def razdel_2(stroki, n, k):
    razdel('2. Что находится ВНУТРИ with db.connect()')
    if n is None:
        return
    bloki = []
    for j in range(n, k):
        s = stroki[j]
        if re.search('with\\s+db\\.connect\\s*\\(|with\\s+connect\\s*\\(', s) and (not s.strip().startswith('#')):
            otstup = len(s) - len(s.lstrip())
            konec = j + 1
            while konec < k:
                t = stroki[konec]
                if t.strip() and len(t) - len(t.lstrip()) <= otstup:
                    break
                konec += 1
            bloki.append((j, konec))
    if not bloki:
        p('  внутри run_ai_draft нет ни одного with db.connect()')
        ITOG['NO AWAIT INSIDE DB TRANSACTION'] = 'PASS (блока нет)'
        return
    est_await = False
    for nachalo, konec in bloki:
        p('')
        p('--- блок соединения: строки %d..%d (длина %d строк) ---' % (nachalo + 1, konec, konec - nachalo))
        ITOG['DB OPEN START'] = 'стр.%d' % (nachalo + 1)
        ITOG['DB CLOSE/COMMIT'] = 'стр.%d (выход из with -> commit)' % konec
        pervyy_sql = posledniy_sql = None
        nayden = []
        for j in range(nachalo, konec):
            s = stroki[j]
            if s.strip().startswith('#') or not s.strip():
                continue
            metki = [imya for rx, imya in DOLGIE if re.search(rx, s)]
            if SQL.search(s):
                if pervyy_sql is None:
                    pervyy_sql = j + 1
                posledniy_sql = j + 1
                metki.append('SQL')
            if metki:
                nayden.append((j + 1, ', '.join(sorted(set(metki))), s.strip()[:150]))
            if 'await' in s and (not s.strip().startswith('#')):
                est_await = True
        for ln, metki, tekst in nayden:
            p('  стр.%-5d [%-28s] %s' % (ln, metki, tekst))
        if pervyy_sql:
            ITOG['FIRST SQL'] = 'стр.%d' % pervyy_sql
            ITOG['LAST SQL'] = 'стр.%d' % posledniy_sql
        trans = [x for x in nayden if 'транскрибация' in x[1]]
        seti = [x for x in nayden if 'сеть/AI' in x[1] or 'телеграм' in x[1] or 'внешний процесс' in x[1]]
        ITOG['TRANSCRIPTION START/END'] = 'стр. %s' % ', '.join((str(x[0]) for x in trans)) if trans else 'внутри соединения нет'
        ITOG['AI/NETWORK START/END'] = 'стр. %s' % ', '.join((str(x[0]) for x in seti)) if seti else 'внутри соединения нет'
    ITOG['NO AWAIT INSIDE DB TRANSACTION'] = 'FAIL — await внутри транзакции есть' if est_await else 'PASS'

def razdel_3():
    razdel('3. Замер по журналу трассировщика (факты, не предположения)')
    t = chitat(ZHURNAL)
    if not t.strip():
        p('  журнала нет или он пуст: %s' % ZHURNAL)
        return
    dlitelnosti = []
    sboi = []
    for s in t.split('\n'):
        ch = s.split('\t')
        if len(ch) < 6:
            continue
        tip, info = (ch[1].strip(), ch[5])
        if not re.search('run_ai_draft|team_bot\\.py:3[5-9]\\d|team_bot\\.py:4\\d\\d', info):
            continue
        try:
            sek = float(ch[2])
        except Exception:
            sek = 0.0
        if tip.startswith('CLOSE'):
            dlitelnosti.append((sek, ch[0], info[:150]))
        if tip == 'COMMIT_FAIL':
            sboi.append((sek, ch[0], info[:170]))
    p('-- закрытия соединений, открытых из этого участка --')
    for sek, kogda, info in sorted(dlitelnosti, reverse=True)[:12]:
        p('  %7.2f сек  %s  %s' % (sek, kogda, info))
    if not dlitelnosti:
        p('  записей нет (журнал начат позже или участок не срабатывал)')
    p('')
    p('-- сбои commit на этом участке --')
    for sek, kogda, info in sboi[-6:]:
        p('  ждал %.2f сек  %s  %s' % (sek, kogda, info))
    if not sboi:
        p('  сбоев не записано')
    if dlitelnosti:
        maks = max((x[0] for x in dlitelnosti))
        ITOG['RUN_AI_DRAFT DB HOLD MAX (замер)'] = '%.2f сек' % maks
    else:
        ITOG['RUN_AI_DRAFT DB HOLD MAX (замер)'] = 'нет данных в журнале'

def trassirovka_obd(k):

    vse_ok = True
    for kod in NOVYE:
        p('')
        p('=== %s ===' % kod)
        stroka = k.crm(kod)
        pole = 'diag_link'
        znach = stroka.get(pole)
        p('%s DB поле        : %s' % (kod, pole))
        p('%s DB значение    : %s' % (kod, znach if znach else 'ПУСТО'))
        p('%s DB тип         : %s' % (kod, type(znach).__name__))
        ist = k.materialy_diagnostiki(kod)
        p('%s BUILDER obd    : %s' % (kod, ist['obd'] if ist['obd'] else 'ПУСТО'))
        vyhod = k.render_obd_link(kod)
        p('%s render input   : %s' % (kod, (ist['obd'] or '')[:60]))
        p('%s render output  : %d знаков, ссылка внутри: %s' % (kod, len(vyhod), 'да' if 'reportview' in vyhod else 'НЕТ'))
        try:
            hd = k.build_diag(kod)
        except Exception as e:
            p('%s build_diag не удался: %r' % (kod, e))
            vse_ok = False
            continue
        syroy, norm = k.obd_iz_stranicy(kod, hd)
        p('%s DIAG HTML raw  : %s' % (kod, syroy or 'НЕТ'))
        p('%s DIAG HTML norm : %s' % (kod, norm or 'НЕТ'))
        p('%s вхождение      : %s' % (kod, 'PASS' if k.obd_na_stranice(kod, hd) else 'FAIL'))
        fayl = chitat(SB_VIDEO + '/' + kod + '-diag.html')
        v_fayle = k.obd_na_stranice(kod, fayl)
        p('%s SANDBOX файл   : %s' % (kod, 'PASS' if v_fayle else 'FAIL'))
        kod_http = 0
        try:
            import urllib.request
            with urllib.request.urlopen(urllib.request.Request(ist['obd'], method='HEAD'), timeout=20) as r:
                kod_http = r.status
        except Exception as e:
            kod_http = repr(e)[:40]
        p('%s LINK TARGET    : HTTP %s' % (kod, kod_http))
        horosho = bool(ist['obd']) and k.obd_na_stranice(kod, hd) and bool(v_fayle)
        vse_ok = vse_ok and horosho
        itog('OBD ' + kod, horosho, 'сквозная трассировка')
    itog('OBD END-TO-END', vse_ok, 'обе карточки')
    return vse_ok

async def auto_catch(msg, card, actor_id, context):
    """Кладёт присланное в открытую карточку само, без кнопок.
    Возвращает True, если разобралось; False — пусть идёт обычным путём."""
    if (context.user_data.get('car_wait') or {}).get('field') in ('price_uah', 'price_georgia'):
        return False
    cid = card['id']
    number = card.get('auto_number') or '#%d' % cid
    if msg.photo or msg.video or msg.video_note or msg.document:
        return False
    text = (msg.text or '').strip()
    if not text:
        return False
    if VIN_RE.search(text):
        return False
    link = _re.search('https?://\\S+', text)
    if link and len(text) < 200:
        set_field(cid, 'diag_link', link.group(0), actor_id)
        await msg.reply_text('Ссылка на отчёт записана в %s.' % number)
        return True
    golo = _re.fullmatch('[\\d\\s]{3,12}\\s*[$₴]?', text)
    if golo:
        ok, answer = apply_value(cid, 'price_uah', text, actor_id)
        if ok:
            await msg.reply_text('%s · цена: %s' % (number, answer.replace('Записано: ', '')), reply_markup=InlineKeyboardMarkup([[InlineKeyboardButton('← Вернуться к карточке', callback_data='car_open:%d' % cid)]]))
            return True
    try:
        import assistant
        if assistant.is_write(text):
            polya = assistant.extract(text, card)
            zapisano = []
            for field, value in (polya or {}).items():
                if field not in dict(EDITABLE) and field not in ('brand', 'model', 'condition_text', 'sea_container', 'diag_link', 'fuel'):
                    continue
                ok, _ = apply_value(cid, field, str(value), actor_id)
                if ok:
                    zapisano.append(LABELS_ALL.get(field, field))
            if zapisano:
                await msg.reply_text('%s · записано: %s' % (number, ', '.join(zapisano)), reply_markup=InlineKeyboardMarkup([[InlineKeyboardButton('← Вернуться к карточке', callback_data='car_open:%d' % cid)]]))
                return True
    except Exception as e:
        log.warning('Запись через ИИ: %s', e)
    try:
        import assistant
        if assistant.is_request(text):
            otvet = assistant.ask(text, card)

def validate_card(kod, html, bylo=None):
    bedy = []
    if not html or len(html) < 2000:
        return ['stranica pustaya ili slishkom korotkaya']
    s = svodka(html)
    if s['dark']:
        bedy.append('staraya temnaya knopka diagnostiki: %d' % s['dark'])
    if est_diagnostika(kod):
        if s['diag_links'] != 1:
            bedy.append('ssylok na diagnostiku %d, dolzhna byt 1' % s['diag_links'])
        if s['gold'] != 1:
            bedy.append('zolotyh CTA %d, dolzhna byt 1' % s['gold'])
    elif s['diag_links']:
        bedy.append('est CTA diagnostiki pri otsutstvii materialov')
    for i, src in enumerate(s['video']):
        if not src:
            bedy.append('video bez src')
            continue
        if not godnoe_video(os.path.join(VIDEO, src.lstrip('/'))):
            bedy.append('video bez fayla: ' + src)
        pos = s['poster'][i] if i < len(s['poster']) else ''
        if not pos:
            bedy.append('net poster u ' + src)
        elif razmer(os.path.join(VIDEO, pos.lstrip('/'))) < 1000:
            bedy.append('poster ne fayl: ' + pos)
    if len(s['video']) != len(set(s['video'])):
        bedy.append('dubli src video')
    if s['chat'] < 1:
        bedy.append('net knopki svyazi')
    if bylo:
        b = svodka(bylo)
        if b['cena'] and b['cena'] != s['cena']:
            bedy.append('cena %s -> %s' % (b['cena'], s['cena']))
        if b['probeg'] and b['probeg'] != s['probeg']:
            bedy.append('probeg %s -> %s' % (b['probeg'], s['probeg']))
        if b['foto'] - s['foto']:
            bedy.append('poteryany foto: %d' % len(b['foto'] - s['foto']))
        poteryano = set((x.split('?')[0] for x in b['video'])) - set((x.split('?')[0] for x in s['video']))
        if poteryano:
            bedy.append('poteryany roliki: %s' % ', '.join(sorted(poteryano)))
    return bedy

def est_materialy(kod, m=None):
    """Активная кнопка только при реальных материалах именно этой машины."""
    if m is None:
        m = dannye(kod)
    if not os.path.exists(os.path.join(VIDEO, '%s-diag.html' % kod)):
        return False
    for pole in ('diag_link', 'diag_text', 'diag_report'):
        if ('%s' % (m.get(pole) or '')).strip():
            return True
    for pole in ('condition_photos', 'condition_videos'):
        v = ('%s' % (m.get(pole) or '')).strip()
        if v and v not in ('[]', 'null', '{}'):
            return True
    papka = os.path.join(VIDEO, 'diag', kod)
    if os.path.isdir(papka):
        for f in os.listdir(papka):
            if not f.startswith('.') and f.lower().endswith(('.jpg', '.jpeg', '.png', '.webp', '.mp4', '.mov', '.webm')):
                return True
    return False

def obrabotat_diagnostiku(html, kod):
    """Страница UA-XXXX-diag.html: если в CRM есть отчёт (diag_link/diag_text),
    а страница показывает заглушку — материалы выводятся клиенту."""
    if not html or 'mcf-diag-block' in html:
        return html
    m = dannye(kod)
    link = ('%s' % (m.get('diag_link') or '')).strip()
    tekst = ('%s' % (m.get('diag_text') or '')).strip()
    if not link and (not tekst):
        return html
    chasti = ['<div class="mcf-diag-block" style="' + _O + '">', '<div style="' + _Z + '">Результаты комплексной диагностики</div>']
    if link:
        chasti.append('<div style="margin:10px 0"><a href="%s" target="_blank" rel="noopener" style="display:inline-block;padding:11px 16px;border-radius:12px;background:rgba(212,175,55,.20);border:1px solid rgba(212,175,55,.55);color:#f4e3ae;font-weight:800;text-decoration:none">Открыть отчёт компьютерной диагностики →</a></div>' % link)
    if tekst:
        bezopasno = tekst.replace('&', '&amp;').replace('<', '&lt;').replace('>', '&gt;')
        chasti.append('<div style="white-space:pre-line;margin-top:8px">%s</div>' % bezopasno)
    chasti.append('</div>')
    blok = ''.join(chasti)
    for yakor in ('Материалы комплексной диагностики пока не добавлены.', 'Материалы диагностики пока не добавлены', 'Данные диагностики пока не добавлены.'):
        if yakor in html:
            return html.replace(yakor, blok, 1)
    if '</body>' in html:
        return html.replace('</body>', blok + '</body>', 1)
    return html + blok

def _ua068_card_errors(source, kod, row):
    errors = []
    if _ua068_re.search('carhistory(?:\\.kr)?|проверить\\s+vin|перевірити\\s+vin|2\\s*200\\s*krw', source, _ua068_re.I):
        errors.append('VIN advertisement forbidden')
    if source.count(_UA068_VIN_START) != 1 or source.count(_UA068_VIN_END) != 1:
        errors.append('VIN Guard blocks != 1')
    errors.extend(_ua068_primary_action_errors(source, kod, row))
    if source.count(_UA068_STAGE_START) != 1 or source.count(_UA068_STAGE_END) != 1:
        errors.append('stage anchors != 1')
    if source.count(_UA068_DIAG) != 1:
        errors.append('diagnostics anchors != 1')
    diag_links = _ua068_re.findall('href=["\\\'](?:[^"\\\']*/)?' + _ua068_re.escape(str(kod)) + '-diag\\.html(?:\\?[^"\\\']*)?["\\\']', source, _ua068_re.I)
    if len(diag_links) != 1:
        errors.append('diagnostics links != 1')
    if _ua068_forbidden_count(source):
        errors.append('forbidden sea wording remains')
    if _ua068_re.search('class=["\\\'][^"\\\']*\\b(?:mcf-etap|mcf-track|mcf-diag-off)\\b[^"\\\']*["\\\']', source, _ua068_re.I):
        errors.append('legacy duplicate UI remains')
    if _ua068_native_stage_span(source) is not None:
        errors.append('native stage duplicate remains')
    vin = str((row or {}).get('vin') or '').strip().upper()
    if not vin or vin not in source:
        errors.append('VIN missing')
    engine = _ua068_engine(row)
    if engine and (not _ua068_re.search(_ua068_re.escape(engine) + '\\s*см³', source)):
        errors.append('engine mismatch')
    stage = _ua068_stage(row)
    if 'data-ua-stage="%d"' % stage not in source:
        errors.append('stage copy mismatch')
    if stage == 4 and ('Выдача и осмотр — Киев' in source or 'Видача та огляд — Київ' in source):
        errors.append('redundant Kyiv issue/inspection copy remains')
    expected_videos = _ua068_video_count(kod, row, source)
    if 'data-ua-video-count="%d"' % expected_videos not in source:
        errors.append('video count mismatch')
    return errors

def obd_iz_crm(kod):
    try:
        c = sqlite3.connect('file:%s?mode=ro' % BAZA, uri=True)
        kol = [r[1] for r in c.execute('PRAGMA table_info(cars)')]
        nom = 'auto_number' if 'auto_number' in kol else kol[0]
        r = list(c.execute('SELECT * FROM cars WHERE %s=?' % nom, (kod,)))
        c.close()
        if not r:
            return ''
        d = dict(zip(kol, r[0]))
        return (d.get('diag_link') or '').strip()
    except Exception:
        return ''

def _validate_primary(code: str, row: dict[str, Any], source: str) -> dict[str, Any]:
    if len(source.encode('utf-8')) < 5000 or code not in source or '</html>' not in source.lower():
        raise PublishError('PRIMARY_HTML_INVALID:' + code)
    links = re.findall('href\\s*=\\s*[\'\\"](?:[^\'\\"]*/)?' + re.escape(code) + '-diag\\.html(?:\\?[^\'\\"]*)?[\'\\"]', source, re.I)
    if len(links) != 1:
        raise PublishError('PRIMARY_DIAGNOSTIC_LINK_COUNT:%s:%d' % (code, len(links)))
    foreign = {value.upper() for value in re.findall('UA-[0-9]{4,}', source, re.I) if value.upper() != code}
    if foreign:
        raise PublishError('PRIMARY_FOREIGN_IDENTIFIER:%s:%s' % (code, ','.join(sorted(foreign))))
    stage = _stage(row)
    stage_tokens = ('data-ua-stage-current="%d"' % stage, "data-ua-stage-current='%d'" % stage, 'data-ua-stage="%d"' % stage, "data-ua-stage='%d'" % stage)
    if not any((token in source for token in stage_tokens)):
        raise PublishError('PRIMARY_STAGE_MISMATCH:%s:%d' % (code, stage))
    return {'bytes': len(source.encode('utf-8')), 'diag_links': 1, 'stage': stage}

def _verify_bundle(codes: list[str]) -> dict[str, Any]:
    rows, digest = _row_map()
    targets: dict[str, Any] = {}
    for code in codes:
        row = rows.get(code)
        if row is None:
            raise PublishError('TARGET_NOT_PUBLISHED:' + code)
        files: dict[str, Any] = {}
        for root in ROOTS:
            primary_path = root / (code + '.html')
            diag_path = root / (code + '-diag.html')
            if not primary_path.is_file() or not diag_path.is_file():
                raise PublishError('TARGET_FILE_MISSING:%s:%s' % (code, root))
            primary_text = _read(primary_path).decode('utf-8', 'replace')
            diag_text = _read(diag_path).decode('utf-8', 'replace')
            if code not in diag_text or '</html>' not in diag_text.lower():
                raise PublishError('DIAGNOSTIC_READBACK_INVALID:%s:%s' % (code, root))
            files[str(root)] = {'primary_sha256': _sha(_read(primary_path)), 'diag_sha256': _sha(_read(diag_path)), 'primary': _validate_primary(code, row, primary_text)}
        targets[code] = {'status': row.get('status'), 'stage': _stage(row), 'category': _category(_stage(row)), 'files': files}
    catalogs: dict[str, Any] = {}
    for root in ROOTS:
        path = root / 'katalog.html'
        catalogs[str(path)] = _validate_catalog(_read(path).decode('utf-8', 'replace'), rows)
    return {'targets': targets, 'catalogs': catalogs, 'published_rows_sha256': digest}

def register_pokaz(app):
    """Вешает /pokaz на CRM-бота."""
    from telegram.ext import CommandHandler, CallbackQueryHandler
    from telegram import InlineKeyboardButton, InlineKeyboardMarkup
    if getattr(app, '_ns_pokaz', False):
        return
    app._ns_pokaz = True
    cols = set(_car_columns())
    have = [c for c in CAR_FIELDS if c in cols]

    async def cmd(update, context):
        rows = _rows('SELECT rowid, %s FROM cars ORDER BY rowid DESC LIMIT 30' % ', '.join(have))
        if not rows:
            await update.message.reply_text('Карточек пока нет.')
            return
        buttons = []
        for car in rows:
            name = ' '.join((str(x) for x in [car.get('auto_number'), car.get('brand'), car.get('model')] if x)) or 'Карточка %s' % car['rowid']
            buttons.append([InlineKeyboardButton(name[:60], callback_data='nspokaz:%d' % car['rowid'])])
        await update.message.reply_text('Как видит покупатель — выбери карточку:', reply_markup=InlineKeyboardMarkup(buttons))

    async def press(update, context):
        q = update.callback_query
        await q.answer()
        rowid = int(q.data.split(':')[1])

def _ua068_card_errors(source, kod, row):
    errors = []
    if _ua068_re.search('carhistory(?:\\.kr)?|проверить\\s+vin|перевірити\\s+vin|2\\s*200\\s*krw', source, _ua068_re.I):
        errors.append('VIN advertisement forbidden')
    if source.count(_UA068_VIN_START) != 1 or source.count(_UA068_VIN_END) != 1:
        errors.append('VIN Guard blocks != 1')
    errors.extend(_ua068_primary_action_errors(source, kod, row))
    if source.count(_UA068_STAGE_START) != 1 or source.count(_UA068_STAGE_END) != 1:
        errors.append('stage anchors != 1')
    if source.count(_UA068_DIAG) != 1:
        errors.append('diagnostics anchors != 1')
    diag_links = _ua068_re.findall('href=["\\\'](?:[^"\\\']*/)?' + _ua068_re.escape(str(kod)) + '-diag\\.html(?:\\?[^"\\\']*)?["\\\']', source, _ua068_re.I)
    if len(diag_links) != 1:
        errors.append('diagnostics links != 1')
    if _ua068_forbidden_count(source):
        errors.append('forbidden sea wording remains')
    if _ua068_re.search('class=["\\\'][^"\\\']*\\b(?:mcf-etap|mcf-track|mcf-diag-off)\\b[^"\\\']*["\\\']', source, _ua068_re.I):
        errors.append('legacy duplicate UI remains')
    if _ua068_native_stage_span(source) is not None:
        errors.append('native stage duplicate remains')
    vin = str((row or {}).get('vin') or '').strip().upper()
    if not vin or vin not in source:
        errors.append('VIN missing')
    engine = _ua068_engine(row)
    if engine and (not _ua068_re.search(_ua068_re.escape(engine) + '\\s*см³', source)):
        errors.append('engine mismatch')
    stage = _ua068_stage(row)
    if 'data-ua-stage="%d"' % stage not in source:
        errors.append('stage copy mismatch')
    if stage == 4 and ('Выдача и осмотр — Киев' in source or 'Видача та огляд — Київ' in source):
        errors.append('redundant Kyiv issue/inspection copy remains')
    expected_videos = _ua068_video_count(kod, row, source)
    if 'data-ua-video-count="%d"' % expected_videos not in source:
        errors.append('video count mismatch')
    return errors

async def run_ai_draft(update, context, inbox_id, kind, text, file_id):
    """Open a schema-closed CRM draft from photo, text, or voice."""
    import asyncio
    import time
    import ai_fast_schema as fast
    import local_ocr
    msg = update.message
    if ai.is_stopped():
        await msg.reply_text('🛑 ИИ остановлен. Включите ИИ в главном меню.')
        return
    started = time.monotonic()
    hard_deadline = started + 15.0
    data = {}
    thinking = None
    allowed = fast.car_fields(ai_filter)

    async def download_bytes(fid):
        tg_file = await context.bot.get_file(fid)
        return bytes(await tg_file.download_as_bytearray())
    if kind in ('voice', 'audio', 'video_note') and file_id:
        thinking = await msg.reply_text('🎤 ИИ распознаёт голосовое...')
        if not ai.voice_enabled():
            await thinking.edit_text('🎤 Расшифровка голоса не настроена.')
            return
        try:
            remaining = max(0.1, hard_deadline - time.monotonic())
            audio = await asyncio.wait_for(download_bytes(file_id), timeout=min(3.0, remaining))
            import crm_voice_watchdog as _v178_voice
            _v184_voice = msg.voice or msg.audio or msg.video_note
            _v184_filename = 'voice.ogg' if msg.voice else 'audio.mp3' if msg.audio else 'video_note.mp4'
            _v184_result = await _v178_voice.transcribe_with_restart(audio, _v184_filename, getattr(_v184_voice, 'duration', 0))
            text = _v184_result.text if _v184_result.ok else ''
            if not _v184_result.ok:
                logging.warning('AI voice worker failed inbox=%s attempts=%s restarts=%s error=%s', inbox_id, _v184_result.attempts, _v184_result.restarted_workers, _v184_result.error)
        except asyncio.TimeoutError:
            text = ''
            logging.warning('AI voice hard timeout inbox=%s', inbox_id)
        except Exception as exc:
            text = ''
            logging.error('Voice recognition failed inbox=%s error=%s', inbox_id, exc)
        if text:
            data.update(local_ocr.fields_from_text(text, allowed))
            data.update(fast.fast_text_data(text, ai_filter))
    elif kind in ('photo', 'document') and file_id:
        if kind == 'document':
            mime = msg.document.mime_type or '' if msg.document else ''
            if not mime.startswith('image/'):
                await msg.reply_text('📎 Нужен файл-изображение.')
                return
        if msg.media_group_id:
            marker = 'ai_mg_%s' % msg.media_group_id
            if context.chat_data.get(marker):
                return
            context.chat_data[marker] = True
        thinking = await msg.reply_text('🖼 ИИ распознаёт, до 15 секунд...')
        try:
            remaining = max(0.1, hard_deadline - time.monotonic())
            image = await asyncio.wait_for(download_bytes(file_id), timeout=remaining)
            remaining = max(0.5, hard_deadline - time.monotonic())
            image_data = await asyncio.wait_for(asyncio.to_thread(local_ocr.fields_from_image, image, allowed, remaining), timeout=remaining)
            data.update(image_data)
            if (text or '').strip():
                data.update(local_ocr.fields_from_text(text, allowed))
                data.update(fast.fast_text_data(text, ai_filter))
        except asyncio.TimeoutError:
            data = {}
            logging.warning('Local OCR hard timeout inbox=%s', inbox_id)
        except Exception as exc:
            data = {}
            logging.error('Local OCR failed inbox=%s error=%s', inbox_id, exc)
    elif (text or '').strip():
        thinking = await msg.reply_text('🧠 ИИ распознаёт, до 15 секунд...')
        data.update(local_ocr.fields_from_text(text, allowed))
        data.update(fast.fast_text_data(text, ai_filter))
    if data:
        parsed = fast.parsed_from_data(data)
        data = fast.clean_car(parsed, ai_filter, '')
    if data:
        parsed = fast.parsed_from_data(data)
        context.user_data['ai_draft'] = {'inbox_id': inbox_id, 'parsed': parsed, 'clean_data': data, 'source_text': '', 'review': None}
        rows = [[InlineKeyboardButton('✅ Сохранить карточку', callback_data='ai_ok')], [InlineKeyboardButton('✏️ Дополнить', callback_data='ai_edit'), InlineKeyboardButton('❌ Отменить', callback_data='ai_no')]]
        elapsed = time.monotonic() - started
        if thinking is None:
            thinking = await msg.reply_text('🧠 ИИ распознаёт, до 15 секунд...')
        await thinking.edit_text('✅ Новая карточка открыта\n\n' + ai_filter.render(data) + '\n\n⏱ %.1f с · локально · 0 AI-токенов' % elapsed, parse_mode='HTML', reply_markup=InlineKeyboardMarkup(rows))
        return
    if thinking is None:
        thinking = await msg.reply_text('🧠 ИИ распознаёт, до 15 секунд...')
    await thinking.edit_text('Поля CRM не распознаны. Пришлите более чёткое фото, текст или голос.')

def _ua068_card_errors(source, kod, row):
    errors = []
    if _ua068_re.search('carhistory(?:\\.kr)?|проверить\\s+vin|перевірити\\s+vin|2\\s*200\\s*krw', source, _ua068_re.I):
        errors.append('VIN advertisement forbidden')
    if source.count(_UA068_VIN_START) != 1 or source.count(_UA068_VIN_END) != 1:
        errors.append('VIN Guard blocks != 1')
    errors.extend(_ua068_primary_action_errors(source, kod, row))
    if source.count(_UA068_STAGE_START) != 1 or source.count(_UA068_STAGE_END) != 1:
        errors.append('stage anchors != 1')
    if source.count(_UA068_DIAG) != 1:
        errors.append('diagnostics anchors != 1')
    diag_links = _ua068_re.findall('href=["\\\'](?:[^"\\\']*/)?' + _ua068_re.escape(str(kod)) + '-diag\\.html(?:\\?[^"\\\']*)?["\\\']', source, _ua068_re.I)
    if len(diag_links) != 1:
        errors.append('diagnostics links != 1')
    if _ua068_forbidden_count(source):
        errors.append('forbidden sea wording remains')
    if _ua068_re.search('class=["\\\'][^"\\\']*\\b(?:mcf-etap|mcf-track|mcf-diag-off)\\b[^"\\\']*["\\\']', source, _ua068_re.I):
        errors.append('legacy duplicate UI remains')
    if _ua068_native_stage_span(source) is not None:
        errors.append('native stage duplicate remains')
    vin = str((row or {}).get('vin') or '').strip().upper()
    if not vin or vin not in source:
        errors.append('VIN missing')
    engine = _ua068_engine(row)
    if engine and (not _ua068_re.search(_ua068_re.escape(engine) + '\\s*см³', source)):
        errors.append('engine mismatch')
    stage = _ua068_stage(row)
    if 'data-ua-stage="%d"' % stage not in source:
        errors.append('stage copy mismatch')
    if stage == 4 and ('Выдача и осмотр — Киев' in source or 'Видача та огляд — Київ' in source):
        errors.append('redundant Kyiv issue/inspection copy remains')
    expected_videos = _ua068_video_count(kod, row, source)
    if 'data-ua-video-count="%d"' % expected_videos not in source:
        errors.append('video count mismatch')
    return errors
