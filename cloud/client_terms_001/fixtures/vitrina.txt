def mashiny():
    put = os.path.join(BAZA_DIR, "crm.db")
    con = sqlite3.connect("file:%s?mode=ro" % put, uri=True)
    con.row_factory = sqlite3.Row
    try:
        stroki = [dict(r) for r in con.execute("SELECT * FROM cars WHERE published=1 ORDER BY id")]
    finally:
        con.close()
    return stroki


def mashina(nomer_id):
    for m in mashiny():
        if str(m.get("id")) == str(nomer_id):
            return m
    return None


def nomer(m):
    return (m.get("auto_number") or "UA-%04d" % int(m.get("id") or 0)).strip()


def spisok(pole):
    if not pole:
        return []
    try:
        z = json.loads(pole) if isinstance(pole, str) else pole
    except Exception:
        return []
    if not isinstance(z, list):
        return []
    itog = []
    for x in z:
        if isinstance(x, dict):
            x = x.get("file_id") or x.get("id") or x.get("path") or ""
        if x:
            itog.append(str(x))
    return itog


def sklonenie(n):
    if n % 10 == 1 and n % 100 != 11:
        return "день"
    if n % 10 in (2, 3, 4) and n % 100 not in (12, 13, 14):
        return "дня"
    return "дней"


def srok(m):
    """Дни считаются с момента, когда владелец перевёл статус."""
    status = (m.get("status") or "").lower()
    if status.startswith("ua_"):
        return "Машина в Киеве — можно приехать и посмотреть"
    if status == "ge_to_kyiv":
        try:
            pamyat = json.loads(chitat(os.path.join(BAZA_DIR, ".etapy.json")))
            s_kogda = (pamyat.get(nomer(m)) or {}).get("s_kogda")
        except Exception:
            s_kogda = None
        if s_kogda:
            ostalos = SROK_DNEY - int((time.time() - s_kogda) // 86400)
            if ostalos > 0:
                return "До прибытия в Киев — %d %s" % (ostalos, sklonenie(ostalos))
            return "Прибытие в Киев ожидается со дня на день"
        return "В Киеве через %d дней" % SROK_DNEY
    return "Срок прибытия — %d дней после оплаты" % SROK_DNEY


def cena(m):
    c = m.get("price_uah") or m.get("price_total") or 0
    try:
        c = int(float(c))
    except Exception:
        c = 0
    if not c:
        return "цена по запросу"
    krasivo = "{:,}".format(c).replace(",", " ")
    return "%s $" % krasivo if VALYUTA == "USD" else "%s ₴" % krasivo


def etap(m):
    s = (m.get("status") or "").lower()
    if s.startswith("sold"):
        return "уже продана · в пути", False
    if s.startswith("kr_"):
        return "Выкуплена · ожидает погрузки в Корее", False
    if s.startswith("sea_"):
        return "В море · идёт из Кореи", False
    if s == "ge_to_kyiv":
        return "Выехала из Грузии · едет в Киев", True
    if s.startswith("ge_"):
        return "В Грузии · можно забронировать", True
    if s.startswith("ua_"):
        return "В Киеве · можно посмотреть", True
    return "В работе", False


def zagolovok(m):
    return " ".join(str(x) for x in [m.get("brand"), m.get("model"), m.get("year")] if x)


def korotko(m):
    kuski = []
    if m.get("engine_cc"):
        kuski.append("%s см³" % m["engine_cc"])
    if m.get("fuel"):
        kuski.append(str(m["fuel"]))
    if m.get("gearbox"):
        kuski.append(str(m["gearbox"]))
    if m.get("drive"):
        kuski.append(str(m["drive"]))
    if m.get("mileage_km"):
        try:
            kuski.append("{:,} км".format(int(m["mileage_km"])).replace(",", " "))
        except Exception:
            pass
    return " · ".join(kuski)


def oblozhka(m):
    """Сначала кадр, собранный на диске (его Telegram точно покажет), потом file_id."""
    papka = os.path.join(PAPKA_VID, "foto", nomer(m))
    try:
        kadry = sorted(f for f in os.listdir(papka) if f.lower().endswith(".jpg"))
        if kadry:
            return "%s/foto/%s/%s" % (sajt(), nomer(m), kadry[0])
    except Exception:
        pass
    if m.get("cover_photo"):
        return m.get("cover_photo")
    spisok_kadrov = spisok(m.get("photos"))
    return spisok_kadrov[0] if spisok_kadrov else None

def zapis_strany(kto, chat_id, strana, dannye):
    """Заявка на подбор пишется в crm.db — таблица podbor_zayavki."""
    try:
        import sqlite3
        b = sqlite3.connect("/home/Carix/crm.db", timeout=10)
        b.execute("CREATE TABLE IF NOT EXISTS podbor_zayavki ("
                  "id INTEGER PRIMARY KEY AUTOINCREMENT, kogda TEXT, chat_id TEXT, "
                  "kto TEXT, strana TEXT, byudzhet TEXT, kuzov TEXT, marka TEXT)")
        b.execute("INSERT INTO podbor_zayavki "
                  "(kogda, chat_id, kto, strana, byudzhet, kuzov, marka) "
                  "VALUES (?,?,?,?,?,?,?)",
                  (time.strftime("%Y-%m-%d %H:%M:%S"), str(chat_id), str(kto), strana,
                   str(dannye.get("b") or ""), str(dannye.get("k") or ""),
                   str(dannye.get("m") or "")))
        b.commit()
        b.close()
    except Exception as e:
        try:
            zapis("strana: %s" % e)
        except Exception:
            pass

async def poslat(bot, chat_id, tekst, knopki=None, foto=None):
    from telegram import InlineKeyboardMarkup, InlineKeyboardButton, WebAppInfo
    ryady = []
    for ryad in (knopki or []):
        sobran = []
        for nadpis, cel in ryad:
            if cel.startswith("http") and ".html" in cel:
                sobran.append(InlineKeyboardButton(nadpis, web_app=WebAppInfo(url=cel)))
            elif cel.startswith("http"):
                sobran.append(InlineKeyboardButton(nadpis, url=cel))
            else:
                sobran.append(InlineKeyboardButton(nadpis, callback_data=cel))
        if sobran:
            ryady.append(sobran)
    razmetka = InlineKeyboardMarkup(ryady) if ryady else None
    if foto:
        try:
            return await bot.send_photo(chat_id, photo=foto, caption=tekst,
                                        parse_mode="HTML", reply_markup=razmetka)
        except Exception:
            pass
    return await bot.send_message(chat_id, tekst, parse_mode="HTML",
                                  reply_markup=razmetka, disable_web_page_preview=True)


async def pokazat_katalog(bot, chat_id):
    spisok_mashin = mashiny()
    if not spisok_mashin:
        await poslat(bot, chat_id,
                     "Сейчас в наличии нет машин, но мы привозим под заказ из Кореи "
                     "за 45–60 дней — с фиксированной ценой в договоре.",
                     [[("Подобрать под заказ", "v_podbor")], [("Назад", "v_start")]])
        return

    await poslat(bot, chat_id, "<b>Авто в наличии и в пути — %d</b>" % len(spisok_mashin))

    if len(spisok_mashin) <= PLITKOJ:
        for m in spisok_mashin:
            etap_tekst, _ = etap(m)
            podpis = ("<i>%s</i>\n<b>%s</b>\n<b>%s</b> под ключ в Киеве\n• %s"
                      % (nomer(m), zagolovok(m), cena(m), etap_tekst))
            knopki = [[("Открыть карточку", "v_car:%s" % m.get("id"))]]
            await poslat(bot, chat_id, podpis, knopki, foto=oblozhka(m))
    else:
        for m in spisok_mashin:
            etap_tekst, _ = etap(m)
            await poslat(bot, chat_id,
                         "<b>%s</b> · %s\n%s" % (zagolovok(m), cena(m), etap_tekst),
                         [[("Открыть карточку", "v_car:%s" % m.get("id"))]])
    await poslat(bot, chat_id, "Нужно другое — подберём под заказ из Кореи.",
                 [[("Подобрать под заказ", "v_podbor")], [("Назад", "v_start")]])

async def obrabotat(app, update):
    bot = app.bot
    zapros = getattr(update, "callback_query", None)

    if zapros is not None:
        dannye = zapros.data or ""
        if not dannye.startswith("v_"):
            return False
        chat_id = zapros.message.chat.id
        kto = zapros.from_user.id
        try:
            await bot.answer_callback_query(zapros.id)
        except Exception:
            pass

        # UA-ANALIT-KNOPKA-V2 · раздел аналитики
        if dannye == "v_analytics" or dannye.startswith("v_an_"):
            try:
                import analitika_knopka
                if await analitika_knopka.vetka(bot, chat_id, kto, dannye):
                    return True
            except Exception as _e_an:
                zapis("аналитика: %s" % _e_an)

        if dannye == "v_start":
            t, k = ekran_start(kto); await poslat(bot, chat_id, t, k)
        elif dannye == "v_cars":
            await pokazat_katalog(bot, chat_id)
        elif dannye == "v_kak":
            t, k = ekran_kak(); await poslat(bot, chat_id, t, k)
        elif dannye == "v_kontakty":
            t, k = ekran_kontakty(); await poslat(bot, chat_id, t, k)
        elif dannye.startswith("v_car:"):
            m = mashina(dannye.split(":", 1)[1])
            if m:
                await pokazat_mashinu(bot, chat_id, m)
        elif dannye.startswith("v_bron:"):
            m = mashina(dannye.split(":", 1)[1])
            if m:
                t, k = ekran_bron(m)
                await poslat(bot, chat_id, t, k)
                vladelec = fajl("vladelec.txt")
                if vladelec:
                    try:
                        await bot.send_message(int(vladelec),
                                               "🔔 <b>Бронь</b>\n%s · %s\nКлиент id %s"
                                               % (zagolovok(m), nomer(m), kto),
                                               parse_mode="HTML")
                    except Exception:
                        pass
        elif dannye == "v_podbor":
            await podbor_shag(bot, chat_id, kto, {})
        elif dannye.startswith("v_pb1:"):
            d = _podbor.get(kto, {})
            d["byudzhet"] = BYUDZHET[int(dannye.split(":", 1)[1])]
            await podbor_shag(bot, chat_id, kto, d)
        elif dannye.startswith("v_pb2:"):
            d = _podbor.get(kto, {})
            d["kuzov"] = KUZOV[int(dannye.split(":", 1)[1])]
            await podbor_shag(bot, chat_id, kto, d)
        else:
            return False
        return True

    soobshchenie = getattr(update, "message", None)
    if soobshchenie is None:
        return False

    if getattr(soobshchenie, "web_app_data", None):
        return await prinyat_iz_prilozheniya(bot, soobshchenie)
    chat_id = soobshchenie.chat.id
    kto = soobshchenie.from_user.id if soobshchenie.from_user else 0
    tekst = (soobshchenie.text or "").strip()
    # LEAD-FOLDERS-001: a step of the lead desk (note, contract number) waits for this message
    try:
        import ua_lead_desk
        if ua_lead_desk.pending(app, kto):
            return False
    except Exception as e:
        zapis("ua_lead_desk: %s" % e)

    if tekst.startswith("/start") or tekst.startswith("/help"):
        chasti = tekst.split(None, 1)
        parametr = chasti[1].strip() if len(chasti) > 1 else ""
        if parametr.startswith("bron_") or parametr.startswith("vopros_"):
            nom = parametr.split("_", 1)[1]
            nuzhna = None
            for m in mashiny():
                if nomer(m) == nom:
                    nuzhna = m
                    break
            if nuzhna is not None:
                await privetstvie(bot, chat_id)
                if parametr.startswith("bron_"):
                    tk, kn = ekran_bron(nuzhna)
                    await poslat(bot, chat_id, tk, kn)
                    vladelec = fajl("vladelec.txt")
                    if vladelec:
                        try:
                            await bot.send_message(
                                int(vladelec),
                                "🔔 <b>Бронь со страницы</b>\n%s · %s\nКлиент id %s"
                                % (zagolovok(nuzhna), nom, kto), parse_mode="HTML")
                        except Exception:
                            pass
                else:
                    await poslat(bot, chat_id,
                                 "<b>%s · %s</b>\n\nНапишите ваш вопрос сюда — "
                                 "менеджер Артём ответит в этом чате."
                                 % (zagolovok(nuzhna), nom),
                                 [[("Открыть карточку", "%s/%s.html?v=%d"
                                    % (sajt(), nom, int(time.time())))]])
                return True
        if parametr.startswith("z_"):
            kody = {"b1": "до 15 000 $", "b2": "15 000 – 25 000 $",
                    "b3": "25 000 – 40 000 $", "b4": "больше 40 000 $",
                    "k1": "седан", "k2": "кроссовер", "k3": "хэтчбек",
                    "k4": "электро/гибрид"}
            chasti_zayavki = parametr.split("_")[1:]
            byudzhet = kody.get(next((x for x in chasti_zayavki if x.startswith("b")), ""), "—")
            kuzov = kody.get(next((x for x in chasti_zayavki if x.startswith("k")), ""), "—")
            marka = next((x for x in chasti_zayavki
                          if not x.startswith("b") and not x.startswith("k")), "")
            await privetstvie(bot, chat_id)
            await poslat(bot, chat_id,
                         "<b>Заявка на подбор принята</b>\n\n"
                         "Бюджет: %s\nКузов: %s\nМарка: %s\n\n"
                         "Менеджер Артём подберёт варианты с аукционов Кореи и пришлёт их сюда. "
                         "Обычно это занимает до суток.\n\n"
                         "<i>Напишите сюда номер телефона, если удобнее по звонку.</i>"
                         % (byudzhet, kuzov, marka or "не важно"),
                         [[("Посмотреть авто в наличии", "v_cars")],
                          [("Написать в WhatsApp", "[URL]" % WHATSAPP)]])
            vladelec = fajl("vladelec.txt")
            if vladelec:
                try:
                    imya_klienta = (soobshchenie.from_user.full_name
                                    if soobshchenie.from_user else "клиент")
                    await bot.send_message(
                        int(vladelec),
                        "🔔 <b>Заявка на подбор</b>\n\nОт: %s (id %s)\n"
                        "Бюджет: %s\nКузов: %s\nМарка: %s"
                        % (imya_klienta, kto, byudzhet, kuzov, marka or "не важно"),

                    zapis("заявка владельцу не ушла: %s" % e)

        if parametr == "podbor":
            await privetstvie(bot, chat_id)
            await podbor_shag(bot, chat_id, kto, {})
            return True
        tk, kn = ekran_start(kto)
        await poslat(bot, chat_id, tk, kn)
        await klaviatura_snizu(bot, chat_id)
        return True
    if tekst.startswith("/cars") or tekst.startswith("/mycar"):
        await pokazat_katalog(bot, chat_id)
        return True
    if tekst.startswith("/terms"):
        t, k = ekran_kak(); await poslat(bot, chat_id, t, k)
        return True

    if kto in _podbor and tekst and not tekst.startswith("/"):
        imya = (soobshchenie.from_user.full_name if soobshchenie.from_user else "клиент")
        await zayavka(bot, chat_id, kto, imya, tekst)

    if tekst and not tekst.startswith("/"):

        vladelec = fajl("vladelec.txt")
        if vladelec:
            try:
                await bot.send_message(
                    int(vladelec),
                    "🔔 <b>Сообщение из бота</b>\n\nОт: %s (id %s)\n\n%s"
                    % (imya, kto, tekst),
                    parse_mode="HTML")
            except Exception as e:

        await poslat(bot, chat_id,
                     "<b>Сообщение передано</b>\n\n"
                     "Менеджер Артём ответит в этом чате.",
                     [[("Написать в WhatsApp", "[URL]" % WHATSAPP)]])

def vklyuchit_zamenu():
    """Старые текстовые простыни клиентского бота заменяем на короткое
    сообщение с кнопкой на дизайн-страницу."""
    try:
        from telegram.ext import ExtBot
        from telegram import InlineKeyboardMarkup, InlineKeyboardButton, WebAppInfo
    except Exception:
        return False
    if getattr(ExtBot, "_zamena_vitriny", False):
        return True

    klientskij = fajl("bot_token.txt")
    staryj = ExtBot.send_message

    async def novyj(self, *args, **kw):
        try:
            if klientskij and getattr(self, "token", "") == klientskij:
                tekst = kw.get("text") or ""
                if not tekst:
                    for a in args:
                        if isinstance(a, str) and len(a) > 30:
                            tekst = a
                            break
                if tekst and sum(1 for m in MARKERY_STAROGO if m in tekst) >= 2:
                    najdeno = re.search(r"\bUA-\d{3,5}\b", tekst)
                    nuzhna = None
                    if najdeno:
                        for m in mashiny():
                            if nomer(m) == najdeno.group(0):
                                nuzhna = m
                                break
                    if nuzhna is not None:
                        nom = nomer(nuzhna)
                        korotkij, _ = tekst_mashiny(nuzhna)
                        ryady = [[InlineKeyboardButton(
                            "🚗  Открыть карточку",
                            web_app=WebAppInfo(url="%s/%s.html" % (sajt(), nom)))]]
                        if stranica_est(nuzhna):
                            ryady.append([InlineKeyboardButton(
                                "Все машины", callback_data="v_cars")])
                        kw["text"] = korotkij
                        kw["parse_mode"] = "HTML"
                        kw["reply_markup"] = InlineKeyboardMarkup(ryady)
                        chat = kw.get("chat_id")
                        if chat is None and args:
                            chat = args[0]
                        zapis("%s: старый экран заменён коротким" % nom)
                        return await staryj(self, chat_id=chat, **{
                            k: v for k, v in kw.items() if k != "chat_id"})
        except Exception as e:
            zapis("замена экрана: %s" % e)
        return await staryj(self, *args, **kw)

    ExtBot.send_message = novyj
    ExtBot._zamena_vitriny = True
    zapis("замена старых экранов включена")
    return True


def vklyuchit():
    try:
        from telegram.ext import Application
    except Exception:
        return False
    vklyuchit_zamenu()
    if getattr(Application, "_vitrina", False):
        return True

    klientskij = fajl("bot_token.txt")
    staryj = Application.process_update

    async def novyj(self, update):
        try:
            if klientskij and getattr(self.bot, "token", "") == klientskij:
                if await obrabotat(self, update):
                    return
        except Exception as e:
            zapis("разбор: %s" % e)
        return await staryj(self, update)

    Application.process_update = novyj
    Application._vitrina = True
    zapis("витрина по макету включена")
    return True
