def who(update: Update):
    """Возвращает запись сотрудника или None. Владелец создаётся сам."""
    user = update.effective_user
    if user.id == OWNER_ID:
        db.ensure_owner(user.id, user.full_name, user.username or "")
    staff = db.get_staff(user.id)
    if staff and not staff["active"]:
        return None
    return staff

async def ai_stop_cb(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Останавливает ИИ, но не трогает доступ к базе, карточкам и журналу."""
    query = update.callback_query
    staff = who(update)
    if not is_admin_or_owner(staff):
        await query.answer("Недостаточно прав", show_alert=True)
        return
    await query.answer("ИИ остановлен")
    ai.set_stopped(True, staff.get("full_name") or staff["user_id"])
    db.log_action(staff["user_id"], "ai_stop", "system", None, "ai", "on", "off")

    # Незавершённые черновики не удаляем — они переходят в приостановленные
    context.user_data.pop("ai_draft", None)

    await query.message.reply_text(
        "🛑 <b>ИИ остановлен.</b>\n\n"
        "Новые сообщения будут просто складываться в «Необработанное» "
        "без разбора. Незавершённые черновики отброшены, исходные записи целы.\n\n"
        "Доступ к базе, карточкам, поиску и журналу работает как обычно.",
        parse_mode="HTML",
        reply_markup=main_menu(staff),
    )
    # Сообщаем остальным, чтобы не гадали, почему бот молчит
    for uid in set(db.staff_ids_by_role(db.ROLE_OWNER) + db.staff_ids_by_role(db.ROLE_ADMIN)):
        if uid == staff["user_id"]:
            continue
        try:
            await context.bot.send_message(
                uid, f"🛑 ИИ остановлен: {staff.get('full_name') or staff['user_id']}"
            )
        except Exception:
            pass

async def try_invite_code(update: Update, context: ContextTypes.DEFAULT_TYPE) -> bool:
    """Незарегистрированный пользователь прислал текст — пробуем как код."""
    user = update.effective_user
    text = (update.message.text or "").strip()
    if len(text) > 12:
        return False
    role = db.use_invite(text, user.id, user.full_name, user.username or "")
    if not role:
        return False
    staff = db.get_staff(user.id)
    await update.message.reply_text(
        f"✅ Доступ открыт.\nВаша роль: {db.ROLE_LABELS.get(role, role)}"
    )
    await show_menu(update, context, staff)
    # Сообщаем владельцу, что человек вошёл
    for oid in db.staff_ids_by_role(db.ROLE_OWNER):
        try:
            await context.bot.send_message(
                oid,
                f"👤 Новый сотрудник в системе:\n"
                f"{user.full_name} (@{user.username or 'без username'})\n"
                f"Роль: {db.ROLE_LABELS.get(role, role)}",
            )
        except Exception:
            pass
    return True

def detect_kind(msg) -> tuple:
    """Определяем тип вложения и достаём file_id."""
    if msg.photo:
        return "photo", msg.photo[-1].file_id
    if msg.video:
        return "video", msg.video.file_id
    if msg.voice:
        return "voice", msg.voice.file_id
    if msg.audio:
        return "audio", msg.audio.file_id
    if msg.document:
        return "document", msg.document.file_id
    if msg.video_note:
        return "video_note", msg.video_note.file_id
    if msg.contact:
        return "contact", None
    if msg.location:
        return "location", None
    return "text", None

async def run_ai_draft(update, context, inbox_id, kind, text, file_id):
    """Open a schema-closed CRM draft from photo, text, or voice."""
    import asyncio
    import time
    import ai_fast_schema as fast
    import local_ocr

    msg = update.message
    if ai.is_stopped():
        await msg.reply_text("🛑 ИИ остановлен. Включите ИИ в главном меню.")
        return

    started = time.monotonic()
    hard_deadline = started + 15.0
    data = {}
    thinking = None
    allowed = fast.car_fields(ai_filter)

    async def download_bytes(fid):
        tg_file = await context.bot.get_file(fid)
        return bytes(await tg_file.download_as_bytearray())

    if kind in ("voice", "audio", "video_note") and file_id:
        thinking = await msg.reply_text("🎤 ИИ распознаёт голосовое...")
        if not ai.voice_enabled():
            await thinking.edit_text("🎤 Расшифровка голоса не настроена.")
            return
        try:
            remaining = max(0.1, hard_deadline - time.monotonic())
            audio = await asyncio.wait_for(download_bytes(file_id), timeout=min(3.0, remaining))
            # UA-TASK084-KILLABLE-DRAFT-VOICE
            import crm_voice_watchdog as _v178_voice
            _v184_voice = msg.voice or msg.audio or msg.video_note
            _v184_filename = (
                "voice.ogg" if msg.voice else
                "audio.mp3" if msg.audio else "video_note.mp4"
            )
            _v184_result = await _v178_voice.transcribe_with_restart(
                audio,
                _v184_filename,
                getattr(_v184_voice, "duration", 0),
            )
            text = _v184_result.text if _v184_result.ok else ""
            if not _v184_result.ok:
                logging.warning(
                    "AI voice worker failed inbox=%s attempts=%s restarts=%s error=%s",
                    inbox_id, _v184_result.attempts,
                    _v184_result.restarted_workers, _v184_result.error,
                )
        except asyncio.TimeoutError:
            text = ""
            logging.warning("AI voice hard timeout inbox=%s", inbox_id)
        except Exception as exc:
            text = ""
            logging.error("Voice recognition failed inbox=%s error=%s", inbox_id, exc)
        if text:
            data.update(local_ocr.fields_from_text(text, allowed))
            data.update(fast.fast_text_data(text, ai_filter))

    elif kind in ("photo", "document") and file_id:
        if kind == "document":
            mime = (msg.document.mime_type or "") if msg.document else ""
            if not mime.startswith("image/"):
                await msg.reply_text("📎 Нужен файл-изображение.")
                return
        if msg.media_group_id:
            marker = "ai_mg_%s" % msg.media_group_id
            if context.chat_data.get(marker):
                return
            context.chat_data[marker] = True
        import photo_card  # CRM-PHOTO-AUTOFILL-001: photo -> new card by VIN
        await photo_card.handle(update, context, inbox_id, file_id, text)
        return

    elif (text or "").strip():
        thinking = await msg.reply_text("🧠 ИИ распознаёт, до 15 секунд...")
        data.update(local_ocr.fields_from_text(text, allowed))
        data.update(fast.fast_text_data(text, ai_filter))

    if data:
        parsed = fast.parsed_from_data(data)
        data = fast.clean_car(parsed, ai_filter, "")
    if data:
        parsed = fast.parsed_from_data(data)
        context.user_data["ai_draft"] = {
            "inbox_id": inbox_id,
            "parsed": parsed,
            "clean_data": data,
            "source_text": "",
            "review": None,
        }
        rows = [[InlineKeyboardButton("✅ Сохранить карточку", callback_data="ai_ok")], [
            InlineKeyboardButton("✏️ Дополнить", callback_data="ai_edit"),
            InlineKeyboardButton("❌ Отменить", callback_data="ai_no"),
        ]]
        elapsed = time.monotonic() - started
        if thinking is None:
            thinking = await msg.reply_text("🧠 ИИ распознаёт, до 15 секунд...")
        await thinking.edit_text(
            "✅ Новая карточка открыта\n\n" + ai_filter.render(data)
            + "\n\n⏱ %.1f с · локально · 0 AI-токенов" % elapsed,
            parse_mode="HTML",
            reply_markup=InlineKeyboardMarkup(rows),
        )
        return

    if thinking is None:
        thinking = await msg.reply_text("🧠 ИИ распознаёт, до 15 секунд...")
    await thinking.edit_text(
        "Поля CRM не распознаны. Пришлите более чёткое фото, текст или голос."
    )

def collect_media(inbox_id: int):
    """Собирает фото и видео записи. Альбом приходит несколькими
    сообщениями — их надо взять все, а не только первое."""
    zapis = db.get_inbox(inbox_id)
    if not zapis:
        return [], []

    foto, video = [], []

    def _dobavit(row):
        if not row.get("file_id"):
            return
        if row["kind"] == "photo" and row["file_id"] not in foto:
            foto.append(row["file_id"])
        elif row["kind"] in ("video", "video_note") and row["file_id"] not in video:
            video.append(row["file_id"])

    _dobavit(zapis)

    # Остальные части альбома
    if zapis.get("media_group"):
        with db.connect() as c:
            rows = c.execute(
                "SELECT * FROM inbox WHERE media_group=? ORDER BY id",
                (zapis["media_group"],),
            ).fetchall()
        for row in rows:
            _dobavit(dict(row))

    return foto, video


async def ai_save(update: Update, context: ContextTypes.DEFAULT_TYPE):
    import ai_fast_schema as fast

    query = update.callback_query
    await query.answer()
    staff = who(update)
    draft = context.user_data.get("ai_draft")
    if not draft or not staff:
        await query.message.reply_text("Черновик потерялся. Отправьте сообщение заново.")
        return

    parsed = draft.get("parsed") or {}
    table = "cars" if parsed.get("type") == "car" else "clients"
    actor = staff["user_id"]
    if table == "cars":
        data = {
            key: value for key, value in (draft.get("clean_data") or {}).items()
            if key in fast.car_fields(ai_filter)
        }
        if not data:
            await query.message.reply_text("Нет полей автомобиля для сохранения.")
            return
        card_id, number, new_card = ai_filter.store(
            data, actor, draft.get("inbox_id"), card_id=None
        )
        if not card_id:
            await query.message.reply_text("Для новой карточки нужен VIN.")
            return
        db.set_card_review("cars", card_id, db.ST_APPROVED_OWNER, actor, publish=True)
        db.add_comment(
            "cars", card_id, actor,
            "Создано распознаванием из входящего #%s; сохранены только поля CRM."
            % draft.get("inbox_id"),
        )
    else:
        allowed = set(getattr(db, "CLIENT_FIELDS", ()))
        data = {key: value for key, value in (draft.get("clean_data") or {}).items() if key in allowed}
        card_id = db.create_card("clients", data, actor)
        db.link_inbox_card(draft.get("inbox_id"), "clients", card_id, actor)
        db.set_inbox_status(draft.get("inbox_id"), db.ST_APPROVED_OWNER, actor)

    card = db.get_card(table, card_id)
    note = "\nОпубликовано в клиентском каталоге." if table == "cars" else ""
    await query.message.edit_reply_markup(reply_markup=None)
    await query.message.reply_text(
        "✅ Сохранено: %s%s" % (db.card_title(table, card), note),
        reply_markup=main_menu(staff),
    )
    context.user_data.pop("ai_draft", None)

async def ai_edit_apply(update: Update, context: ContextTypes.DEFAULT_TYPE):
    draft = context.user_data.get("ai_draft")
    if not draft:
        return ConversationHandler.END
    correction = update.message.text
    thinking = await update.message.reply_text("🧠 Исправляю...")

    parsed = ai.parse_message(draft["source_text"], correction=correction,
                              previous=draft["parsed"])
    if not parsed:
        await thinking.edit_text("Не получилось. Попробуйте сформулировать иначе.")
        return AI_FIX

    draft["parsed"] = parsed
    draft.setdefault("corrections", []).append(correction)
    issues = ai.validate(parsed)
    await thinking.edit_text(ai_filter.render_from(parsed, issues), parse_mode="HTML")
    proverka = ai.review(draft["source_text"], parsed, corrections=draft["corrections"])
    draft["review"] = proverka

    if await ai_filter.autosave(update, context, thinking, draft["inbox_id"], parsed):
        return ConversationHandler.END

    rows = []
    if parsed.get("type") in ("car", "client"):

async def inbox_open(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    staff = who(update)
    if not staff:
        return
    inbox_id = int(query.data.split(":")[1])
    item = db.get_inbox(inbox_id)
    if not item:
        await query.message.reply_text("Запись не найдена.")
        return

    if item["status"] == db.ST_NEW and is_manager(staff):
        db.set_inbox_status(inbox_id, db.ST_ASSIGNED, staff["user_id"],
                            assigned_to=staff["user_id"])

                "audio": context.bot.send_audio,
                "document": context.bot.send_document,

    when = (item["created_at"] or "")[:16].replace("T", " ")
    body = (
        f"📄 <b>Запись #{inbox_id}</b>\n"
        f"Статус: {db.STATUS_LABELS.get(item['status'], item['status'])}\n"
        f"Получено: {when}\n\n"

    )
    rows = [
        [InlineKeyboardButton("🚗 Создать карточку авто", callback_data=f"mk_car:{inbox_id}")],
        [InlineKeyboardButton("👤 Создать карточку клиента", callback_data=f"mk_client:{inbox_id}")],

async def mk_card_start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    staff = who(update)
    if not staff:
        return ConversationHandler.END
    kind, inbox_id = query.data.split(":")
    inbox_id = int(inbox_id)
    table = "cars" if kind == "mk_car" else "clients"
    steps = CAR_STEPS if table == "cars" else CLIENT_STEPS

    context.user_data["card"] = {
        "table": table, "inbox_id": inbox_id, "step": 0, "data": {}, "steps": steps,
    }
    db.set_inbox_status(inbox_id, db.ST_IN_WORK, staff["user_id"])

    title = "🚗 Новая карточка авто" if table == "cars" else "👤 Новая карточка клиента"
    await query.message.reply_text(
        f"{title}\n\nШаг 1 из {len(steps)} — {steps[0][1]}\n\n"
        f"Отправьте /cancel, чтобы отменить."
    )
    return CARD_STEP


async def mk_card_step(update: Update, context: ContextTypes.DEFAULT_TYPE):
    st = context.user_data.get("card")
    if not st:
        return ConversationHandler.END
    staff = who(update)
    value = (update.message.text or "").strip()
    field, _ = st["steps"][st["step"]]
    st["data"][field] = None if value == "-" else value
    st["step"] += 1

    if st["step"] < len(st["steps"]):
        _, prompt = st["steps"][st["step"]]
        await update.message.reply_text(
            f"Шаг {st['step'] + 1} из {len(st['steps'])} — {prompt}"
        )
        return CARD_STEP

    # Все поля собраны — сохраняем и отправляем на проверку
    table = st["table"]
    card_id = db.create_card(table, st["data"], staff["user_id"])
    db.link_inbox_card(st["inbox_id"], table, card_id, staff["user_id"])
    db.set_inbox_status(st["inbox_id"], db.ST_ON_REVIEW, staff["user_id"])

    card = db.get_card(table, card_id)
    await update.message.reply_text(
        f"✅ Карточка создана: {db.card_title(table, card)}\n"
        f"Статус: {db.STATUS_LABELS[db.ST_ON_REVIEW]}\n\n"
        f"Администратор получил уведомление.",
        reply_markup=main_menu(staff),
    )
    await notify_admins_review(context, table, card_id, staff)
    context.user_data.pop("card", None)
    return ConversationHandler.END

def render_card(table, card) -> str:
    labels = CAR_LABELS if table == "cars" else CLIENT_LABELS
    lines = [f"<b>{db.card_title(table, card)}</b>",
             f"Статус: {db.STATUS_LABELS.get(card['review_status'], card['review_status'])}"]
    if table == "cars":
        lines.append("Публикация в каталоге: " + ("да" if card.get("published") else "нет"))
    lines.append("")
    for f, lab in labels.items():
        v = card.get(f)
        if v:

    return "\n".join(lines)

async def card_review_action(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    staff = who(update)
    if not is_admin_or_owner(staff):
        await query.answer("Недостаточно прав", show_alert=True)
        return
    action, table, card_id = query.data.split(":")
    card_id = int(card_id)
    card = db.get_card(table, card_id)

    if action == "card_ok":
        status = db.ST_APPROVED_OWNER if is_owner(staff) else db.ST_APPROVED_ADMIN
        publish = True if table == "cars" else None
        db.set_card_review(table, card_id, status, staff["user_id"], publish=publish)
        await query.answer("Подтверждено")
        note = "\nКарточка опубликована в клиентском каталоге." if table == "cars" else ""
        await query.message.reply_text(
            f"✅ {db.card_title(table, card)} — подтверждено.{note}"
        )
        target = card.get("created_by")
    elif action == "card_back":
        db.set_card_review(table, card_id, db.ST_RETURNED, staff["user_id"], publish=False)
        await query.answer("Возвращено")
        await query.message.reply_text(
            f"↩️ {db.card_title(table, card)} возвращена менеджеру на доработку."
        )
        target = card.get("created_by")
    else:
        db.set_card_review(table, card_id, db.ST_REJECTED, staff["user_id"], publish=False)
        await query.answer("Отклонено")
        await query.message.reply_text(f"❌ {db.card_title(table, card)} отклонена.")
        target = card.get("created_by")

    if target:
        try:
            await context.bot.send_message(
                target,
                f"{db.STATUS_LABELS[db.get_card(table, card_id)['review_status']]}\n"
                f"{db.card_title(table, card)}",
                reply_markup=InlineKeyboardMarkup([[
                    InlineKeyboardButton("Открыть", callback_data=f"card_open:{table}:{card_id}")
                ]]),
            )
        except Exception:
            pass

async def do_search(update: Update, context: ContextTypes.DEFAULT_TYPE, text: str):
    staff = who(update)
    res = db.search(text)
    rows, lines = [], []
    if res["cars"]:
        lines.append(f"🚗 Найдено авто: {len(res['cars'])}")
        for c in res["cars"][:10]:
            rows.append([InlineKeyboardButton(
                f"🚗 {db.card_title('cars', c)}", callback_data=f"card_open:cars:{c['id']}")])
    if res["clients"]:
        lines.append(f"👤 Найдено клиентов: {len(res['clients'])}")
        for c in res["clients"][:10]:
            rows.append([InlineKeyboardButton(
                f"👤 {db.card_title('clients', c)}", callback_data=f"card_open:clients:{c['id']}")])
    if not rows:
        await update.message.reply_text(
            f"По запросу «{text}» ничего не найдено.",
            reply_markup=main_menu(staff),
        )
        return
    rows.append([back_btn()])
    await update.message.reply_text("\n".join(lines),
                                    reply_markup=InlineKeyboardMarkup(rows))

async def audit_list(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer()
    staff = who(update)
    if not staff:
        return
    entries = db.get_audit(limit=25)
    if not entries:
        await query.message.reply_text("Журнал пуст.")
        return
    lines = ["📋 <b>Журнал действий</b>", ""]
    for e in entries:
        when = (e["created_at"] or "")[11:16]
        actor = db.get_staff(e["actor_id"]) or {}
        name = (actor.get("full_name") or str(e["actor_id"]))[:18]
        where = f"{e['entity_type']}#{e['entity_id']}" if e["entity_type"] else ""
        if e["old_value"] is not None and e["new_value"] is not None:
            lines.append(f"{when} {name}: {where} {e['old_value']} → {e['new_value']}")
        else:
            lines.append(f"{when} {name}: {e['action']} {where}")
    await query.message.reply_text("\n".join(lines), parse_mode="HTML",
                                   reply_markup=InlineKeyboardMarkup([[back_btn()]]))

async def inbox_ask(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Менеджер просит владельца уточнить — вопрос уходит владельцу."""
    query = update.callback_query
    await query.answer()
    staff = who(update)
    inbox_id = int(query.data.split(":")[1])
    db.set_inbox_status(inbox_id, db.ST_NEED_INFO, staff["user_id"])
    for oid in db.staff_ids_by_role(db.ROLE_OWNER):
        try:
            await context.bot.send_message(
                oid,
                f"❓ <b>Менеджер просит уточнить по записи #{inbox_id}</b>\n\n"
                f"{(db.get_inbox(inbox_id) or {}).get('text') or ''}\n\n"
                f"Ответьте сообщением — оно попадёт в систему как новая запись.",
                parse_mode="HTML",
            )
        except Exception:
            pass
    await query.message.reply_text(f"❓ Запрос уточнения по записи #{inbox_id} отправлен владельцу.")


async def error_handler(update, context):

    никто не читает — теперь о ней узнаёт владелец."""
    logging.error("Ошибка в team_bot:", exc_info=context.error)
    tb = "".join(traceback.format_exception(
        None, context.error, context.error.__traceback__
    ))
    for uid in db.staff_ids_by_role(db.ROLE_OWNER):
        try:
            await context.bot.send_message(
                uid,
                f"⚠️ <b>Сбой во внутреннем боте</b>\n\n<code>{tb[-1000:]}</code>",
                parse_mode="HTML",
            )
        except Exception:
            pass

async def vvodnye_job(context: ContextTypes.DEFAULT_TYPE):
    """После запуска бот сам присылает владельцу текущие вводные."""
    # UA-TASK084-STARTUP-NOTICE-DEBOUNCE
    import fcntl as _v184_fcntl
    import os as _v184_os
    import time as _v184_time

    _v184_state = "/home/Carix/.team_bot_start_notice"
    _v184_lock = "/home/Carix/.team_bot_start_notice.lock"
    try:
        with open(_v184_lock, "a+", encoding="utf-8") as _v184_guard:
            _v184_fcntl.flock(_v184_guard.fileno(), _v184_fcntl.LOCK_EX)
            try:
                with open(_v184_state, "r", encoding="utf-8") as _v184_handle:
                    _v184_last = int((_v184_handle.read() or "0").strip())
            except (FileNotFoundError, ValueError):
                _v184_last = 0
            _v184_now = int(_v184_time.time())
            if _v184_now - _v184_last < 600:
                logging.info("Повторное стартовое уведомление подавлено.")
                return
            _v184_temp = "%s.tmp.%d" % (_v184_state, _v184_os.getpid())
            with open(_v184_temp, "w", encoding="utf-8") as _v184_handle:
                _v184_handle.write(str(_v184_now))
                _v184_handle.flush()
                _v184_os.fsync(_v184_handle.fileno())
            _v184_os.replace(_v184_temp, _v184_state)
    except Exception as e:
        logging.warning("Стартовое уведомление подавлено: защитный шлюз недоступен: %s", e)
        return

    try:
        ids = db.staff_ids_by_role(db.ROLE_OWNER)
    except Exception as e:
        logging.warning("владелец не найден: %s", e)
        return
    for uid in ids or []:
        try:
            await context.bot.send_message(
                uid, "🟢 <b>Бот запущен и готов к работе.</b>\n\n" + vvodnye_text(),
                parse_mode="HTML")
        except Exception as e:
            logging.warning("вводные не отправились %s: %s", uid, e)
