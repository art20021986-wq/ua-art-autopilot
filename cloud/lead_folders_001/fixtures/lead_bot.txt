def _load_json(path, default):
    if not os.path.exists(path):
        return default
    try:
        with open(path, "r", encoding="utf-8") as f:
            return json.load(f)
    except (json.JSONDecodeError, OSError) as e:
        logging.error(f"Не удалось прочитать {path}: {e}. Использую пустое значение.")
        return default


def _save_json(path, data):
    """Пишем через временный файл: если процесс упадёт в момент записи,
    основной файл с базой останется целым."""
    tmp = path + ".tmp"
    try:
        with open(tmp, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        os.replace(tmp, path)
    except OSError as e:
        logging.error(f"Не удалось записать {path}: {e}")

# Машины из CRM получают id со сдвигом, чтобы не сталкиваться с теми,
# что заведены напрямую в клиентском боте через /addcar.
CRM_ID_OFFSET = 100000


def _crm_to_catalog(row: dict) -> dict:
    """Карточка из crm.db в том виде, в каком её ждёт клиентский каталог.
    Без этого перевода подтверждённые машины до клиентов не доходили."""
    nazvanie = " ".join(
        str(x) for x in (row.get("brand"), row.get("model"), row.get("year")) if x
    ) or "Автомобиль"

    # Характеристики собираем из отдельных полей CRM
    chasti = []
    for pole, podpis in (
        ("mileage", "пробег"), ("color", "цвет"), ("engine", "двигатель"),
        ("fuel", "топливо"), ("trim", "комплектация"), ("condition", "состояние"),
        ("accidents", "ДТП"),
    ):
        znachenie = row.get(pole)
        if znachenie:
            chasti.append(f"{podpis}: {znachenie}")

    def _spisok(raw):
        if not raw:
            return []
        try:
            data = json.loads(raw) if isinstance(raw, str) else raw
            return data if isinstance(data, list) else []
        except (json.JSONDecodeError, TypeError):
            return []

    return {
        "id": CRM_ID_OFFSET + row["id"],
        "crm_id": row["id"],
        "model": nazvanie,
        "specs": ", ".join(chasti),
        "price": row.get("price_total") or row.get("price_buy") or "уточняйте у менеджера",
        "status": row.get("stage") or "purchased",
        "vin": row.get("vin"),
        "photo_file_ids": _spisok(row.get("photos")),
        "video_file_ids": _spisok(row.get("videos")),
        "tech_report_desc": row.get("condition"),
        "created_at": row.get("created_at"),
        "status_changed_at": row.get("updated_at") or row.get("created_at"),
        "viewers": [],
        "buyers": [],
    }


def load_cars():
    """Каталог = машины, заведённые в клиентском боте, плюс подтверждённые
    карточки из CRM. Раньше вторые никуда не попадали."""
    cars = _load_json(CARS_DB_FILE, [])
    try:
        import db as crm_db
        for row in crm_db.published_cars():
            cars.append(_crm_to_catalog(row))
    except Exception as e:
        # CRM недоступна — каталог всё равно должен работать
        logging.warning(f"Не удалось прочитать машины из CRM: {e}")
    return cars


def save_cars(cars):
    """Пишем только свои машины. Карточки из CRM редактируются
    во внутреннем боте, здесь их трогать нельзя."""
    svoi = [c for c in cars if c.get("id", 0) < CRM_ID_OFFSET]
    _save_json(CARS_DB_FILE, svoi)


def is_crm_car(car_id: int) -> bool:
    return car_id >= CRM_ID_OFFSET


def next_car_id(cars):
    svoi = [c["id"] for c in cars if c.get("id", 0) < CRM_ID_OFFSET]
    return (max(svoi, default=0)) + 1


def get_car(car_id: int):

def remaining_days_text(car: dict) -> str:
    status = car.get("status")
    duration = STATUS_DURATIONS_DAYS.get(status)
    if not duration:
        return ""
    base_iso = car.get("ship_date") or car.get("status_changed_at") or car.get("created_at")
    if not base_iso:
        return f"⏳ Ориентировочно ~{duration} дней на этом этапе"
    try:
        base_date = datetime.fromisoformat(base_iso)
    except ValueError:
        return f"⏳ Ориентировочно ~{duration} дней на этом этапе"
    target = base_date + timedelta(days=duration)
    remaining = (target - datetime.now()).days
    if remaining > 0:
        return f"⏳ Осталось ~{remaining} дней"
    return "⏳ Со дня на день"


# Сколько дней остаётся до Украины, считая от начала текущего этапа
DAYS_TO_UKRAINE = {
    "purchased": 14 + 60 + 14,
    "korea_georgia": 60 + 14,
    "arrived_georgia": 14,
}


def arrival_date_text(car: dict) -> str:
    """Конкретная дата убеждает сильнее, чем «осталось N дней»."""
    status = car.get("status")
    if status == "in_ukraine":
        return "📍 Уже в Украине — можно забирать"
    days_left = DAYS_TO_UKRAINE.get(status)
    if not days_left:
        return ""
    base_iso = car.get("ship_date") or car.get("status_changed_at") or car.get("created_at")
    if not base_iso:
        return ""
    try:
        base_date = datetime.fromisoformat(base_iso)
    except ValueError:
        return ""
    arrival = base_date + timedelta(days=days_left)
    if arrival < datetime.now():
        return "📅 Ожидаем в Украине со дня на день"
    return f"📅 Ожидаем в Украине: ~{arrival.strftime('%d.%m.%Y')}"

def load_users():
    return _load_json(USERS_DB_FILE, {})


def save_users(users):
    _save_json(USERS_DB_FILE, users)


def track_user(user, source=None):
    users = load_users()
    uid = str(user.id)
    entry = users.get(uid, {"cars_viewed": [], "cars_bought": []})
    entry["username"] = user.username
    entry["full_name"] = user.full_name
    entry.setdefault("cars_viewed", [])
    entry.setdefault("cars_bought", [])
    entry.setdefault("first_seen", datetime.now().isoformat())
    # Источник фиксируем только первый раз — важно, откуда человек пришёл изначально
    if source and not entry.get("source"):
        entry["source"] = source
    entry["last_seen"] = datetime.now().isoformat()
    users[uid] = entry
    save_users(users)


def get_user_source(user_id) -> str:
    users = load_users()
    entry = users.get(str(user_id), {})
    raw = entry.get("source")
    if not raw:
        return "Прямой заход в бота"
    return SOURCE_LABELS.get(raw.lower(), raw)


def track_view(user, car_id: int):
    track_user(user)
    users = load_users()
    uid = str(user.id)
    if car_id not in users[uid]["cars_viewed"]:
        users[uid]["cars_viewed"].append(car_id)
    save_users(users)

    if is_crm_car(car_id):
        return  # интерес к машинам из CRM отслеживается по users_db
    cars = _load_json(CARS_DB_FILE, [])
    for c in cars:
        if c["id"] == car_id:
            c.setdefault("viewers", [])
            if user.id not in c["viewers"]:
                c["viewers"].append(user.id)
            break
    _save_json(CARS_DB_FILE, cars)


def track_buy(user, car_id: int):
    track_user(user)
    users = load_users()
    uid = str(user.id)
    if car_id not in users[uid]["cars_bought"]:
        users[uid]["cars_bought"].append(car_id)
    save_users(users)

    if is_crm_car(car_id):
        return
    cars = _load_json(CARS_DB_FILE, [])
    for c in cars:
        if c["id"] == car_id:
            c.setdefault("buyers", [])
            if user.id not in c["buyers"]:
                c["buyers"].append(user.id)
            break
    _save_json(CARS_DB_FILE, cars)

def save_lead(lead: dict) -> int:
    """Лид сначала пишется на диск, и только потом уходит менеджеру.
    Если Telegram недоступен или менеджер заблокировал бота — заявка всё равно
    сохранена и её видно через /leads."""
    leads = _load_json(LEADS_DB_FILE, [])
    lead["id"] = (max([l.get("id", 0) for l in leads], default=0)) + 1
    lead["created_at"] = datetime.now().isoformat()
    lead["delivered_to_manager"] = False
    leads.append(lead)
    _save_json(LEADS_DB_FILE, leads)
    mirror_lead_to_crm(lead)
    return lead["id"]


def mirror_lead_to_crm(lead: dict):
    """Заявка становится карточкой клиента в CRM. Без этого лиды жили
    отдельно, и найти человека поиском по телефону было невозможно."""
    try:
        import db as crm_db
    except Exception:
        return
    try:
        crm_db.init_db()
        telegram_handle = f"@{lead['username']}" if lead.get("username") else None

        # Тот же человек мог обращаться раньше — не плодим дубли
        with crm_db.connect() as c:
            existing = None
            if lead.get("user_id"):
                existing = c.execute(
                    "SELECT id FROM clients WHERE tg_user_id=?", (lead["user_id"],)
                ).fetchone()
            if existing:
                if lead.get("phone"):
                    c.execute("UPDATE clients SET phone=?, updated_at=? WHERE id=?",
                              (lead["phone"], crm_db.now(), existing["id"]))
                return

        data = {
            "name": lead.get("full_name"),
            "phone": lead.get("phone"),
            "telegram": telegram_handle,
            "car_interest": lead.get("car_model") or lead.get("model"),
            "budget": lead.get("budget"),
            "wishes": lead.get("message") or lead.get("timing"),
            "source": lead.get("source"),
            "stage": lead.get("type"),
        }
        data = {k: v for k, v in data.items() if v}
        card_id = crm_db.create_card("clients", data, lead.get("user_id") or 0)
        with crm_db.connect() as c:
            c.execute("UPDATE clients SET tg_user_id=? WHERE id=?",
                      (lead.get("user_id"), card_id))
        crm_db.log_action(lead.get("user_id") or 0, "lead_from_bot", "clients", card_id,
                          None, None, lead.get("type"))
    except Exception as e:
        logging.warning(f"Лид не продублирован в CRM: {e}")


def mark_lead_delivered(lead_id: int):
    leads = _load_json(LEADS_DB_FILE, [])
    for l in leads:
        if l.get("id") == lead_id:
            l["delivered_to_manager"] = True
            break
    _save_json(LEADS_DB_FILE, leads)


def mark_lead_taken(lead_id: int) -> dict:
    """Менеджер жмёт «Взял в работу» — с этого момента напоминания
    по заявке прекращаются."""
    leads = _load_json(LEADS_DB_FILE, [])
    found = {}
    for l in leads:
        if l.get("id") == lead_id:
            l["taken"] = True
            l["taken_at"] = datetime.now().isoformat()
            found = l
            break
    _save_json(LEADS_DB_FILE, leads)
    return found


def mark_lead_reminded(lead_id: int):
    leads = _load_json(LEADS_DB_FILE, [])
    for l in leads:
        if l.get("id") == lead_id:
            l["reminded"] = True
            break
    _save_json(LEADS_DB_FILE, leads)


def taken_markup(lead_id: int) -> InlineKeyboardMarkup:
    return InlineKeyboardMarkup([
        [InlineKeyboardButton("✅ Взял в работу", callback_data=f"lead_taken:{lead_id}")]
    ])


async def send_to_manager(context: ContextTypes.DEFAULT_TYPE, text: str, lead_id=None) -> bool:
    """Единая точка отправки менеджеру. Все ошибки ловим — падение отправки
    не должно ломать диалог с клиентом."""
    try:
        await context.bot.send_message(
            chat_id=MANAGER_CHAT_ID,
            text=text,
            parse_mode="HTML",
            reply_markup=taken_markup(lead_id) if lead_id else None,
        )
        if lead_id:
            mark_lead_delivered(lead_id)
        return True
    except Exception as e:
        logging.error(f"НЕ ОТПРАВЛЕНО МЕНЕДЖЕРУ (lead_id={lead_id}): {e}")
        return False


async def lead_taken_cb(update: Update, context: ContextTypes.DEFAULT_TYPE):
    query = update.callback_query
    await query.answer("Отмечено")
    if not is_admin(query.from_user.id):
        return
    lead_id = int(query.data.split(":")[1])
    mark_lead_taken(lead_id)
    try:
        await query.edit_message_reply_markup(reply_markup=None)
    except Exception:
        pass


def user_block(user, source_label=None) -> str:
    username_line = f"@{user.username}" if user.username else "(username не указан)"
    src = source_label or get_user_source(user.id)
    return (
        f"<b>Telegram:</b> {username_line}\n"
        f"<b>Имя в Telegram:</b> {user.full_name}\n"
        f"<b>Написать напрямую:</b> <a href=\"tg://user?id={user.id}\">открыть чат</a>\n"
        f"<b>ID:</b> <code>{user.id}</code>\n"
        f"<b>Источник:</b> {src}"
    )

async def notify_car_watchers(context: ContextTypes.DEFAULT_TYPE, car: dict, message: str):
    watcher_ids = set(car.get("viewers", []) + car.get("buyers", []))
    if car.get("reserved_by"):
        watcher_ids.add(car["reserved_by"])
    for uid in watcher_ids:
        try:
            await context.bot.send_message(chat_id=uid, text=message, parse_mode="HTML")
        except Exception as e:
            logging.warning(f"Не удалось уведомить {uid}: {e}")


async def manager_reminder_job(context: ContextTypes.DEFAULT_TYPE):
    """Заявка без отметки «Взял в работу» дольше часа — напоминаем.
    Один раз, чтобы не превращать чат в спам."""
    leads = _load_json(LEADS_DB_FILE, [])
    now = datetime.now()
    for l in leads:
        if l.get("taken") or l.get("reminded"):
            continue
        try:
            created = datetime.fromisoformat(l.get("created_at", ""))
        except ValueError:
            continue
        if (now - created) < timedelta(minutes=REMINDER_AFTER_MINUTES):
            continue
        mins = int((now - created).total_seconds() // 60)
        who = f"@{l['username']}" if l.get("username") else l.get("full_name", "клиент")
        phone = f"\n📱 {l['phone']}" if l.get("phone") else ""
        detail = l.get("car_model") or l.get("model") or (l.get("message", "") or "")[:60]
        try:
            await context.bot.send_message(
                chat_id=MANAGER_CHAT_ID,
                text=(
                    f"⏰ <b>Заявка #{l.get('id')} висит без ответа {mins} мин.</b>\n\n"
                    f"{who}{phone}\n"
                    f"{detail}\n\n"
                    f"<a href=\"tg://user?id={l.get('user_id')}\">Написать клиенту</a>"
                ),
                parse_mode="HTML",
                reply_markup=taken_markup(l.get("id")),
            )
            mark_lead_reminded(l.get("id"))
        except Exception as e:
            logging.warning(f"Не удалось напомнить по лиду {l.get('id')}: {e}")


def user_has_lead(user_id) -> bool:
    leads = _load_json(LEADS_DB_FILE, [])
    return any(str(l.get("user_id")) == str(user_id) for l in leads)


async def warmup_job(context: ContextTypes.DEFAULT_TYPE):
    """Прогрев: человек посмотрел машины и пропал — через 2 дня
    напоминаем о себе. Отправляется один раз на пользователя."""
    users = load_users()
    now = datetime.now()
    changed = False
    for uid, data in users.items():
        if data.get("warmed"):
            continue
        if not data.get("cars_viewed"):
            continue
        if user_has_lead(uid):  # уже оставил заявку — прогревать незачем
            continue
        try:
            last_seen = datetime.fromisoformat(data.get("last_seen", ""))
        except ValueError:
            continue
        if (now - last_seen) < timedelta(days=WARMUP_AFTER_DAYS):
            continue
        try:
            await context.bot.send_message(
                chat_id=int(uid),
                text=(
                    "👋 Здравствуйте! Это UA Art Company LLC.\n\n"
                    "Вы смотрели у нас автомобили — актуален ли ещё поиск?\n\n"
                    "Если не нашли подходящий вариант, напишите марку и бюджет — "
                    "подберём под ваш запрос и выкупим в Корее."
                ),
                reply_markup=lead_cta_markup(with_terms=True),
            )
            data["warmed"] = True
            changed = True
        except Exception as e:
            logging.warning(f"Прогрев не доставлен {uid}: {e}")
            data["warmed"] = True  # чтобы не долбиться в заблокировавших
            changed = True
    if changed:
        save_users(users)


async def weekly_reminder_job(context: ContextTypes.DEFAULT_TYPE):
    users = load_users()
    sent = 0
    for uid, data in users.items():
        if not data.get("cars_viewed"):
            continue
        try:
            await context.bot.send_message(
                chat_id=int(uid),
                text=(
                    "👋 Привет! Это UA Art Company LLC.\n\n"
                    "Хотели напомнить о себе — вы интересовались нашими автомобилями. "
                    "Актуален ли ещё поиск авто? Если что-то поменялось в бюджете или "
                    "модели — напишите, подберём заново."
                ),
                reply_markup=lead_cta_markup(),
            )
            sent += 1
        except Exception as e:
            logging.warning(f"Не удалось отправить напоминание {uid}: {e}")
    logging.info(f"Еженедельная рассылка: отправлено {sent}")

async def _stats_header(update, context):
    """Считает заход и показывает владельцу сводку. Клиенты её не видят."""
    try:
        user = update.effective_user
        istochnik = context.args[0] if getattr(context, "args", None) else None
        visits.hit(user, istochnik)
    except Exception:
        pass
    try:
        if not is_admin(update.effective_user.id):
            return
        await update.effective_message.reply_text(
            visits.short_line(),
            parse_mode="HTML",
            reply_markup=InlineKeyboardMarkup([[
                InlineKeyboardButton("Подробно", callback_data="stats_full"),
                InlineKeyboardButton("За неделю", callback_data="stats_week"),
            ]]),
        )
    except Exception:
        pass


async def start(update: Update, context: ContextTypes.DEFAULT_TYPE):
    await _stats_header(update, context)
    # Метка источника: ссылка вида t.me/ваш_бот?start=insta
    source = context.args[0] if context.args else None
    track_user(update.effective_user, source=source)
    context.user_data.clear()

    await update.message.reply_text(
        "Здравствуйте! Помогу подобрать корейский автомобиль. 🚗\n\n"
        "Выберите, что вас интересует:",
        reply_markup=main_menu_markup(),
    )
    return CAR_TYPE

        "Нажмите кнопку ниже, чтобы поделиться номером в один тап, "

        reply_markup=contact_request_markup(),
    )
    return CONTACT


async def contact_received(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Принимаем и нажатие кнопки «Поделиться номером», и набранный текст."""
    msg = update.message
    if msg.contact:
        context.user_data["phone"] = msg.contact.phone_number
        context.user_data["contact_text"] = f"{msg.contact.phone_number} (кнопкой)"
    else:
        text = msg.text or ""
        context.user_data["contact_text"] = text
        found = PHONE_RE.search(text)
        context.user_data["phone"] = found.group(0).strip() if found else None
    await finalize_lead(update, context)
    return ConversationHandler.END

async def finalize_lead(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user = update.effective_user
    data = context.user_data

    lead = {
        "type": "Анкета",
        "user_id": user.id,
        "username": user.username,
        "full_name": user.full_name,
        "car_type": data.get("car_type"),
        "model": data.get("model"),
        "budget": data.get("budget"),
        "timing": data.get("timing"),
        "contact_text": data.get("contact_text"),
        "phone": data.get("phone"),
        "source": get_user_source(user.id),
    }
    lead_id = save_lead(lead)  # сначала на диск, потом уже отправка

    timing_line = f"<b>Когда планирует покупку:</b> {data.get('timing')}\n" if data.get("timing") else ""
    phone_line = f"<b>📱 Телефон:</b> <code>{data.get('phone')}</code>\n" if data.get("phone") else ""

    text = (
        f"🔔 <b>Новый лид #{lead_id}</b>\n\n"
        f"<b>Тип:</b> {data.get('car_type', '-')}\n"
        f"<b>Модель:</b> {data.get('model', '-')}\n"
        f"<b>Бюджет:</b> {data.get('budget', '-')}\n"
        f"{timing_line}"
        f"{phone_line}"
        f"<b>Контакт (как написал):</b> {data.get('contact_text', '-')}\n\n"
        + user_block(user)
    )
    await send_to_manager(context, text, lead_id=lead_id)

    await update.message.reply_text(
        "Спасибо! ✅ Ваша заявка передана менеджеру, он свяжется с вами в ближайшее время.",
        reply_markup=ReplyKeyboardRemove(),
    )
    await update.message.reply_text(
        MANAGER_INTRO_TEXT,
        reply_markup=lead_cta_markup(with_catalog=True, with_terms=True),
    )
    context.user_data.clear()

INTEREST_WORDS = [
    "хочу", "нужен", "нужна", "интересует", "подобрать", "заказать", "куплю",
    "купить", "цена", "стоимость", "сколько", "бюджет", "рассрочка", "доставка",
    "kia", "hyundai", "genesis", "киа", "хендай", "хёндай", "хундай", "джензис",
    "sorento", "sportage", "tucson", "santa", "carnival", "sonata", "elantra",
    "ioniq", "ev6", "niro", "seltos", "palisade", "k5", "k8",
]


async def free_text_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Ловит любое сообщение вне анкеты: отвечает клиенту и показывает
    менеджеру, что человек пишет. Это и есть «видеть всех, кто проявил интерес»."""
    user = update.effective_user
    text = (update.message.text or "").strip()
    track_user(user)

    lower = text.lower()
    has_phone = bool(PHONE_RE.search(text))
    is_hot = has_phone or any(w in lower for w in INTEREST_WORDS)

    # Менеджеру — уведомление о живом сообщении
    if FORWARD_ALL_MESSAGES:
        marker = "🔥 <b>Горячее сообщение в боте</b>" if is_hot else "💬 <b>Сообщение в боте</b>"
        lead_id = None
        if is_hot:
            lead_id = save_lead({
                "type": "Свободное сообщение",
                "user_id": user.id,
                "username": user.username,
                "full_name": user.full_name,
                "message": text,
                "phone": PHONE_RE.search(text).group(0).strip() if has_phone else None,
                "source": get_user_source(user.id),
            })
        await send_to_manager(
            context,
            f"{marker}\n\n"
            f"<b>Текст:</b> {text[:800]}\n\n" + user_block(user),
            lead_id=lead_id,
        )

    # Клиенту — осмысленный ответ, а не тишина
    if has_phone:
        reply = (
            "Спасибо, записал ваш номер! 📱\n"
            "Менеджер свяжется с вами в ближайшее время.\n\n"
            "Пока можете посмотреть, что есть в наличии:"
        )
    elif is_hot:
        reply = (
            "Понял вас! 🚗 Передал сообщение менеджеру — он ответит в ближайшее время.\n\n"
            "Чтобы он сразу подобрал подходящие варианты, ответьте на 3 коротких вопроса:"
        )
    else:
        reply = (
            "Спасибо за сообщение! Менеджер его увидит и ответит.\n\n"
            "А пока — вот что можно сделать:"
        )

    await update.message.reply_text(reply, reply_markup=lead_cta_markup(with_terms=True))


async def non_text_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Фото, голосовые, стикеры вне анкеты."""
    user = update.effective_user
    track_user(user)
    if FORWARD_ALL_MESSAGES:
        await send_to_manager(
            context,
            "📎 <b>Клиент прислал медиа-сообщение в бота</b>\n\n" + user_block(user),

def visible_cars_for(user_id: int):
    cars = load_cars()
    result = []
    for c in cars:
        if c["status"] in ("sold", "issued"):
            continue
        if c["status"] == "reserved" and c.get("reserved_by") != user_id:
            continue
        result.append(c)
    return result


async def _render_catalog(chat_id, user, context):
    """«Авто в Украине» — только те, что уже здесь. Машины в пути
    показывает отдельная кнопка, чтобы списки не дублировались."""
    cars = [c for c in visible_cars_for(user.id) if c["status"] in ("in_ukraine", "reserved")]
    if not cars:
        transit_count = len([c for c in load_cars() if c.get("status") in IN_TRANSIT_STATUSES])
        hint = (
            f"\n\nСейчас в пути {transit_count} авто — посмотрите их."
            if transit_count else ""
        )
        await context.bot.send_message(
            chat_id=chat_id,
            text=(
                "🇺🇦 <b>Авто в Украине</b>\n\n"
                "Сейчас в наличии ничего нет.\n\n"
                "Мы возим авто под заказ из Кореи под ваш бюджет — "
                "оставьте заявку, и менеджер подберёт варианты лично для вас."
                + hint
            ),
            parse_mode="HTML",
            reply_markup=InlineKeyboardMarkup([
                [InlineKeyboardButton("📝 Оставить заявку", callback_data="lead_start")],
                [InlineKeyboardButton(transit_btn_label(), callback_data="show_transit")],
                [
                    InlineKeyboardButton(TERMS_BTN_TEXT, callback_data="show_terms"),
                    manager_button(),
                ],
            ]),
        )
        return
    await context.bot.send_message(
        chat_id=chat_id,
        text=f"🇺🇦 <b>Авто в Украине — {len(cars)} шт.</b>",
        parse_mode="HTML",

        chat_id=chat_id,
        text="Не нашли подходящее? Оставьте заявку — подберём под ваш бюджет.",

async def send_car_card(chat_id, context: ContextTypes.DEFAULT_TYPE, car: dict, viewer_id: int):
    days_line = remaining_days_text(car)
    arrival_line = arrival_date_text(car)
    tech = car.get("tech_report_desc")
    tech_line = f"\n<b>Техотчёт:</b> {tech}" if tech else ""

    caption = (
        f"<b>{car['model']}</b>\n"
        f"{car.get('specs', '')}\n\n"
        f"<b>Цена:</b> {car.get('price', 'уточняйте у менеджера')}\n"
        f"<b>Статус:</b> {STATUS_LABELS.get(car['status'], car['status'])}\n"
        f"{days_line}\n"
        f"{arrival_line}"
        f"{tech_line}"
    )

    buttons = []
    if car.get("container"):
        buttons.append([InlineKeyboardButton(
            "📍 Отследить контейнер", url=container_tracking_link(car["container"])
        )])
    if car.get("vin"):
        buttons.append([InlineKeyboardButton("🔎 Проверить VIN", url=vin_check_link(car["vin"]))])
    if car["status"] not in ("reserved",) or car.get("reserved_by") == viewer_id:
        buttons.append([InlineKeyboardButton(buy_button_label(car), callback_data=f"buycar:{car['id']}")])
    markup = InlineKeyboardMarkup(buttons)

    photos = car.get("photo_file_ids") or ([car["photo_file_id"]] if car.get("photo_file_id") else [])
    try:
        if photos:
            for i, photo_id in enumerate(photos):
                if i == 0:
                    await context.bot.send_photo(
                        chat_id=chat_id, photo=photo_id, caption=caption,
                        parse_mode="HTML", reply_markup=markup,
                    )
                else:
                    await context.bot.send_photo(chat_id=chat_id, photo=photo_id)
        else:
            await context.bot.send_message(
                chat_id=chat_id, text=caption, parse_mode="HTML", reply_markup=markup
            )

        videos = car.get("video_file_ids") or ([car["video_file_id"]] if car.get("video_file_id") else [])
        for video_id in videos:
            await context.bot.send_video(chat_id=chat_id, video=video_id)

        for p in car.get("tech_report_photos", []):
            await context.bot.send_photo(chat_id=chat_id, photo=p, caption="📋 Техотчёт: фото")
        for v in car.get("tech_report_videos", []):
            await context.bot.send_video(chat_id=chat_id, video=v, caption="📋 Техотчёт: видео")
    except Exception as e:
        # Например, устаревший file_id — карточка не должна ронять весь каталог
        logging.error(f"Ошибка показа машины #{car.get('id')}: {e}")
        await context.bot.send_message(
            chat_id=chat_id, text=caption, parse_mode="HTML", reply_markup=markup
        )

async def leads_cmd(update: Update, context: ContextTypes.DEFAULT_TYPE):
    msg = await _ack(update)
    if not is_admin(update.effective_user.id):
        return
    leads = _load_json(LEADS_DB_FILE, [])
    if not leads:
        await msg.reply_text("Заявок пока нет.", reply_markup=admin_menu_markup())
        return

    undelivered = [l for l in leads if not l.get("delivered_to_manager")]
    lines = [f"📊 Всего заявок: <b>{len(leads)}</b>"]
    untaken = [l for l in leads if not l.get("taken")]
    if untaken:
        lines.append(f"🔴 Не взято в работу: <b>{len(untaken)}</b>")
    if undelivered:
        lines.append(f"⚠️ Не дошло до чата менеджера: <b>{len(undelivered)}</b>")
    lines.append("\n<b>Последние 10:</b>")

    for l in leads[-10:][::-1]:
        when = l.get("created_at", "")[:16].replace("T", " ")
        who = f"@{l['username']}" if l.get("username") else l.get("full_name", "?")
        phone = f" 📱{l['phone']}" if l.get("phone") else ""
        detail = l.get("car_model") or l.get("model") or l.get("message", "")[:40] or "-"
        mark = "✅" if l.get("taken") else "🔴"
        lines.append(f"{mark} #{l.get('id')} {when} — {who}{phone} — {detail}")

    await msg.reply_text("\n".join(lines), parse_mode="HTML", reply_markup=admin_menu_markup())

    # Выгрузка в CSV — открывается в Excel и Google Таблицах
    try:
        csv_path = os.path.join(BASE_DIR, "leads_export.csv")
        fields = ["id", "created_at", "type", "full_name", "username", "phone",
                  "car_type", "model", "car_model", "budget", "timing",
                  "contact_text", "message", "source", "user_id", "taken", "taken_at"]
        with open(csv_path, "w", encoding="utf-8-sig", newline="") as f:
            writer = csv.DictWriter(f, fieldnames=fields, extrasaction="ignore")
            writer.writeheader()
            for l in leads:
                writer.writerow(l)
        with open(csv_path, "rb") as f:
            await context.bot.send_document(
                chat_id=update.effective_chat.id,
                document=f,
                filename="leads_export.csv",
                caption="Все заявки одним файлом (открывается в Excel).",
            )
    except Exception as e:
        logging.error(f"Не удалось выгрузить CSV: {e}")

async def editstatus_ship_date(update: Update, context: ContextTypes.DEFAULT_TYPE):
    text = update.message.text.strip()
    try:
        ship_date = datetime.strptime(text, "%d.%m.%Y")
    except ValueError:
        await update.message.reply_text("Не получилось распознать дату. Формат: ДД.ММ.ГГГГ, попробуйте ещё раз.")
        return ES_SHIP_DATE

    car_id = context.user_data["edit_car_id"]
    update_car(
        car_id,
        status="korea_georgia",
        container=context.user_data.get("edit_container"),
        ship_date=ship_date.isoformat(),
        status_changed_at=datetime.now().isoformat(),
    )
    arrival = ship_date + timedelta(days=STATUS_DURATIONS_DAYS["korea_georgia"])
    await update.message.reply_text(
        f"✅ Статус обновлён: {STATUS_LABELS['korea_georgia']}\n"
        f"Ожидаемое прибытие в Грузию: ~{arrival.strftime('%d.%m.%Y')}"
    )
    car = get_car(car_id)
    await notify_car_watchers(
        context, car,
        f"🚢 Ваш автомобиль {car['model']} отправлен! Ожидаемое прибытие в Грузию: ~{arrival.strftime('%d.%m.%Y')}."
    )
    for key in ("edit_car_id", "edit_new_status", "edit_container"):
        context.user_data.pop(key, None)
    return ConversationHandler.END


async def _apply_status_and_notify(message, context, new_price=None):
    car_id = context.user_data.get("edit_car_id")
    new_status = context.user_data.get("edit_new_status")
    update_car(
        car_id,
        status=new_status,
        price=new_price,
        status_changed_at=datetime.now().isoformat(),
    )
    car = get_car(car_id)
    await message.reply_text(f"✅ Статус обновлён: {STATUS_LABELS.get(new_status, new_status)}")

    notify_texts = {
        "arrived_georgia": f"🎉 Отличная новость! Ваш автомобиль {car['model']} прибыл в Грузию.",
        "in_ukraine": f"🇺🇦 Ваш автомобиль {car['model']} уже в Украине и готов к выдаче!",
    }
    if new_status in notify_texts:
        await notify_car_watchers(context, car, notify_texts[new_status])

    for key in ("edit_car_id", "edit_new_status"):
        context.user_data.pop(key, None)

async def reserve_buyer_id(update: Update, context: ContextTypes.DEFAULT_TYPE):
    text = update.message.text.strip()
    if not text.lstrip("-").isdigit():
        await update.message.reply_text("Это должно быть число (Telegram ID). Попробуйте ещё раз.")
        return RS_BUYER_ID
    buyer_id = int(text)
    car_id = context.user_data["reserve_car_id"]
    update_car(car_id, status="reserved", reserved_by=buyer_id, status_changed_at=datetime.now().isoformat())
    car = get_car(car_id)
    await update.message.reply_text(f"🔒 Машина #{car_id} закреплена за {buyer_id}. Скрыта из общего каталога.")
    try:
        await context.bot.send_message(
            chat_id=buyer_id,
            text=f"🔒 Машина {car['model']} закреплена за вами. Посмотреть — /mycar",
        )
    except Exception as e:
        logging.warning(f"Не удалось уведомить покупателя {buyer_id}: {e}")
        await update.message.reply_text(
            "⚠️ Клиенту не удалось отправить уведомление — возможно, он не запускал бота."
        )
    context.user_data.pop("reserve_car_id", None)
    return ConversationHandler.END

async def error_handler(update: object, context: ContextTypes.DEFAULT_TYPE):
    """Раньше ошибка просто уходила в лог, и вы о ней не узнавали."""
    logging.error("Ошибка при обработке обновления:", exc_info=context.error)
    tb = "".join(traceback.format_exception(None, context.error, context.error.__traceback__))
    try:
        await context.bot.send_message(
            chat_id=MANAGER_CHAT_ID,
            text=f"⚠️ <b>Ошибка в боте</b>\n\n<code>{tb[-1200:]}</code>",
            parse_mode="HTML",
        )
    except Exception:
        pass

async def _setup_menu(application):
    from telegram import BotCommand, BotCommandScopeChat, BotCommandScopeDefault
    klientu = [
        BotCommand("start", "Главное меню"),
        BotCommand("cars", "Авто в наличии и в пути"),
        BotCommand("terms", "Условия и гарантии"),
    ]
    vladelcu = klientu + [
        BotCommand("stats", "Статистика бота"),
        BotCommand("leads", "Заявки и выгрузка"),
        BotCommand("archive", "Архив машин"),
    ]
    try:
        await application.bot.set_my_commands(klientu, scope=BotCommandScopeDefault())
        for admin_id in ADMIN_IDS:
            await application.bot.set_my_commands(
                vladelcu, scope=BotCommandScopeChat(chat_id=admin_id))
    except Exception as e:
        print("Меню команд не обновилось:", e)


def main():
    logging.basicConfig(
        format="%(asctime)s [%(levelname)s] %(message)s", level=logging.INFO
    )
    app = build_application()
    if not app:
        return
    print("Бот запущен.")
    app.run_polling(drop_pending_updates=False)


if __name__ == "__main__":
    main()
