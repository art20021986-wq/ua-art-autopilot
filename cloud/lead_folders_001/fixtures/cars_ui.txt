def _task088_ge_number(raw):
    """Reuse the current parser's amount, never its default price_uah target."""
    import re
    text = str(raw or "").strip()
    if not text or re.search(r"(?:₴|€|₾|\b(?:uah|gel|eur|грн|грив\w*|лари)\b)",
                             text, re.I):
        raise ValueError("Укажите цену Грузии в долларах, одним числом.")
    try:
        from price_parser import parse_sale_price_message
    except ModuleNotFoundError as exc:
        if exc.name != "price_parser":
            raise
        normalized = re.sub(r"\s+", " ", text.replace("\u00a0", " ")).lower()
        normalized = re.sub(
            r"^(?:(?:измени|изменить|поставь|запиши|зміни)\s+)?"
            r"(?:(?:цена|цену|ціна|ціну|стоимость|вартість|price)\s*)?"
            r"(?:(?:грузии|грузії|georgia|украины|україни|ukraine|авто|автомобиля)\s*)?(?:на\s+)?", "", normalized)
        match = re.fullmatch(
            r"\$?\s*(\d{1,3}(?:[ ,.]\d{3})+|\d+)\s*"
            r"(?:\$|usd|доллар(?:ов|а)?|долар(?:ів|и)?|долл\.?)?", normalized)
        if match:
            value = int(re.sub(r"[ ,.]", "", match.group(1)))
        else:
            scaled = re.fullmatch(
                r"\$?\s*(\d+(?:[.,]\d{1,3})?)\s*(?:тысяч|тыс\.?|тис\.?|к)\s*"
                r"(?:\$|usd|доллар(?:ов|а)?|долар(?:ів|и)?|долл\.?)?", normalized)
            if not scaled:
                raise ValueError("Укажите одну цену Грузии, например 11 400 USD.")
            from decimal import Decimal
            value = int(Decimal(scaled.group(1).replace(",", ".")) * 1000)
    else:
        parsed = parse_sale_price_message(text, in_price_uah_wait=True)
        if not parsed.ok:
            raise ValueError("Не разобрал цену Грузии. Укажите одно число в долларах.")
        value = parsed.value
    if isinstance(value, bool) or not isinstance(value, int) or not 0 < value < 2**63:
        raise ValueError("Цена должна быть положительным целым числом.")
    return value

def _task088_apply_selected_price(card_id, field, raw, actor_id):
    """Commit the selected market price + audit; verify an independent DB read."""
    if field not in ("price_uah", "price_georgia"):
        return False, "Неизвестное поле цены."
    label = "Цена Украины" if field == "price_uah" else "Цена Грузии"
    try:
        value = _task088_ge_number(raw)
    except ValueError as exc:
        return False, str(exc).replace("цену Грузии", label.lower())
    except Exception:
        # A broken optional parser must not escape the Telegram callback or
        # proceed to the DB. Keep exception details out of the user response.
        return False, "%s не записана: не удалось проверить ввод. Повторите позже." % label
    conn = None
    try:
        conn = db.connect()
        conn.execute("BEGIN IMMEDIATE")
        all_before = {item["id"]: dict(item) for item in conn.execute("SELECT * FROM cars")}
        row = conn.execute("SELECT * FROM cars WHERE id=?", (int(card_id),)).fetchone()
        if row is None:
            conn.rollback()
            return False, "Карточка не найдена. %s не записана." % label
        before = dict(row)
        if not {"price_uah", "price_georgia", "updated_at"} <= set(before):
            raise RuntimeError("TASK088_GE_SCHEMA_MISSING")
        changed_at = db.now()
        if field == "price_uah":
            history = price_history(before)
            history[str(S.stage_of(before.get("status")) or 1)] = value
            history_text = jdump(history)
            conn.execute("UPDATE cars SET price_uah=?,price_history=?,updated_at=? WHERE id=?",
                         (value, history_text, changed_at, int(card_id)))
        else:
            conn.execute("UPDATE cars SET price_georgia=?,updated_at=? WHERE id=?",
                         (value, changed_at, int(card_id)))
        after = dict(conn.execute("SELECT * FROM cars WHERE id=?",
                                  (int(card_id),)).fetchone())
        expected = dict(before, updated_at=changed_at)
        expected[field] = value
        if field == "price_uah":
            expected["price_history"] = history_text
        if after != expected:
            raise RuntimeError("TASK088_GE_CROSS_WRITE")
        # The existing Ukraine path calls set_field twice: once for the price,
        # then remember_price -> set_field(price_history). Preserve both audit
        # records, in the same transaction as their corresponding values.
        changes = [(field, before[field], value)]
        if field == "price_uah":
            changes.append(("price_history", before["price_history"], history_text))
        audit_rows = []
        for changed_field, old_value, new_value in changes:
            audit_value = (actor_id, "card_edit", "cars", int(card_id), changed_field,
                           str(old_value) if old_value is not None else None,
                           str(new_value) if new_value is not None else None, changed_at)
            audit_cursor = conn.execute(
                "INSERT INTO audit (actor_id,action,entity_type,entity_id,field,"
                "old_value,new_value,created_at) VALUES (?,?,?,?,?,?,?,?)", audit_value)
            audit_rows.append((audit_cursor.lastrowid, audit_value))
        audit_sql = ("SELECT actor_id,action,entity_type,entity_id,field,"
                     "old_value,new_value,created_at FROM audit WHERE rowid=?")
        for audit_id, audit_value in audit_rows:
            recorded = conn.execute(audit_sql, (audit_id,)).fetchone()
            if recorded is None or tuple(recorded) != audit_value:
                raise RuntimeError("TASK088_AUDIT_RECORD_MISMATCH")
        checked = dict(conn.execute("SELECT * FROM cars WHERE id=?",
                                    (int(card_id),)).fetchone())
        if checked != expected:
            raise RuntimeError("TASK088_AUDIT_CROSS_WRITE")
        all_expected = dict(all_before)
        all_expected[int(card_id)] = expected
        all_after = {item["id"]: dict(item) for item in conn.execute("SELECT * FROM cars")}
        if all_after != all_expected:
            raise RuntimeError("TASK088_OTHER_CAR_CROSS_WRITE")
        conn.commit()
        conn.close()
        conn = None
        conn = db.connect()
        saved = conn.execute("SELECT * FROM cars WHERE id=?",
                             (int(card_id),)).fetchone()
        saved = dict(saved) if saved is not None else None
        if saved != expected:
            return False, ("Не удалось подтвердить сохранение: %s. "
                           "Откройте карточку повторно и проверьте значение.") % label
        for audit_id, audit_value in audit_rows:
            recorded = conn.execute(audit_sql, (audit_id,)).fetchone()
            if recorded is None or tuple(recorded) != audit_value:
                return False, ("Не удалось подтвердить журнал изменения: %s. "
                               "Откройте карточку повторно и проверьте значение.") % label
        return True, "%s сохранена: %s USD." % (label, S.num(saved[field]))
    except Exception:
        if conn is not None:
            conn.rollback()
        return False, "%s не подтверждена. Откройте карточку и повторите." % label
    finally:
        if conn is not None:
            conn.close()

def ensure_columns():
    """Добавляет колонки, которых нет ни в старой схеме, ни в cars_schema."""
    extra = [("price_history", "TEXT"), ("cover_photo", "TEXT"),
             ("eta_manual", "TEXT"), ("price_georgia", "INTEGER")]
    with db.connect() as c:
        have = {r["name"] for r in c.execute("PRAGMA table_info(cars)").fetchall()}
        for name, decl in extra:
            if name not in have:
                try:
                    c.execute("ALTER TABLE cars ADD COLUMN %s %s" % (name, decl))
                    log.info("Добавлена колонка cars.%s", name)
                except Exception as e:
                    log.warning("Не добавил cars.%s: %s", name, e)


def jload(value, default=None):
    if not value:
        return default if default is not None else []
    try:
        out = json.loads(value)
        return out if isinstance(out, (list, dict)) else (default or [])
    except Exception:
        return default if default is not None else []


def jdump(value):
    return json.dumps(value, ensure_ascii=False)


def card_of(card_id):
    card = db.get_card("cars", int(card_id))
    if not card:
        return None
    # старые значения подтягиваем в новые поля, если новые пустые
    for old, new in LEGACY.items():
        if not card.get(new) and card.get(old):
            card[new] = card[old]
    return card


def photos_of(card):
    """Список фото. Заглавная — передняя часть, за ней VIN и одометр."""
    raw = jload(card.get("photos"))
    items = []
    for p in raw:
        if isinstance(p, dict):
            items.append(p)
        else:
            items.append({"file_id": p, "tag": ""})
    cover = card.get("cover_photo")
    if cover:
        items.sort(key=lambda p: 0 if p.get("file_id") == cover else 1)
        return items
    return S.order_photos(items)


def videos_of(card):
    """Все видео карточки списком, до 30 штук."""
    out = []
    for v in jload(card.get("videos")):
        out.append(v if isinstance(v, dict) else {"file_id": v, "tag": ""})
    for old, tag in ((card.get("video_h"), "гориз"), (card.get("video_v"), "вертик")):
        if old and not any(p.get("file_id") == old for p in out):
            out.append({"file_id": old, "tag": tag})
    return out


def price_history(card):
    return jload(card.get("price_history"), {}) or {}


def price_stale(card):
    """Цену на текущем этапе ещё не подтверждали — надо напомнить."""
    etap = S.stage_of(card.get("status"))
    if not etap:
        return False
    return str(etap) not in price_history(card)


def set_field(card_id, field, value, actor_id):
    """Пустое значение в карточку не пишем — данные не теряются."""
    if value in (None, "", []):
        return
    db.update_card_field("cars", int(card_id), field, value, actor_id)

def _ua082_clean(value):
    return " ".join(str(value or "").split())


def _ua082_vin4(card):
    raw = str((card or {}).get("vin") or "").upper()
    normalized = "".join(ch for ch in raw if ch not in " -\t\r\n")
    allowed = frozenset("ABCDEFGHJKLMNPRSTUVWXYZ[N]")
    if len(normalized) == 17 and all(ch in allowed for ch in normalized):
        return normalized[-4:]
    return "НЕТ"


def _ua082_title_parts(card):
    card = card or {}
    number = _ua082_clean(card.get("auto_number") or ("#%s" % card.get("id")))
    name = " ".join(
        _ua082_clean(value)
        for value in (card.get("brand"), card.get("model"), card.get("year"))
        if _ua082_clean(value)
    ) or "без названия"
    return number, name, _ua082_vin4(card)


def _ua082_title_plain(card):
    number, name, vin4 = _ua082_title_parts(card)
    return "%s · %s · VIN %s" % (number, name, vin4)


def _ua082_title_html(card):
    import html as _ua082_html
    number, name, vin4 = _ua082_title_parts(card)
    return "%s · %s · VIN <b>%s</b>" % (
        _ua082_html.escape(number, quote=True),
        _ua082_html.escape(name, quote=True),
        _ua082_html.escape(vin4, quote=True),
    )


def _ua082_title_button(card):
    number, name, vin4 = _ua082_title_parts(card)
    suffix = " · VIN " + vin4
    prefix = (number + " · " + name).replace("<", "‹").replace(">", "›")
    limit = 64
    if len(prefix) + len(suffix) > limit:
        room = max(1, limit - len(suffix) - 1)
        prefix = prefix[:room].rstrip() + "…"
    return prefix + suffix


def render(card, staff):
    L = []
    L.append(_ua082_title_html(card))

    miss = S.missing_required(card)
    if miss:
        L.append("<i>Черновик · не хватает %d полей</i>" % len(miss))
    else:
        L.append("<i>%s · %s</i>" % (
            db.STATUS_LABELS.get(card.get("review_status"), ""),
            "в каталоге" if card.get("published") else "не опубликовано"))

    photos = photos_of(card)
    vids = videos_of(card)
    L.append("Фото: %d из %d · видео: %d из %d" % (
        len(photos), S.PHOTO_LIMIT, len(vids), VIDEO_LIMIT))

    L.append("")
    L.append("<b>ИДЕНТИФИКАЦИЯ</b>")
    L.append("Марка: <b>%s</b>" % (card.get("brand") or "—"))
    L.append("Модель: <b>%s</b>" % (card.get("model") or "—"))
    L.append("Год выпуска: %s" % (card.get("year") or "—"))
    L.append("VIN: <b>%s</b>" % (card.get("vin") or "—"))
    if card.get("vin"):
        ok, msg = S.check_vin(card.get("vin"), _year(card))
        L.append("  <i>%s</i>" % msg)

    L.append("")
    L.append("<b>ТЕХНИКА</b>")
    L.append("Топливо: %s" % (card.get("fuel") or "—"))
    L.append("Объём: <b>%s</b>" % S.num(_int(card.get("engine_cc")), " см³"))
    L.append("КПП: %s" % (card.get("gearbox") or "—"))
    L.append("Привод: %s" % (card.get("drive") or "—"))
    L.append("Пробег: %s" % S.num(_int(card.get("mileage_km")), " км"))
    L.append("Цвет: %s" % (card.get("color") or "—"))

    L.append("")
    L.append("<b>ТЕХНИЧЕСКОЕ СОСТОЯНИЕ</b>")
    L.append(card.get("condition_text") or "—")
    L.append("Фото: %d из %d · видео: %d из %d" % (
        len(jload(card.get("condition_photos"))), S.CONDITION_PHOTO_LIMIT,
        len(jload(card.get("condition_videos"))), S.CONDITION_VIDEO_LIMIT))
    if card.get("diag_report") or card.get("diag_link"):
        L.append("Диагностика: отчёт загружен")
        if card.get("diag_link"):
            L.append('  <a href="%s">полный отчёт</a>' % card["diag_link"])

    L.append("")
    stage = S.stage_of(card.get("status"))
    L.append("<b>ЛОГИСТИКА · ЭТАП %d ИЗ 4</b>" % (stage or 1))
    for n, name in S.STAGES:
        mark = "пройден" if n < stage else ("сейчас" if n == stage else "—")
        L.append("%d. %s: <b>%s</b>" % (n, name, mark))
    if card.get("status"):
        L.append("Статус: %s" % S.status_label(card.get("status")))
    if card.get("sea_container"):
        L.append("Контейнер: <b>%s</b>" % card["sea_container"])

    left, eta = eta_of(card)
    # Подпись зависит от того, где машина сейчас.
    # На море счёт идёт до Грузии, до Киева — только когда она уже в Грузии.
    podpis = {2: "ДО ПОТИ/БАТУМИ", 3: "ДО КИЕВА"}.get(stage)
    if left is not None and podpis:
        L.append("")
        L.append("🚚 <b>%s: %d дней</b>" % (podpis, left))
        L.append("ориентировочно %s%s" % (
            eta.strftime("%d.%m.%Y"),
            " · задано вручную" if card.get("eta_manual") else ""))

    klient = client_of(card)
    if klient:
        L.append("")
        L.append("<b>ПОКУПАТЕЛЬ</b>")
        L.append("Имя: %s" % (klient.get("name") or "—"))
        L.append("Телефон: %s" % (klient.get("phone") or "—"))
        if klient.get("telegram"):
            L.append("Telegram: %s" % klient["telegram"])
        if klient.get("prepayment"):
            L.append("Предоплата: %s" % klient["prepayment"])
        if klient.get("wishes"):
            L.append("Заметка: %s" % klient["wishes"])

    if is_owner(staff):
        L.extend(CR.owner_money(card))
        if price_stale(card):
            L.append("⚠️ <b>Цена на этом этапе ещё не обновлена</b>")

    if miss:
        L.append("")
        L.append("<b>НЕ ХВАТАЕТ</b>")
        L.append(", ".join(miss))

    return "\n".join(L)


def eta_of(card):
    """Сколько дней до Киева и когда. Ручной срок важнее расчётного."""
    manual = card.get("eta_manual")
    if manual:
        try:
            eta = _dt.strptime(str(manual)[:10], "%Y-%m-%d").date()
            return max((eta - _date.today()).days, 0), eta
        except Exception:
            pass
    return S.days_to_kyiv(card)


def _int(v):
    try:
        return int(str(v).replace(" ", "").replace("км", "").strip())
    except Exception:
        return None


def _year(card):
    try:
        return int(str(card.get("year"))[:4])
    except Exception:
        return None

def card_kb(card, staff):
    cid = card["id"]
    rows = [
        [InlineKeyboardButton("Редактировать данные", callback_data="car_edit:%d" % cid),
         InlineKeyboardButton("Фото и видео", callback_data="car_media:%d" % cid)],
        [InlineKeyboardButton("🚚 Доставка и этапы", callback_data="car_stage:%d" % cid)],
        [InlineKeyboardButton("Комплексная диагностика", callback_data="car_cond:%d" % cid)],
    ]
    klient = client_of(card)
    rows.append([InlineKeyboardButton(
        "Покупатель" if klient else "Оформить за клиентом",
        callback_data="car_client:%d" % cid)])
    rows.append([InlineKeyboardButton("Разместить объявление",
                                      callback_data="car_ad:%d" % cid)])
    rows.append([
        InlineKeyboardButton("Продано", callback_data="car_sold:%d" % cid),
    ])
    if card.get("published"):
        rows.append([InlineKeyboardButton("Скрыть от клиентов", callback_data="car_hide:%d" % cid)])
    rows.append([InlineKeyboardButton("Удалить", callback_data="car_del:%d" % cid)])
    rows.append([InlineKeyboardButton("← Все автомобили", callback_data="cards_cars")])
    return InlineKeyboardMarkup(rows)

async def open_card(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Open the card immediately; media delivery is bounded and cannot freeze CRM."""
    import asyncio as _v189_asyncio
    q = update.callback_query
    await _v168_ack(q,)
    drop_wait(context)
    staff = db.get_staff(q.from_user.id)
    cid = q.data.split(":")[-1]
    card = card_of(cid)
    if not card:
        await q.message.reply_text("Карточка не найдена.")
        raise ApplicationHandlerStop

    # CRM-VOICE-PHOTO-STAGE-REPAIR-089-V1: the card itself is always first.
    await q.message.reply_text(
        render(card, staff), parse_mode="HTML", reply_markup=card_kb(card, staff),
        disable_web_page_preview=True)
    context.user_data["car_last"] = int(cid)
    context.user_data["car_voice_active"] = int(cid)

def apply_value(card_id, field, raw, actor_id):
    """Возвращает (ок, сообщение). Ничего не додумывает."""
    if field == "price_georgia":
        return _task088_apply_selected_price(card_id, field, raw, actor_id)
    value = (raw or "").strip()
    if not value:
        return False, "Пустое значение не записываю."

    if field == "eta_days":
        digits = "".join(ch for ch in value if ch.isdigit())
        if not digits:
            return False, "Пришлите число дней, например 45."
        n = int(digits)
        if n > 400:
            return False, "Слишком большой срок. Пришлите число дней."
        eta = _date.today() + _td(days=n)
        set_field(card_id, "eta_manual", eta.isoformat(), actor_id)
        set_field(card_id, "days_to_kyiv", n, actor_id)
        return True, "До прибытия %d дней · %s" % (n, eta.strftime("%d.%m.%Y"))

    if field in NUMERIC:
        digits = value.lower().replace(" ", "").replace("\u00a0", "")
        for lishnee in ("цена", "стоимость", "$", "usd", "долларов", "долл"):
            digits = digits.replace(lishnee, "")
        for word, mult in (("тысяч", 1000), ("тыс", 1000), ("к", 1000)):
            if word in digits.lower():
                digits = digits.lower().split(word)[0]
                try:
                    num = int(float(digits.replace(",", ".")) * mult)
                except ValueError:
                    return False, "Не разобрал число. Пришлите цифрами."
                break
        else:
            digits = "".join(ch for ch in digits if ch.isdigit())
            if not digits:
                return False, "Не разобрал число. Пришлите цифрами."
            num = int(digits)
        if field == "engine_cc" and num < 100:
            return False, ("Похоже, это литры. Объём нужен в кубических сантиметрах — "
                           "например, 1598.")
        set_field(card_id, field, num, actor_id)
        return True, "Записано: %s" % S.num(num)

    if field == "vin":
        vin = value.upper().replace(" ", "")
        ok, msg = S.check_vin(vin)
        if not ok:
            return False, "VIN не принят: %s" % msg
        set_field(card_id, field, vin, actor_id)
        return True, "Записано: %s" % vin

    set_field(card_id, field, value, actor_id)
    return True, "Записано."

def _save_media_bazovoe(card, target, file_id, actor_id, tag=""):  # UA-V142
    """Write one unique media id using the latest authoritative card state."""
    cid = card["id"]
    latest = card_of(cid)
    if latest:
        card = latest
    if target in ("video_h", "video_v"):
        target = "videos"
    if target == "diag_report":
        set_field(cid, target, file_id, actor_id)
        return "Отчёт диагностики сохранён."

    limit = {"photos": S.PHOTO_LIMIT,
             "videos": VIDEO_LIMIT,
             "condition_photos": S.CONDITION_PHOTO_LIMIT,
             "condition_videos": S.CONDITION_VIDEO_LIMIT}.get(target, 99)
    items = jload(card.get(target))
    norm = []
    for p in items:
        norm.append(p if isinstance(p, dict) else {"file_id": p, "tag": ""})
    if any(p.get("file_id") == file_id for p in norm):
        return None
    if len(norm) >= limit:
        return "Достигнут предел: %d." % limit
    norm.append({"file_id": file_id, "tag": tag})
    set_field(cid, target, jdump(norm), actor_id)
    return "Сохранено. Всего: %d из %d." % (len(norm), limit)

import re as _re

VIN_RE = _re.compile(r"\b[A-HJ-NPR-Z0-9]{17}\b", _re.I)

# фраза в чате -> поле карточки. Берём только само значение, не всю фразу.
PHRASES = [
    (_re.compile(r"проб(?:ег|іг)\D{0,4}(\d[\d\s]{0,9})\s*(тыс\w*|тис\w*)?", _re.I),
     "mileage_km"),
    (_re.compile(r"(?:объ[её]м|обем|двигател\w*)\D{0,6}(\d{3,4})", _re.I), "engine_cc"),
    (_re.compile(r"(?:цена|стоимость|ціна)\D{0,6}(\d[\d\s]{2,11})", _re.I), "price_uah"),
    (_re.compile(r"цвет\s+([а-яёa-z\-]{3,15})", _re.I), "color"),
    (_re.compile(r"(?:кпп|коробк\w*)\s+([а-яёa-z]{3,12})", _re.I), "gearbox"),
    (_re.compile(r"привод\s+([а-яёa-z]{5,12})", _re.I), "drive"),
    (_re.compile(r"\b(бензин|дизель|гибрид|электро)\b", _re.I), "fuel"),
    (_re.compile(r"(?:до\s+киева|осталось|прибыти\w*|доставк\w*)\D{0,10}(\d{1,3})\s*"
                 r"(?:дн|дней|дня)", _re.I), "eta_days"),
]


def _orient(video):
    """Горизонтальное или вертикальное — по размерам файла.
    Используется, когда видео присылают после нажатия кнопки."""
    w = getattr(video, "width", 0) or 0
    h = getattr(video, "height", 0) or 0
    if not w or not h:
        return None
    return "video_h" if w >= h else "video_v"


async def auto_catch(msg, card, actor_id, context):
    """Кладёт присланное в открытую карточку само, без кнопок.
    Возвращает True, если разобралось; False — пусть идёт обычным путём."""
    if (context.user_data.get("car_wait") or {}).get("field") in ("price_uah", "price_georgia"):
        return False
    cid = card["id"]
    number = card.get("auto_number") or "#%d" % cid

    # ── фото, видео, документы: только по кнопке ──
    # Что нельзя опознать наверняка, бот сам в карточку не кладёт.
    # Нажмите «Фото и видео» или «Тех. состояние» и пришлите файлы туда.
    if msg.photo or msg.video or msg.video_note or msg.document:
        return False

    # ── текст ──
    text = (msg.text or "").strip()
    if not text:
        return False
    if VIN_RE.search(text):
        return False                        # новый VIN — это новая машина, не правка

    link = _re.search(r"https?://\S+", text)
    if link and len(text) < 200:
        set_field(cid, "diag_link", link.group(0), actor_id)
        await msg.reply_text("Ссылка на отчёт записана в %s." % number)
        return True

    golo = _re.fullmatch(r"[\d\s]{3,12}\s*[$₴]?", text)
    if golo:
        ok, answer = apply_value(cid, "price_uah", text, actor_id)
        if ok:
            await msg.reply_text(
                "%s · цена: %s" % (number, answer.replace("Записано: ", "")),
                reply_markup=InlineKeyboardMarkup([[
                    InlineKeyboardButton("← Вернуться к карточке",
                                         callback_data="car_open:%d" % cid)]]))
            return True

    # «запиши / добавь / поставь» — ИИ сам вносит данные в карточку
    try:
        import assistant
        if assistant.is_write(text):
            polya = assistant.extract(text, card)
            zapisano = []
            for field, value in (polya or {}).items():
                if field not in dict(EDITABLE) and field not in (
                        "brand", "model", "condition_text", "sea_container",
                        "diag_link", "fuel"):
                    continue
                ok, _ = apply_value(cid, field, str(value), actor_id)
                if ok:
                    zapisano.append(LABELS_ALL.get(field, field))
            if zapisano:
                await msg.reply_text(
                    "%s · записано: %s" % (number, ", ".join(zapisano)),
                    reply_markup=InlineKeyboardMarkup([[InlineKeyboardButton(
                        "← Вернуться к карточке",
                        callback_data="car_open:%d" % cid)]]))
                return True
    except Exception as e:
        log.warning("Запись через ИИ: %s", e)

    # свободная просьба — отвечает ИИ, а не «необработанное»
    try:
        import assistant
        if assistant.is_request(text):
            otvet = assistant.ask(text, card)
            if otvet:
                context.user_data["ai_otvet"] = otvet[:1500]
                await msg.reply_text(
                    assistant.clean_html(otvet)[:3800], parse_mode="HTML",
                    reply_markup=InlineKeyboardMarkup([
                        [InlineKeyboardButton("Записать в тех. состояние",
                                              callback_data="car_saveai:%d" % cid)],
                        [InlineKeyboardButton("← Вернуться к карточке",
                                              callback_data="car_open:%d" % cid)]]))
                return True
    except Exception as e:
        log.warning("Помощник: %s", e)

    for rx, field in PHRASES:
        m = rx.search(text)
        if not m:
            continue
        raw = (m.group(1) or "").strip()
        if not raw:
            continue
        if field == "mileage_km" and len(m.groups()) > 1 and m.group(2):
            try:
                if int(raw.replace(" ", "")) < 1000:
                    raw = str(int(raw.replace(" ", "")) * 1000)
            except ValueError:
                continue
        ok, answer = apply_value(cid, field, raw, actor_id)
        if ok:
            await msg.reply_text(
                "%s · %s" % (number, answer),
                reply_markup=InlineKeyboardMarkup([[
                    InlineKeyboardButton("← Вернуться к карточке",
                                         callback_data="car_open:%d" % cid)]]))
            return True
    return False

def stats_line():
    """Одна строка про новых клиентов — для шапки главного меню."""
    try:
        import visits
        dni = visits.new_days(4)
        podpisi = ["сегодня", "вчера"]
        chasti = []
        for i, (d, n) in enumerate(dni):
            imya = podpisi[i] if i < len(podpisi) else d.strftime("%d.%m")
            chasti.append("%s <b>%d</b>" % (imya, n) if i == 0
                          else "%s %d" % (imya, n))
        return "👥 Новые клиенты: " + " · ".join(chasti)
    except Exception as e:
        log.warning("Статистика не собралась: %s", e)
        return ""


def ai_warning():
    """Если ИИ-помощник не работает — предупреждаем сразу на главной."""
    try:
        import ai
    except Exception:
        return "⚠️ <b>ИИ-помощник не подключён</b> — разбор сообщений не работает."
    try:
        if not ai.enabled():
            return ("⚠️ <b>ИИ-помощник не работает</b> — нет ключа или связи. "
                    "Сообщения сохраняются, но не разбираются.")
        if hasattr(ai, "is_stopped") and ai.is_stopped():
            return "⚠️ <b>ИИ-помощник остановлен</b> — включите его, чтобы разбор шёл."
    except Exception as e:
        return "⚠️ <b>ИИ-помощник отвечает с ошибкой:</b> %s" % e
    return ""


def menu_header(text):
    """Строка аналитики и предупреждение об ИИ над главным меню."""
    parts = []
    line = stats_line()
    if line:
        parts.append(line)
    warn = ai_warning()
    if warn:
        parts.append(warn)
    parts.append(text or "")
    return "\n\n".join(p for p in parts if p)

def with_stats_button(markup):
    """Главное меню владельца: только рабочие кнопки, без «Ещё» и его разделов."""
    try:
        rows = list(markup.inline_keyboard)
    except Exception:
        return markup
    clean = []
    for row in rows:
        keep = [b for b in row
                if getattr(b, "callback_data", None) not in HIDE
                and getattr(b, "callback_data", None) not in MORE]
        if keep:
            clean.append(keep)
    return InlineKeyboardMarkup(clean)

def collect_svodka():
    """Собирает цифры по обоим ботам — их и разбирает ИИ."""
    L = ["СВОДКА ПО СОСТОЯНИЮ ДЕЛ", ""]

    try:
        cards = db.list_cards("cars", limit=200)
    except Exception:
        cards = []
    chernoviki = 0
    opublikovano = 0
    bez_foto = 0
    bez_ceny = 0
    po_etapam = {}
    propuski = {}
    for c in cards:
        c = card_of(c["id"]) or c
        miss = S.missing_required(c)
        if miss:
            chernoviki += 1
            for m in miss:
                propuski[m] = propuski.get(m, 0) + 1
        if c.get("published"):
            opublikovano += 1
        if not photos_of(c):
            bez_foto += 1
        if not c.get("price_uah"):
            bez_ceny += 1
        etap = S.status_label(c.get("status")) if c.get("status") else "этап не задан"
        po_etapam[etap] = po_etapam.get(etap, 0) + 1

    L.append("АВТОМОБИЛИ")
    L.append("Всего карточек: %d" % len(cards))
    L.append("Черновиков (не хватает обязательных полей): %d" % chernoviki)

def _v168_cas_write(card_id, field, expected_old, new_value, actor_id, correction=False):
    import re as _v168_re
    if not _v168_re.fullmatch(r"[a-z_][a-z0-9_]*", str(field or "")):
        return False, "invalid"
    if _v168_empty(new_value):
        return False, "empty"
    with db.connect() as _v168_con:
        columns = {row[1] for row in _v168_con.execute("PRAGMA table_info(cars)")}
        if field not in columns:
            return False, "unknown"
        row = _v168_con.execute(
            "SELECT %s FROM cars WHERE id=?" % field, (int(card_id),)).fetchone()
        if row is None:
            return False, "missing_card"
        current = row[0]
        if str(current or "") == str(new_value):
            return False, "same"
        if correction:
            if str(current or "") != str(expected_old or ""):
                return False, "conflict"
        elif not _v168_empty(current):
            return False, "filled"
        cursor = _v168_con.execute(
            "UPDATE cars SET %s=?,updated_at=? WHERE id=? AND %s IS ?" % (field, field),
            (new_value, db.now(), int(card_id), current))
        if cursor.rowcount != 1:
            return False, "conflict"
    try:
        db.log_action(actor_id, "card_edit", "cars", int(card_id),
                      field, current, new_value)
    except Exception as _v168_exc:
        log.warning("CRM CAS audit failed card=%s field=%s error=%s",
                    card_id, field, _v168_exc)
    return True, "applied"

async def _v168_ack(query, *args, **kwargs):
    try:
        import crm_online_guard as _v168_guard
        import asyncio as _v168_asyncio
        action = "crm_" + str(getattr(query, "data", "callback") or "callback").split(":", 1)[0]
        if args or kwargs:
            token, _started = _v168_guard.start_operation(action)
            try:
                await _v168_asyncio.wait_for(query.answer(*args, **kwargs), timeout=0.75)
                _v168_guard.finish_operation(token, "ack")
            except Exception as exc:
                _v168_guard.finish_operation(token, "continued", type(exc).__name__)
        else:
            token, _started = await _v168_guard.safe_callback_answer(query, action)
            _v168_guard.finish_operation(token, "ack")
    except Exception:
        try:
            await query.answer(*args, **kwargs)
        except Exception:
            pass

def voice_change_plan(card, data, allowed, override=False):
    """Return bounded field updates; filled values change only by explicit request."""
    changes = []
    skipped = []
    permitted = set(allowed or ())
    for field, value in (data or {}).items():
        if field not in permitted or value in (None, "", []):
            continue
        old = card.get(field)
        if old not in (None, "", 0, []) and not override:
            skipped.append(field)
            continue
        if str(old or "") != str(value):
            changes.append((field, old, value))
    return changes, skipped

def _v165_spool_enqueue(card_id, target, file_id, actor_id, tag="", unique_id=None, message_id=None):
    """Durable, idempotent acceptance; no SQLite write and no getFile wait."""
    import fcntl as _v165_fcntl
    import hashlib as _v165_hashlib
    import json as _v165_json
    import os as _v165_os
    import time as _v165_time
    identity = str(unique_id or file_id)
    event_key = _v165_key(card_id, target, identity)
    try:
        import crm_online_guard as _v168_guard
        if _v168_guard.media_seen(event_key):
            return {"state": "duplicate_saved", "queued": 0}
    except Exception:
        pass
    try:
        if str(file_id) in set(_v165_media_ids(card_of(card_id), target)):
            return {"state": "duplicate_saved", "queued": 0}
    except Exception:
        pass
    _v165_os.makedirs(_v165_os.path.dirname(_V165_SPOOL), exist_ok=True)
    event = {
        "key": event_key, "card_id": int(card_id), "target": str(target),
        "file_id": str(file_id), "actor_id": int(actor_id), "tag": str(tag or ""),
        "unique_id_sha256": _v165_hashlib.sha256(identity.encode()).hexdigest(),
        "message_id": int(message_id) if message_id is not None else None,
        "accepted_at": int(_v165_time.time()),
    }
    with open(_V165_SPOOL_LOCK, "a+", encoding="utf-8") as guard:
        _v165_fcntl.flock(guard.fileno(), _v165_fcntl.LOCK_EX)
        try:
            try:
                lines = open(_V165_SPOOL, "r", encoding="utf-8").readlines()
            except FileNotFoundError:
                lines = []
            keys, same = set(), 0
            for line in lines:
                try:
                    row = _v165_json.loads(line)
                except Exception:
                    continue
                keys.add(row.get("key"))
                if row.get("card_id") == int(card_id) and row.get("target") == str(target):
                    same += 1
            if event_key in keys:
                return {"state": "duplicate_queued", "queued": same}
            try:
                saved_count = len(set(_v165_media_ids(card_of(card_id), target)))
            except Exception:
                saved_count = 0
            if saved_count + same >= _v165_limit(target):
                return {"state": "limit", "queued": same}
            payload = (_v165_json.dumps(event, ensure_ascii=False, separators=(",", ":")) + "\n").encode()
            descriptor = _v165_os.open(_V165_SPOOL, _v165_os.O_WRONLY | _v165_os.O_CREAT | _v165_os.O_APPEND, 0o600)
            try:
                _v165_os.write(descriptor, payload)
                _v165_os.fsync(descriptor)
            finally:
                _v165_os.close(descriptor)
            return {"state": "accepted", "queued": same + 1, "key": event_key}
        finally:
            _v165_fcntl.flock(guard.fileno(), _v165_fcntl.LOCK_UN)



def _v165_spool_remove(keys):
    import fcntl as _v165_fcntl
    import json as _v165_json
    import os as _v165_os
    import tempfile as _v165_tempfile
    _v165_keys = set(keys)
    if not _v165_keys:
        return
    with open(_V165_SPOOL_LOCK, "a+", encoding="utf-8") as _v165_guard:
        _v165_fcntl.flock(_v165_guard.fileno(), _v165_fcntl.LOCK_EX)
        try:
            try:
                with open(_V165_SPOOL, "r", encoding="utf-8") as _v165_file:
                    _v165_lines = _v165_file.readlines()
            except FileNotFoundError:
                return
            _v165_keep = []
            for _v165_line in _v165_lines:
                try:
                    if _v165_json.loads(_v165_line).get("key") in _v165_keys:
                        continue
                except Exception:
                    pass
                _v165_keep.append(_v165_line)
            _v165_dir = _v165_os.path.dirname(_V165_SPOOL)
            _v165_fd, _v165_tmp = _v165_tempfile.mkstemp(prefix=".media-spool-", dir=_v165_dir)
            try:
                with _v165_os.fdopen(_v165_fd, "w", encoding="utf-8") as _v165_file:
                    _v165_file.writelines(_v165_keep)
                    _v165_file.flush()
                    _v165_os.fsync(_v165_file.fileno())
                _v165_os.replace(_v165_tmp, _V165_SPOOL)
            finally:
                try:
                    _v165_os.unlink(_v165_tmp)
                except FileNotFoundError:
                    pass
        finally:
            _v165_fcntl.flock(_v165_guard.fileno(), _v165_fcntl.LOCK_UN)


def _v166_drain_owned(max_items=100):
    """Drain once; mark a durable receipt only after successful CRM save."""
    rows = _v165_read_rows()[:max(1, int(max_items))]
    done, error = [], ""
    for row in rows:
        try:
            card = card_of(row["card_id"])
            if not card:
                error = "card unavailable"
                continue
            answer = save_media(card, row["target"], row["file_id"],
                                row["actor_id"], row.get("tag", ""))
            if answer and any(word in str(answer).casefold() for word in ("не принят", "не проверен")):
                error = str(answer)[:160]
                continue
            done.append(row["key"])
            try:
                import crm_online_guard as _v168_guard
                _v168_guard.media_mark(row["key"], row["card_id"], row["target"])
            except Exception:
                pass
        except Exception as exc:
            error = "%s: %s" % (type(exc).__name__, exc)
            if "locked" in str(exc).casefold() or "busy" in str(exc).casefold():
                break
    _v165_spool_remove(done)
    return {"processed": len(done), "remaining": len(_v165_read_rows()),
            "error": error, "busy": False}



def _v165_spool_drain(max_items=100):
    """Exactly one worker may mutate card media at a time."""
    import fcntl as _v166_fcntl
    with open(_V165_DRAIN_LOCK, "a+", encoding="utf-8") as _v166_guard:
        try:
            _v166_fcntl.flock(
                _v166_guard.fileno(), _v166_fcntl.LOCK_EX | _v166_fcntl.LOCK_NB)
        except BlockingIOError:
            return {"processed": 0, "remaining": len(_v165_read_rows()),
                    "error": "", "busy": True}
        try:
            return _v166_drain_owned(max_items)
        finally:
            _v166_fcntl.flock(_v166_guard.fileno(), _v166_fcntl.LOCK_UN)

            "limit": _v165_limit(target), "session_saved": session_saved,
            "session_queued": len(queued_ids & session)}

def _v167_voice_explicit_fields(text, allowed):
    """Deterministic RU/UA/EN automotive fields; ambiguous input writes nothing."""
    import re as _v167_re
    value = str(text or "").casefold().replace("ё", "е")
    for words in ("две тысячи", "дві тисячі", "дві тисячі", "two thousand"):
        value = value.replace(words, "2000")
    result, permitted = {}, set(allowed or ())
    if "fuel" in permitted:
        fuel = None
        if _v167_re.search(r"(?<![a-zа-яіїєґ0-9])(?:l\s*p\s*i|л\s*п\s*и|[еэ]л\s*пи\s*ай)(?![a-zа-яіїєґ0-9])", value):
            fuel = "LPI"
        elif _v167_re.search(r"(?<![a-zа-яіїєґ0-9])(?:l\s*p\s*g|л\s*п\s*г|[еэ]л\s*пи\s*джи)(?![a-zа-яіїєґ0-9])", value):
            fuel = "LPG"
        elif _v167_re.search(r"\b(?:дизел[ьяь]?|дизельне|diesel)\b", value):
            fuel = "diesel"
        elif _v167_re.search(r"\b(?:бензин|бензинов|gasoline|petrol)\b", value):
            fuel = "gasoline"
        elif _v167_re.search(r"\b(?:гибрид|гібрид|hybrid)\b", value):
            fuel = "hybrid"
        elif _v167_re.search(r"\b(?:электро|електро|electric)\b", value):
            fuel = "electric"
        elif _v167_re.search(r"(?:топлив|палив|fuel)\s*(?:[:=\-–—]|тип)?\s*газ\b", value):
            fuel = "LPG"
        if fuel:
            result["fuel"] = fuel
    if "engine_cc" in permitted:
        label = r"(?:объ?ем(?:\s+двигателя)?|об['’]?єм(?:\s+двигуна)?|двигател[ья]|двигун[ау]?|мотор[ау]?|engine(?:\s+volume)?)"
        match = (_v167_re.search(label + r"\s*(?:[:=\-–—]|составляет|становить|is)?\s*(\d+(?:[.,]\d+)?)", value)
                 or _v167_re.search(r"(\d+(?:[.,]\d+)?)\s*(?:см3|см³|cc|куб(?:ов)?)?\s*" + label, value))
        if match:
            try:
                number = float(match.group(1).replace(",", "."))
                cc = int(round(number * 1000 if number < 20 else number))
                if 400 <= cc <= 12000:
                    result["engine_cc"] = cc
            except Exception:
                pass
    if "color" in permitted:
        colors = {
            "бел": "белый", "білий": "белый", "біла": "белый", "white": "белый",
            "черн": "чёрный", "чорн": "чёрный", "black": "чёрный",
            "серебр": "серебристый", "сріб": "серебристый", "silver": "серебристый",
            "сер": "серый", "сір": "серый", "gray": "серый", "grey": "серый",
            "красн": "красный", "червон": "красный", "red": "красный",
            "син": "синий", "blue": "синий", "зелен": "зелёный", "green": "зелёный",
            "желт": "жёлтый", "жовт": "жёлтый", "yellow": "жёлтый",
            "беж": "бежевый", "beige": "бежевый", "корич": "коричневый", "brown": "коричневый",
        }
        match = _v167_re.search(
            r"(?:(?:цвет|колір|color)\s*(?:[:=\-–—]|автомобиля|авто|машины|car)?\s*([a-zа-яіїєґ]+)|"
            r"([a-zа-яіїєґ]+)\s+(?:цвет|колір|color))", value)
        if match:
            word = next((item for item in match.groups() if item), "")
            for prefix, normalized in colors.items():
                if word.startswith(prefix):
                    result["color"] = normalized
                    break
    return result

    """UA-V104 · файлы, память, пересборка и проверка HTML.

    Возвращает (сколько файлов убрано, список сбоев). Пустой список сбоев
    значит, что на собранной странице ссылок на эти ролики больше нет.
    """
    import glob as _g
    import json as _j
    import re as _re

    sboi = []
    nom = str(nomer or "").upper()
    if not nom:
        return 0, ["у карточки нет номера машины"]

    papka = os.path.join("/home/Carix", "video")
    puti = (sorted(_g.glob(os.path.join(papka, "%s.mp4" % nom)))
            + sorted(_g.glob(os.path.join(papka, "%s-*.mp4" % nom))))
    ubrano = 0
    for put in puti:
        try:
            os.remove(put)
            ubrano += 1
        except Exception as e:
            sboi.append("не стёрся файл %s: %s" % (os.path.basename(put), e))
        try:
            if os.path.exists(put + ".poster.jpg"):
                os.remove(put + ".poster.jpg")
        except Exception:
            pass

    try:
        pamyat = "/home/Carix/.video_sinhron.json"
        if os.path.exists(pamyat):
            d = _j.load(open(pamyat, encoding="utf-8"))
            if nom in d:
                d.pop(nom)
                with open(pamyat, "w", encoding="utf-8") as f:
                    _j.dump(d, f, ensure_ascii=False)
    except Exception as e:
        sboi.append("не вышло почистить список синхронизации: %s" % e)

    if not _peresobrat_stranicy():
        sboi.append("страницы сайта не пересобрались")

        put_html = os.path.join(papka, "%s.html" % nom)

def _v163_status(nomer, novyj, staryj=None):
    """Меняет статус записей видео машины и пересобирает старое поле."""
    zatronuto = []
    try:
        con = _s163.connect("/home/Carix/crm.db", timeout=20)
        con.row_factory = _s163.Row
        car = con.execute("SELECT id FROM cars WHERE auto_number=?", (nomer,)).fetchone()
        if not car:
            return []
        usloviye = "AND status=?" if staryj else ""
        parametry = [car["id"]] + ([staryj] if staryj else [])
        for r in con.execute("SELECT id, file_id, tag FROM media WHERE car_id=? AND vid='video' "
                             + usloviye, parametry):
            con.execute("UPDATE media SET status=? WHERE id=?", (novyj, r["id"]))
            zatronuto.append(r["id"])
        spisok = [{"file_id": r["file_id"], "tag": r["tag"] or ""}
                  for r in con.execute("SELECT file_id, tag FROM media WHERE car_id=? "
                                       "AND vid='video' AND status='ready' ORDER BY poryadok",
                                       (car["id"],))]
        import json as _j163
        con.execute("UPDATE cars SET videos=? WHERE id=?",
                    (_j163.dumps(spisok, ensure_ascii=False), car["id"]))
        con.commit()
        con.close()
    except Exception as e:
        try:
            _v140_zapis("media: смена статуса видео %s не удалась — %r" % (nomer, e))
        except Exception:
            pass
    return zatronuto


def _ubrat_video_polno(nomer):
    nom = str(nomer or "").upper()
    _v163_status(nom, "rejected", "ready")          # сначала таблица
    ubrano, sboi = _v163_ubrat_ishodnyy(nomer)      # потом файлы и пересборка
    if sboi:
        _v163_status(nom, "ready", "rejected")      # не вышло — вернули как было
        sboi.append("записи в таблице возвращены в прежний вид")
    return ubrano, sboi

def _ua083_publish_preimage(card):
    return {name: card.get(name) for name in ("published", "status", "publish_pending")}


def _ua083_restore_publish_preimage(cid, preimage, actor_id):
    try:
        for field in ("published", "status", "publish_pending"):
            current = card_of(cid) or {}
            if current.get(field) != preimage.get(field):
                db.update_card_field("cars", int(cid), field, preimage.get(field), actor_id)
        final = card_of(cid) or {}
        return all(final.get(field) == preimage.get(field)
                   for field in ("published", "status", "publish_pending"))
    except Exception as exc:
        log.error("TASK083: не удалось восстановить preimage %s: %s", cid, exc)
        return False

# UA-ART-LEGACY-EMERGENCY-SPEC-20260910: automatic queue handoff only.
_ua_emergency_spec_tasks = set()

async def _ua_emergency_enqueue_spec(uid):
    import asyncio as _ua_emergency_asyncio
    try:
        queued = await _ua_emergency_asyncio.to_thread(_ua110_vin_service.retry_card, uid)
        if not queued:
            log.warning("Автоматический сбор спецификации не поставлен в очередь: %s", uid)
    except Exception:
        # A supplier/queue failure cannot undo an already completed publication.
        log.exception("Автоматический сбор спецификации не запущен: %s", uid)

def _ua_emergency_schedule_spec(uid):
    import asyncio as _ua_emergency_asyncio
    coro = _ua_emergency_enqueue_spec(uid)
    try:
        task = _ua_emergency_asyncio.create_task(coro)
    except BaseException:
        coro.close()
        raise
    _ua_emergency_spec_tasks.add(task)
    task.add_done_callback(_ua_emergency_spec_tasks.discard)

    after = _UA122_CATALOG.stat()

            after.st_ino, after.st_size, after.st_mtime_ns):
        raise ValueError('Catalog changed while reading')
    return _ua122_parse_catalog(source)

def _ua_oneclick_verified(receipt):
    from ua_crm_public_sync import snapshot
    from ua_public_freshness import verify_public
    revision = snapshot().get(receipt['identity'])
    if not revision or revision['delivery_status'] == 'hidden':
        return False
    if revision['sha256'] != receipt['verified_revision']:
        return False
    verify_public(receipt['code'])
    return snapshot().get(receipt['identity']) == revision

async def _ua_oneclick_receipts(context):
    import asyncio
    from ua_crm_public_sync import ROOT, snapshot, notify, start
    import ua_publish_requests as requests
    start()
    for receipt in await asyncio.to_thread(requests.receipts, ROOT):
        try:
            live = await asyncio.to_thread(requests.current, ROOT, receipt['token'])
            if live != receipt:
                continue
            if receipt['state'] == 'verified':
                try:
                    fresh = await asyncio.to_thread(_ua_oneclick_verified, receipt)
                except Exception:
                    fresh = False
                if not fresh:
                    revision = (await asyncio.to_thread(snapshot)).get(receipt['identity'])
                    if revision and revision['delivery_status'] != 'hidden':
                        await asyncio.to_thread(requests.enqueue, ROOT, receipt['identity'],
                            revision, receipt['chat_id'], receipt['message_id'])
                        notify()
                        continue
                    text = 'Публикация отменена: карточка удалена, скрыта или перенесена в архив.'
                else:
                    text = ('✅ Объявление опубликовано на сайте.\n'
                            'https://www.uaart.com.ua/video/%s.html' % receipt['code'])
            else:
                text = 'Публикация отменена: карточка удалена, скрыта или перенесена в архив.'
            # Recheck after network verification so a newer request is not acknowledged.
            if await asyncio.to_thread(requests.current, ROOT, receipt['token']) != receipt:
                continue
            try:
                await context.bot.edit_message_text(text=text,
                    chat_id=receipt['chat_id'], message_id=receipt['message_id'],
                    reply_markup=_ua099_back(int(receipt['identity'])),
                    disable_web_page_preview=True)
            except Exception as exc:
                if 'message is not modified' not in str(exc).lower():
                    raise
            await asyncio.to_thread(requests.delivered, ROOT, receipt['token'])
            if receipt['state'] == 'verified' and fresh:
                _ua_emergency_schedule_spec(receipt['code'])
        except Exception:
            # A Telegram outage never undoes a publication or loses its receipt.
            log.exception('Publication receipt delivery will retry')
