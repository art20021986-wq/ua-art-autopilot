def kadry_ne_dokachany(est, pole):
    """Сколько кадров диагностики в базе и сколько файлов на диске."""
    if not est or 'condition_photos' not in est:
        return []
    berem = [k for k in (pole, 'id', 'condition_photos', 'condition_videos') if k and k in est]
    try:
        soed = sqlite3.connect(BD, timeout=20)
        soed.row_factory = sqlite3.Row
        ryady = soed.execute('SELECT ' + ', '.join(berem) + ' FROM cars').fetchall()
        soed.close()
    except Exception:
        return []
    otstayut = []
    for ryad in ryady:
        nom = ''
        if pole:
            try:
                nom = str(ryad[pole] or '').strip()
            except Exception:
                nom = ''
        if not nom:
            try:
                nom = 'UA-%04d' % int(ryad['id'])
            except Exception:
                continue
        try:
            v_baze = len(json.loads(ryad['condition_photos'] or '[]'))
        except Exception:
            v_baze = 0
        if not v_baze:
            continue
        papka = os.path.join(PAPKA_VID, 'diag', nom)
        na_diske = len(glob.glob(os.path.join(papka, '*.jpg')))
        if na_diske < v_baze:
            otstayut.append((nom, v_baze, na_diske))
    return otstayut

def perekodirovat(put):
    """Переписывает файл в вид, который играет везде."""
    vremenno = put + '.novoe.mp4'
    komanda = ['ffmpeg', '-y', '-i', put, '-c:v', 'libx264', '-preset', 'veryfast', '-crf', KACHESTVO, '-profile:v', 'high', '-level', '4.0', '-pix_fmt', 'yuv420p', '-vf', "scale='min(1920,iw)':'min(%d,ih)':force_original_aspect_ratio=decrease,scale=trunc(iw/2)*2:trunc(ih/2)*2" % VYSOTA, '-c:a', 'aac', '-b:a', '128k', '-ac', '2', '-movflags', '+faststart', vremenno]
    try:
        itog = subprocess.run(komanda, capture_output=True, text=True, timeout=1800)
    except Exception as e:
        log('перекодировать не вышло: ' + repr(e)[:120])
        try:
            os.remove(vremenno)
        except Exception:
            pass
        return False
    if itog.returncode != 0 or not os.path.exists(vremenno) or os.path.getsize(vremenno) < 20000:
        hvost = (itog.stderr or '').strip().split('\n')[-1:]
        log('ffmpeg отказался: ' + ' '.join(hvost)[:150])
        try:
            os.remove(vremenno)
        except Exception:
            pass
        return False
    iskhodnoe = put + '.iskhodnoe'
    try:
        if not os.path.exists(iskhodnoe):
            shutil.copy2(put, iskhodnoe)
    except Exception:
        pass
    os.replace(vremenno, put)
    return True
