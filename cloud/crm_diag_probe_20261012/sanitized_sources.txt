def zastavka(otn):
    """Возвращает poster='...', если рядом с роликом лежит его первый кадр."""

        put = os.path.join(PAPKA_VID, otn + '.poster.jpg')
        if os.path.exists(put) and os.path.getsize(put) > 1000:
            return " poster='%s.poster.jpg'" % otn
    except Exception:

    return ''

def kadry_diagnostiki(m):
    """Фото и видео проверки из /home/Carix/video/diag/UA-XXXX/."""
    nom = nomer(m)
    papka = os.path.join(PAPKA_VID, 'diag', nom)
    if not os.path.isdir(papka):
        return ([], [])
    foto = []
    for obraz in ('*.jpg', '*.jpeg', '*.JPG', '*.png'):
        foto += glob.glob(os.path.join(papka, obraz))
    foto = sorted(set(foto))
    foto = [p for p in foto if os.path.getsize(p) > 1000][:50]
    video = sorted(glob.glob(os.path.join(papka, '*.mp4')))[:10]
    otn_foto = ['diag/%s/%s' % (nom, os.path.basename(x)) for x in foto]
    otn_video = ['diag/%s/%s' % (nom, os.path.basename(x)) for x in video]
    return (otn_foto, otn_video)

def zapisat(imya_bez, soderzhimoe):
    """Пишем во все копии: обе папки, все адреса с хэшем."""
    try:
        _sh_imya = os.path.basename(str(imya_bez)).strip().lower()
        if _sh_imya.replace('.html', '') in ('index', 'katalog', 'podbor', 'info'):
            try:
                with io.open('/home/Carix/shchit_log.txt', 'a', encoding='utf-8') as _sh_zh:
                    _sh_zh.write('%s  запрет zapisat: %s\n' % (time.strftime('%Y-%m-%d %H:%M:%S'), imya_bez))
            except Exception:
                pass
            return True
    except Exception:
        pass
    from site_ge_inject import inject_html as _ua_site_ge_inject_html
    soderzhimoe = _ua_site_ge_inject_html(soderzhimoe)
    celi = []
    for papka in (PAPKA_VID, PAPKA_SITE):
        os.makedirs(papka, exist_ok=True)
        celi.append(os.path.join(papka, imya_bez + '.html'))
        for _variant in glob.glob(os.path.join(papka, imya_bez + '-*.html')):
            _base = os.path.basename(_variant)
            if str(imya_bez).startswith('UA-') and (not str(imya_bez).endswith('-diag')) and _base.startswith(str(imya_bez) + '-diag'):
                continue
            celi.append(_variant)
    for put in set(celi):
        try:
            with io.open(put, 'w', encoding='utf-8') as f:
                f.write(soderzhimoe)
        except Exception:
            continue
    return len(set(celi))

def obnovit_etalon(imya_bez, soderzhimoe):
    """Если стоит замок вида — закрепляем в нём новую страницу, иначе он её откатит."""
    papka = os.path.join(BAZA_DIR, 'etalon')
    if not os.path.isdir(papka):
        return 0
    from site_ge_inject import inject_html as _ua_site_ge_inject_html
    soderzhimoe = _ua_site_ge_inject_html(soderzhimoe)
    skolko = 0
    for put in glob.glob(os.path.join(papka, '**', imya_bez + '*.html'), recursive=True):
        _base = os.path.basename(put)
        if str(imya_bez).startswith('UA-') and (not str(imya_bez).endswith('-diag')) and _base.startswith(str(imya_bez) + '-diag'):
            continue
        if 'otklonennye' in put:
            continue
        try:
            with io.open(put, 'w', encoding='utf-8') as f:
                f.write(soderzhimoe)
            skolko += 1
        except Exception:
            continue
    return skolko
