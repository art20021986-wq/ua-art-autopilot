async def delete_ask(update: Update, context: ContextTypes.DEFAULT_TYPE):
    adapter = context.application.bot_data.get('ua_delete_recovery_002')
    if adapter is None:
        await _v168_ack(update.callback_query)
        await update.callback_query.message.reply_text(
            'Удаление временно недоступно: проверяемый механизм не подключён. Карточка не удалена.')
        raise ApplicationHandlerStop
    await adapter.delete_ask(update, context)
