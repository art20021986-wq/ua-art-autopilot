def stage_po_statusu(status):
    s = (status or "").strip().lower()
    if s.startswith("ua_"):
        return "kiev"
    if s.startswith("ge_"):
        return "georgia"
    if s.startswith("sea_"):
        return "sea"
    if s.startswith("kr_"):
        return "korea"
    return ""

def _json_fajl(put):
    try:
        with io.open(put, "r", encoding="utf-8") as f:
            return json.load(f)
    except Exception:
        return {}

def _zapisat_json(put, d):
    try:
        vrem = put + ".tmp"
        with io.open(vrem, "w", encoding="utf-8") as f:
            f.write(json.dumps(d, ensure_ascii=False, indent=1))
        os.replace(vrem, put)
    except Exception:
        pass

LINII = [
    (("ONEY", "ONEU", "NYKU", "MOLU", "KKLU"), "ONE",
     "[URL]"),
]

def tracking(nomer):
    n = (nomer or "").strip().upper().replace(" ", "")
    if not n:
        return (None, None)
    ruchnoy = _json_fajl(TRACK)
    if n in ruchnoy and ruchnoy[n]:
        return ("CRM", ruchnoy[n])
    for prefiksy, imya, shablon in LINII:
        for pref in prefiksy:
            if n.startswith(pref):
                hvost = n[4:] if (imya == "ONE" and n.startswith("ONEY") and len(n) > 12) else n
                return (imya, shablon % hvost)
    return (None, None)

def _razmer(put):
    try:
        from PIL import Image
        with Image.open(put) as im:
            return im.size
    except Exception:
        return (0, 0)

def fokus_fajla(put):
    """Точка фокуса: центр масс детализации (кузов детальный, небо и асфальт ровные)."""
    kesh = _json_fajl(FOKUS)
    klyuch = put.replace(DOM, "")
    try:
        metka = "%d" % int(os.path.getmtime(put))
    except Exception:
        metka = "0"
    if kesh.get(klyuch, {}).get("m") == metka:
        return (kesh[klyuch]["x"], kesh[klyuch]["y"])
    x, y = 50, 50
    try:
        from PIL import Image
        import numpy as np
        with Image.open(put) as im:
            im = im.convert("L")
            im.thumbnail((180, 180))
            a = np.asarray(im, dtype="float32")
        e = np.abs(np.diff(a, axis=1))[:-1, :] + np.abs(np.diff(a, axis=0))[:, :-1]
        m = (e > float(np.median(e)) * 1.6 + 1.0).astype("float32")
        if m.sum() > 20:
            h, w = m.shape
            ys, xs = np.nonzero(m)
            x = max(20, min(80, int(round(float(xs.mean()) / max(w - 1, 1) * 100))))
            y = max(25, min(75, int(round(float(ys.mean()) / max(h - 1, 1) * 100))))
    except Exception:
        pass
    kesh[klyuch] = {"x": x, "y": y, "m": metka}
    _zapisat_json(FOKUS, kesh)
    return (x, y)

def kandidat_fasada(kod):
    """ПОДСКАЗКА для владельца, а не решение: горизонтальный крупный кадр.
    Никогда не используется как «фасад» — перед от зада без модели не отличить."""
    papka = os.path.join(VIDEO, "foto", kod)
    if not os.path.isdir(papka):
        return ""
    luchshee, ball = "", -1.0
    for f in sorted(os.listdir(papka))[:16]:
        if f.startswith(".") or not f.lower().endswith((".jpg", ".jpeg", ".png", ".webp")):
            continue
        put = os.path.join(papka, f)
        w, h = _razmer(put)
        if not w or not h or float(w) / float(h) < 1.15:
            continue
        b = min(float(w * h) / 1200000.0, 1.5)
        try:
            b += min(os.path.getsize(put) / 250000.0, 1.0)
        except Exception:
            pass
        if b > ball:
            luchshee, ball = f, b
    return luchshee

    """1) cover_photo из CRM (если менеджер задал), 2) первое фото автомобиля.
    Правило одинаково для существующих и всех будущих карточек."""
    papka = os.path.join(VIDEO, "foto", kod)
    if not os.path.isdir(papka):
        return ("001.jpg", "папки фото нет")
    fajly = sorted([f for f in os.listdir(papka)
                    if not f.startswith(".")
                    and f.lower().endswith((".jpg", ".jpeg", ".png", ".webp"))])
    if not fajly:
        return ("001.jpg", "фотографий нет")
    if m is None:
        m = dannye(kod)
    from photo_visibility import visible_names
    fajly = visible_names(m if m is not None else dannye(kod), fajly)
    if not fajly:
        raise ValueError("No visible photo available for cover")
    cover = os.path.basename((m.get("cover_photo") or "").strip()) if m else ""
    if cover and cover in fajly:
        return (cover, "cover_photo из CRM")
    return (fajly[0], "первое фото автомобиля")

def zakrepit_pervoe_foto_v_crm(kody):
    """Пишет главное фото в cars.cover_photo, чтобы источник истины был один
    и повторная публикация его не меняла. Меняется только эта колонка."""
    sdelano = []
    try:
        con = sqlite3.connect(BAZA)
        cur = con.cursor()
        for kod in kody:
            imya = pervoe_foto(kod)
            if not imya:
                continue
            cur.execute("select cover_photo from cars where auto_number=?", (kod,))
            r = cur.fetchone()
            tek = os.path.basename((r[0] or "").strip()) if r and r[0] else ""
            if tek == imya:
                continue
            cur.execute("update cars set cover_photo=? where auto_number=?", (imya, kod))
            if cur.rowcount:
                sdelano.append("%s → %s" % (kod, imya))
        con.commit()
        con.close()
    except Exception as e:
        return ["ошибка записи: %s" % e]
    return sdelano

def bbox_obekta(put):
    """Прямоугольник, в котором лежит сам автомобиль (доли 0..1).
    Кузов детальный, небо и асфальт ровные — берём границы массы деталей."""
    try:
        from PIL import Image
        import numpy as np
        with Image.open(put) as im:
            im = im.convert("L")
            im.thumbnail((260, 260))
            a = np.asarray(im, dtype="float32")
    except Exception:
        return (0.0, 0.0, 1.0, 1.0)
    h, w = a.shape
    if w < 30 or h < 30:
        return (0.0, 0.0, 1.0, 1.0)
    e = np.abs(np.diff(a, axis=1))[:-1, :] + np.abs(np.diff(a, axis=0))[:, :-1]
    porog = float(np.median(e)) * 1.5 + 1.0
    m = (e > porog).astype("float32")
    if m.sum() < 30:
        return (0.0, 0.0, 1.0, 1.0)
    def granicy(proekciya, dolya=0.04):
        s = proekciya.cumsum()
        vsego = s[-1]
        if vsego <= 0:
            return (0.0, 1.0)
        import numpy as np
        n0 = int(np.searchsorted(s, vsego * dolya))
        n1 = int(np.searchsorted(s, vsego * (1 - dolya)))
        return (n0 / float(len(proekciya)), max(n1, n0 + 1) / float(len(proekciya)))
    x0, x1 = granicy(m.sum(axis=0))
    y0, y1 = granicy(m.sum(axis=1))
    return (x0, y0, x1, y1)

def _cvet_kraya(im, storona):
    """Средний цвет крайней полосы снимка — им заполняется свободное поле.
    Это не blur и не дорисовка: сплошной цвет фона того же кадра."""
    w, h = im.size
    polosa = im.crop((0, 0, w, max(int(h * 0.06), 1))) if storona == "verh" else \
             im.crop((0, h - max(int(h * 0.06), 1), w, h)) if storona == "niz" else \
             im.crop((0, 0, max(int(w * 0.06), 1), h)) if storona == "levo" else \
             im.crop((w - max(int(w * 0.06), 1), 0, w, h))
    malenkiy = polosa.resize((1, 1))
    return malenkiy.getpixel((0, 0))

def sdelat_vitrinnoe(kod, imya_fajla):
    """РЕЖИМ C — реальное кадрирование: из снимка вырезается окно точно в
    пропорции витринного блока, центрированное по автомобилю. Кадр заполняет
    окно целиком, срезается только фон. Ни заливки краёв, ни полей, ни blur."""
    ishod = os.path.join(VIDEO, "foto", kod, imya_fajla)
    if not os.path.exists(ishod):
        return ("", {"ok": False, "prichina": "файла нет"})
    try:
        os.makedirs(STAGE)
    except Exception:
        pass
    try:
        from PIL import Image
        with Image.open(ishod) as im:
            im = im.convert("RGB")
            w, h = im.size
            aspekt = float(CEL_W) / float(CEL_H)
            bx0, by0, bx1, by1 = bbox_obekta(ishod)
            px0, py0, px1, py1 = bx0 * w, by0 * h, bx1 * w, by1 * h
            # максимально возможное окно нужных пропорций внутри снимка
            if w / float(h) >= aspekt:
                ch = float(h)
                cw = ch * aspekt
            else:
                cw = float(w)
                ch = cw / aspekt
            cx, cy = (px0 + px1) / 2.0, (py0 + py1) / 2.0
            left = min(max(cx - cw / 2.0, 0.0), max(w - cw, 0.0))
            top = min(max(cy - ch / 2.0, 0.0), max(h - ch, 0.0))
            srezano = 1.0 - (cw * ch) / float(w * h)
            vidno_x = max(0.0, min(px1, left + cw) - max(px0, left))
            vidno_y = max(0.0, min(py1, top + ch) - max(py0, top))
            ploshad = max((px1 - px0) * (py1 - py0), 1.0)
            srez_kuzova = 1.0 - (vidno_x * vidno_y) / ploshad
            kadr = im.crop((int(left), int(top),
                            int(left + cw), int(top + ch))).resize(
                                (CEL_W, CEL_H), Image.LANCZOS)
            put_jpg = os.path.join(STAGE, "%s.jpg" % kod)
            put_webp = os.path.join(STAGE, "%s.webp" % kod)
            kadr.save(put_jpg, quality=86, optimize=True, progressive=True)
            try:
                kadr.save(put_webp, quality=84, method=5)
            except Exception:
                put_webp = ""
    except Exception as e:
        return ("", {"ok": False, "prichina": "%s" % e})
    legkiy = put_webp or put_jpg
    return ("/video/stage/%s" % os.path.basename(legkiy),
            {"ok": True, "celikom": srez_kuzova < 0.02, "bez_poley": True,
             "dolya_polya": 0, "srez": round(srezano * 100),
             "srez_kuzova": round(srez_kuzova * 100), "ishod": imya_fajla,
             "ves": os.path.getsize(legkiy), "razmer": "%dx%d" % (CEL_W, CEL_H)})

def podgotovit_vitrinu(kod, m=None):
    """ЖИВОЕ ФОТО = ПЕРВОЕ ФОТО АВТОМОБИЛЯ. Никакого перебора кадров,

    if m is None:

    imya, istochnik = vybrat_glavnoe(kod, m)
    pervoe = pervoe_foto(kod)
    put, otchet = sdelat_vitrinnoe(kod, imya)
    otchet["foto"] = imya
    otchet["pervoe_foto"] = pervoe
    otchet["istochnik"] = istochnik
    otchet["sovpadaet_s_pervym"] = bool(imya and imya == pervoe)
    otchet["podtverzhden"] = True
    return (put, otchet)

def stil_fokusa(kod, imya_fajla, vitrina=False, vazhno=False):
    """Кадр витрины уже подготовлен точно под окно, поэтому здесь только

    put = os.path.join(VIDEO, "foto", kod, imya_fajla or "")

def _pravka_tega_foto(teg, kod, glav, vitrina):
    src = re.search(r'src=["\']([^"\']*/foto/(UA-\d{4})/([^"\']+))["\']', teg)
    if not src:
        return teg
    tek = src.group(3)
    if glav and glav != tek:
        teg = teg.replace("/foto/%s/%s" % (kod, tek), "/foto/%s/%s" % (kod, glav))
    if "mcf-foto" in teg:
        return teg
    stil = stil_fokusa(kod, glav or tek, vitrina=vitrina, vazhno=vitrina)
    m = re.search(r'style=["\']([^"\']*)["\']', teg)
    if m:
        staryy = re.sub(r"object-(fit|position)\s*:[^;]*;?", "", m.group(1))
        teg = teg.replace(m.group(0), 'style="%s;%s" data-mcf-foto="1"'
                          % (staryy.rstrip(";"), stil), 1)
    else:
        teg = teg[:-1].rstrip("/") + ' style="%s" data-mcf-foto="1">' % stil
    return teg

def _dhash(put_kadra):
    try:
        from PIL import Image
        import numpy as np
        with Image.open(put_kadra) as im:
            im = im.convert("L").resize((9, 8))
            a = np.asarray(im, dtype="int16")
        bity = (a[:, 1:] > a[:, :-1]).flatten()
        v = 0
        for b in bity:
            v = (v << 1) | int(b)
        return v
    except Exception:
        return None

def _dlitelnost(put):
    try:
        r = subprocess.run(["ffprobe", "-v", "error", "-show_entries", "format=duration",
                            "-of", "default=nw=1:nk=1", put],
                           capture_output=True, timeout=25)
        return float((r.stdout or b"0").decode().strip() or 0)
    except Exception:
        return 0.0

def _put_iz_src(src):
    if src.startswith("/video/"):
        return os.path.join(DOM, src.lstrip("/"))
    return src if src.startswith("/") else os.path.join(VIDEO, src)

def _rasstoyanie(a, b):
    return bin(a ^ b).count("1")

def otpechatki_kadrov(put):
    """Пять кадров: начало, 25 %, 50 %, 75 %, конец — устойчиво к пережатию."""
    d = _dlitelnost(put)
    if d <= 0:
        return []
    rez = []
    papka = tempfile.mkdtemp(prefix="mcf_kadry_")
    try:
        for dolya in (0.05, 0.25, 0.5, 0.75, 0.95):
            kadr = os.path.join(papka, "k%d.jpg" % int(dolya * 100))
            try:
                subprocess.run(["ffmpeg", "-nostdin", "-v", "error", "-ss", "%.2f" % (d * dolya),
                                "-i", put, "-frames:v", "1", "-y", kadr],
                               capture_output=True, timeout=40)
            except Exception:
                continue
            if os.path.exists(kadr):
                h = _dhash(kadr)
                if h is not None:
                    rez.append(h)
    finally:
        try:
            for f in os.listdir(papka):
                os.remove(os.path.join(papka, f))
            os.rmdir(papka)
        except Exception:
            pass
    return rez

def otpechatok_video(put):
    h = hashlib.sha256()
    try:
        with io.open(put, "rb") as f:
            while True:
                b = f.read(524288)
                if not b:
                    break
                h.update(b)
        return h.hexdigest()
    except Exception:
        return None

def sodejatelno_odno(put1, put2):
    """True — практически один и тот же ролик (пережатый, обрезанный по краям и т. п.)."""
    if otpechatok_video(put1) == otpechatok_video(put2):
        return True
    k1, k2 = otpechatki_kadrov(put1), otpechatki_kadrov(put2)
    if len(k1) < 3 or len(k2) < 3:
        return False
    n = min(len(k1), len(k2))
    sovpalo = sum(1 for i in range(n) if _rasstoyanie(k1[i], k2[i]) <= 8)
    return sovpalo >= max(3, int(n * 0.6))

def ubrat_dubli_video(html, kod=""):
    """Уровень 1 — SHA256, уровень 2 — сравнение кадров. Файлы не удаляются:
    повтор просто не показывается, решение пишется в video_dubli.json."""
    bloki = list(re.finditer(r"<video\b[^>]*>.*?</video>|<video\b[^>]*/?>", html, re.S))
    if len(bloki) < 2:
        return html, 0
    zhurnal = _json_fajl(DUBLI)
    ostavleny, ubrano, rez, kon = [], 0, [], 0
    for m in bloki:
        src = re.search(r'src="([^"]+)"', m.group(0))
        if not src:
            continue
        put = _put_iz_src(src.group(1))
        if not os.path.exists(put):
            continue
        dubl = False
        for bylo, bylo_src in ostavleny:
            # ТОЛЬКО один и тот же файл, вставленный дважды (защита от
            # размножения при повторной публикации). Разные ролики не трогаем.
            if os.path.abspath(bylo) == os.path.abspath(put) or \
                    otpechatok_video(bylo) == otpechatok_video(put):
                dubl = True
                break
        if dubl:
            rez.append(html[kon:m.start()])
            kon = m.end()
            ubrano += 1
        else:
            ostavleny.append((put, src.group(1)))
    _zapisat_json(DUBLI, zhurnal)
    if ubrano:
        rez.append(html[kon:])
        return ("".join(rez), ubrano)
    return (html, 0)


def video_uzhe_est(kod, put_novogo):
    """Для точки загрузки видео в CRM: True → VIDEO ALREADY EXISTS."""

def oblegchit(src, max_w=900, kachestvo=82):
    """Делает лёгкую webp-копию тяжёлой картинки в /video/opt и возвращает её адрес.
    Оригиналы не трогаются. Если копия уже свежая — используется она."""
    if not src.startswith("/video/") or "/opt/" in src or "/stage/" in src:
        return src
    ishod = os.path.join(DOM, src.lstrip("/").split("?")[0])
    if not os.path.exists(ishod):
        return src
    try:
        if os.path.getsize(ishod) < 70000:
            return src               # и так лёгкая
    except Exception:
        return src
    imya = re.sub(r"[^A-Za-z0-9_.-]", "_", src.lstrip("/"))
    imya = os.path.splitext(imya)[0] + ".webp"
    cel = os.path.join(OPT, imya)
    try:
        if os.path.exists(cel) and os.path.getmtime(cel) >= os.path.getmtime(ishod):
            return "/video/opt/%s" % imya
    except Exception:
        pass
    try:
        os.makedirs(OPT)
    except Exception:
        pass
    try:
        from PIL import Image
        with Image.open(ishod) as im:
            im = im.convert("RGB")
            if im.size[0] > max_w:
                k = max_w / float(im.size[0])
                im = im.resize((max_w, max(int(im.size[1] * k), 1)), Image.LANCZOS)
            im.save(cel, quality=kachestvo, method=5)
    except Exception:
        return src
    try:
        if os.path.getsize(cel) >= os.path.getsize(ishod):
            return src               # выигрыша нет
    except Exception:
        pass
    return "/video/opt/%s" % imya

def kanonicheskiy_put(kod, papka=None):

def versiya_kartochki(kod):
    put = kanonicheskiy_put(kod)
    try:
        with io.open(put, "rb") as f:
            return hashlib.sha256(f.read()).hexdigest()[:8]
    except Exception:
        return ""

ARHIV_DUBLEY = os.path.join(DOM, "archive", "video_dubli")

def video_fajly_mashiny(kod):
    spisok = []
    for koren, _, fajly in os.walk(VIDEO):
        if "/preview" in koren or "/archive" in koren or "/stage" in koren:
            continue
        for f in sorted(fajly):
            if f.startswith(".") or not f.lower().endswith((".mp4", ".mov", ".m4v", ".webm")):

def svedeniya_video(put):
    d = {"put": put, "razmer": 0, "dlitelnost": 0.0, "razreshenie": "?", "sha": ""}
    try:
        d["razmer"] = os.path.getsize(put)
    except Exception:
        pass
    d["sha"] = otpechatok_video(put) or ""
    d["dlitelnost"] = round(_dlitelnost(put), 2)
    try:
        r = subprocess.run(["ffprobe", "-v", "error", "-select_streams", "v:0",
                            "-show_entries", "stream=width,height",
                            "-of", "csv=p=0:s=x", put], capture_output=True, timeout=25)
        d["razreshenie"] = (r.stdout or b"").decode().strip() or "?"
    except Exception:
        pass
    return d

def ubrat_dubli_v_istochnike(kod):
    """Содержательные дубли ФАЙЛОВ машины уезжают в archive/video_dubli.
    Оригинал не уничтожается, генератор их больше не видит — значит и cikl2 тоже."""
    fajly = video_fajly_mashiny(kod)
    otchet = {"bylo": [svedeniya_video(f) for f in fajly], "ubrano": [], "shozhest": []}
    if len(fajly) < 2:
        otchet["stalo"] = len(fajly)
        return otchet
    try:
        os.makedirs(ARHIV_DUBLEY)
    except Exception:
        pass
    ostavleny = []
    for put in sorted(fajly, key=lambda f: -os.path.getsize(f)):
        dubl_dlya = None
        for bylo in ostavleny:
            if sodejatelno_odno(bylo, put):
                dubl_dlya = bylo
                break
        if dubl_dlya and False:
            k1, k2 = otpechatki_kadrov(dubl_dlya), otpechatki_kadrov(put)
            n = min(len(k1), len(k2))
            sovpalo = sum(1 for i in range(n) if _rasstoyanie(k1[i], k2[i]) <= 8)
            otchet["shozhest"].append("%s ≈ %s: совпало кадров %d из %d"
                                      % (os.path.basename(dubl_dlya),
                                         os.path.basename(put), sovpalo, n))
            cel = os.path.join(ARHIV_DUBLEY, os.path.basename(put))
            try:
                shutil.move(put, cel)
                otchet["ubrano"].append("%s → archive/video_dubli/" % os.path.basename(put))
            except Exception as e:
                otchet["ubrano"].append("не удалось убрать %s: %s" % (put, e))
        else:
            ostavleny.append(put)
    otchet["stalo"] = len(video_fajly_mashiny(kod))
    return otchet

def unikalnye_video(puti):
    """Сколько содержательно РАЗНЫХ роликов в списке файлов."""
    gruppy = []
    for put in puti:
        nashli = False
        for g in gruppy:
            if sodejatelno_odno(g[0], put):
                g.append(put)
                nashli = True
                break
        if not nashli:
            gruppy.append([put])
    return gruppy

def svesti_k_unikalnym(kod):
    """ХУК ДЕДУПА: вызывается публикацией ДО сборки HTML. Копии одного и того же
    материала уводятся в archive/video_dubli, разные ролики остаются."""
    fajly = video_fajly_mashiny(kod)
    if len(fajly) < 2:
        return {"fajlov": len(fajly), "unikalnyh": len(fajly), "ubrano": []}
    gruppy = unikalnye_video(fajly)
    ubrano = []
    arhiv = os.path.join(DOM, "archive", "video_dubli")
    for g in gruppy:
        for lishniy in g[1:]:
            try:
                os.makedirs(arhiv)
            except Exception:
                pass
            try:
                shutil.move(lishniy, os.path.join(arhiv, os.path.basename(lishniy)))
                ubrano.append(os.path.basename(lishniy))
            except Exception:
                pass
    ostalos = video_fajly_mashiny(kod)
    return {"fajlov": len(ostalos), "unikalnyh": len(unikalnye_video(ostalos)),
            "ubrano": ubrano}

    mesta = [os.path.join(DOM, "uploads"), os.path.join(DOM, "archive")]

MARKERY_STARYE = ("mcf-ch-full", "mcf-ch-short")
_OTKR = ("<", "&lt;", "\\u003c", "\u003c", "&#60;", "&#x3c;", "&#x3C;")
_ZAKR = (">", "&gt;", "\\u003e", "\u003e", "&#62;", "&#x3e;", "&#x3E;")

def _granica_vlevo(html, poz):
    luchshe, dlina = -1, 0
    for o in _OTKR:
        i = html.rfind(o, max(0, poz - 400), poz)
        if i > luchshe:
            luchshe, dlina = i, len(o)
    return luchshe

def _granica_vpravo(html, poz):
    luchshe, dlina = len(html), 0
    for z in _ZAKR:
        i = html.find(z, poz, poz + 400)
        if i != -1 and i < luchshe:
            luchshe, dlina = i, len(z)
    return luchshe + dlina if dlina else -1

def fragment(html, vokrug="Чат online", shirina=130):
    """Кусок страницы вокруг надписи кнопки — чтобы видеть реальный вид."""
    poz = html.find(vokrug)
    if poz < 0:
        for mk in MARKERY_STARYE:
            poz = html.find(mk)
            if poz >= 0:
                break
    if poz < 0:
        return ""
    return html[max(0, poz - shirina):poz + shirina].replace("\n", " ")

def ochistit_staroe(html):
    """Вырезает вставки прежней версии по МАРКЕРУ, а не по форме записи:
    границы тега ищутся во всех видах (< , &lt; , \\u003c и т. п.),
    поэтому работает и в обычном HTML, и в экранированном, и внутри JS-строки."""
    if not any(mk in html for mk in MARKERY_STARYE):
        return html
    for _ in range(60):
        poz = -1
        for mk in MARKERY_STARYE:
            i = html.find(mk)
            if i >= 0 and (poz < 0 or i < poz):
                poz = i
        if poz < 0:
            break
        a = _granica_vlevo(html, poz)
        b = _granica_vpravo(html, poz)
        if a < 0 or b <= a:
            html = html.replace(html[poz:poz + len("mcf-ch-full")], "", 1)
            continue
        kusok = html[a:b]
        if (b - a) > 140 or "<a " in kusok or "href=" in kusok:
            # опасно: можно снести саму ссылку — убираем только сам маркер
            html = html[:poz] + html[poz + len("mcf-ch-full"):] \
                if html[poz:poz + len("mcf-ch-full")] == "mcf-ch-full" \
                else html[:poz] + html[poz + len("mcf-ch-short"):]
            continue
        html = html[:a] + html[b:]
    # осиротевшие закрывающие теги рядом с надписью кнопки
    poz = html.find("Чат online")
    if poz >= 0:
        n, k = max(0, poz - 400), min(len(html), poz + 400)
        okno = html[n:k]
        for zakr in ("&lt;/span&gt;", "\\u003c/span\\u003e", "\u003c/span\u003e",
                     "&#60;/span&#62;", "</span>"):
            okno = okno.replace(zakr, "")
        okno = okno.replace("Чат onlineЧат", "Чат online").replace("Чат online Чат", "Чат online")
        html = html[:n] + okno + html[k:]
    return html

def obrabotat_diagnostiku(html, kod):
    """Страница UA-XXXX-diag.html: если в CRM есть отчёт (diag_link/diag_text),
    а страница показывает заглушку — материалы выводятся клиенту."""
    if not html or "mcf-diag-block" in html:
        return html
    m = dannye(kod)
    link = ("%s" % (m.get("diag_link") or "")).strip()
    tekst = ("%s" % (m.get("diag_text") or "")).strip()
    if not link and not tekst:
        return html
    chasti = ['<div class="mcf-diag-block" style="' + _O + '">',
              '<div style="' + _Z + '">Результаты комплексной диагностики</div>']
    if link:
        chasti.append(
            '<div style="margin:10px 0"><a href="%s" target="_blank" rel="noopener" '
            'style="display:inline-block;padding:11px 16px;border-radius:12px;'
            'background:rgba(212,175,55,.20);border:1px solid rgba(212,175,55,.55);'
            'color:#f4e3ae;font-weight:800;text-decoration:none">'
            'Открыть отчёт компьютерной диагностики →</a></div>' % link)
    if tekst:
        bezopasno = (tekst.replace("&", "&amp;").replace("<", "&lt;")
                     .replace(">", "&gt;"))
        chasti.append('<div style="white-space:pre-line;margin-top:8px">%s</div>' % bezopasno)
    chasti.append("</div>")
    blok = "".join(chasti)
    for yakor in ("Материалы комплексной диагностики пока не добавлены.",
                  "Материалы диагностики пока не добавлены",
                  "Данные диагностики пока не добавлены."):
        if yakor in html:
            return html.replace(yakor, blok, 1)
    if "</body>" in html:
        return html.replace("</body>", blok + "</body>", 1)
    return html + blok

def proverit(html, kod=""):
    """КРИТИЧНОЕ — из-за этого публиковать нельзя. Пустой список = можно."""
    bedy = []
    if not html or len(html) < 3000:
        bedy.append("страница пустая или слишком короткая")
        return bedy
    if ("mcf-ch-full" in html) or ("mcf-ch-short" in html):
        bedy.append("в кнопке чата остались служебные теги прежней версии")
    syroy = re.search(r"&lt;(span|div|a)\b[^&]{0,40}class", html)
    if syroy:
        nach = max(0, syroy.start() - 30)
        bedy.append("сырой markup в тексте: …%s…"
                    % html[nach:syroy.end() + 30].replace("\n", " "))
    n = html.count('class="mcf-chat')
    if n != 1:
        bedy.append("виджетов чата на странице: %d (нужен ровно 1)" % n)
    if "Комплексная диагностика" not in html:
        bedy.append("пропал раздел «Комплексная диагностика»")
    if "UA ART" not in html:
        bedy.append("пропала шапка UA ART")
    if kod:
        m = dannye(kod)
        if stage_po_statusu(m.get("status") or m.get("stage")) == "kiev" and SHAG_KIEV_STARYY in html:
            bedy.append("для машины в Киеве осталась старая формулировка этапа")
    return bedy

def _vstavit_pered_diagnostikoy(html, blok):
    poz = html.find("Комплексная диагностика")
    if poz < 0:
        poz = html.find("Купить авто")
    if poz < 0:
        return html, False
    nach = max(html.rfind("<section", 0, poz), html.rfind("<div", 0, poz),
               html.rfind("<h2", 0, poz), html.rfind("<h3", 0, poz))
    if nach < 0:
        return html, False
    return (html[:nach] + blok + html[nach:], True)

def obrabotat_kartochku(html, kod):
    """ЕДИНОЕ мастер-правило. Работает при каждой записи любой карточки."""
    if not html:
        return html
    html = ochistit_staroe(html)
    # раннего выхода нет: каждый шаг сам проверяет, нужен ли он,
    # иначе после очистки чата его было бы некому вставить
    m = dannye(kod)
    stage = stage_po_statusu(m.get("status") or m.get("stage"))

    if stage == "kiev":
        html = html.replace(SHAG_KIEV_STARYY, SHAG_KIEV_NOVYY)

    nomer = (m.get("sea_container") or "").strip()
    if nomer and "mcf-track" not in html:
        liniya, url = tracking(nomer)
        if url:
            knopka = (
                '<div class="mcf-track" style="margin:6px 0 10px">'
                '<a href="%s" target="_blank" rel="noopener" '
                'style="display:inline-block;padding:9px 14px;border-radius:10px;'
                'background:rgba(212,175,55,.18);border:1px solid rgba(212,175,55,.55);'
                'color:#f4e3ae;font-weight:700;text-decoration:none">'
                'Отследить контейнер онлайн →</a>'
                '<span style="opacity:.7;font-size:12px;margin-left:8px">%s</span></div>'
                % (url, liniya))
            novyy = re.sub(r"Контейнер:\s*<b>[^<]{4,24}</b>\s*(<br\s*/?>)?",
                           lambda mm: mm.group(0) + knopka, html, count=1)
            html = novyy

    if "mcf-etap" not in html:
        blok = TEKSTY.get(stage)
        if blok:
            html, _ = _vstavit_pered_diagnostikoy(html, blok)

    if "mcf-diag-cta" not in html:
        poz = html.find("Комплексная диагностика")
        if poz >= 0:
            konec = html.find("</", poz)
            konec = html.find(">", konec) + 1 if konec > 0 else poz
            html = html[:konec] + cta_diagnostiki(kod, m) + html[konec:]

    html, _ = obespechit_chat(html)
    html, _ = ubrat_dubli_video(html, kod)

    glav, prichina = vybrat_glavnoe(kod, m)
    sdelano = [0]

    def _foto(mm):
        if sdelano[0]:
            return mm.group(0)
        sdelano[0] = 1
        return _pravka_tega_foto(mm.group(0), kod, glav, vitrina=False)

    html = re.sub(r'<img[^>]*/foto/%s/[^>]*>' % kod, _foto, html)
    # метка ставится ровно одна: повторная сборка не должна менять страницу
    html = re.sub(r"<!-- MCF-[^>]*-->\s*", "", html)
    html = html.replace("</body>", "<!-- %s / %s / %s / stage=%s -->\n</body>"
                        % (METKA, glav, prichina, stage or "?"), 1)
    return html

STARYE_PODPISI = [
    "После отправки в Украину — фиксированные 15 дней",
    "После відправлення в Україну — фіксовані 15 днів",
    "фиксированные 15 дней",
]
PODPIS_GRUZIYA = "Отправки каждую неделю · ориентировочно около 10 дней до Киева"

def obrabotat_obshuyu(html):
    if not html:
        return html
    html = ochistit_staroe(html)
    for staraya in STARYE_PODPISI:
        html = html.replace(staraya, PODPIS_GRUZIYA)

    def _zhivoe(mm):
        teg = mm.group(0)
        src = re.search(r'src=["\']([^"\']*/(?:foto|stage)/(UA-\d{4})[^"\']*)["\']', teg)
        if not src:
            return teg
        kod = src.group(2)
        gotovyy, _ = podgotovit_vitrinu(kod)
        if gotovyy:
            teg = teg.replace(src.group(1), gotovyy)
        stil = stil_fokusa(kod, "", vitrina=True, vazhno=True)
        m2 = re.search(r'style=["\']([^"\']*)["\']', teg)
        if m2:
            staryy = re.sub(r"object-(fit|position)\s*:[^;]*;?", "", m2.group(1))
            teg = teg.replace(m2.group(0), 'style="%s;%s"' % (staryy.rstrip(";"), stil), 1)
        else:
            teg = teg[:-1].rstrip("/") + ' style="%s">' % stil
        return teg

    html = re.sub(r'<img[^>]*/(?:foto|stage)/UA-\d{4}[^>]*>', _zhivoe, html)

    def _oblegchit_teg(mm):
        teg = mm.group(0)
        src = re.search(r'src="(/video/[^"]+\.(?:jpg|jpeg|png))"', teg)
        if not src:
            return teg
        shirina = 1200 if "hero" in src.group(1) else 900
        novyy = oblegchit(src.group(1), shirina)
        return teg.replace(src.group(1), novyy) if novyy != src.group(1) else teg

    html = re.sub(r"<img[^>]*>", _oblegchit_teg, html)

    def _oblegchit_fon(mm):
        adres = mm.group(1).strip("'\"")
        novyy = oblegchit(adres, 900)
        return mm.group(0).replace(adres, novyy) if novyy != adres else mm.group(0)

    html = re.sub(r"background-image\s*:\s*url\(([^)]+)\)", _oblegchit_fon, html)
    # тяжёлые webp-исходники в <source srcset> тоже облегчаем
    def _istochnik(mm):
        adres = mm.group(1)
        novyy = oblegchit(adres, 900)
        return mm.group(0).replace(adres, novyy) if novyy != adres else mm.group(0)

    html = re.sub(r'srcset="(/video/[^"]+\.(?:jpg|jpeg|png))"', _istochnik, html)

    def _otlozhit(mm):
        teg = mm.group(0)
        if "loading=" in teg:
            return teg
        return teg[:-1].rstrip("/") + ' loading="lazy" decoding="async">'

    html = re.sub(r"<img[^>]*>", _otlozhit, html)
    html, _ = obespechit_chat(html)
    return html

def zastavka_rolika(src):
    put = (src or "").split("?")[0]
    imya = os.path.basename(put)
    if not imya.lower().endswith((".mp4", ".mov", ".m4v")):
        return ""
    osnova = os.path.splitext(imya)[0]
    papka = os.path.dirname(put)
    baza = os.path.join(VIDEO, papka.lstrip("/")) if papka else VIDEO
    if os.path.exists(os.path.join(baza, osnova + ".poster.jpg")):
        return (papka + "/" if papka else "") + osnova + ".poster.jpg"
    return ""

def rasstavit_zastavki(html):
    def _zamena(m):
        blok = m.group(0)
        src = re.search(r"<source[^>]*src=['\"]([^'\"]+)['\"]", blok)
        if not src:
            # ролик может быть задан прямо в теге video, без вложенного source
            src = re.search(r"<video[^>]*\ssrc=['\"]([^'\"]+)['\"]", blok)
        if not src:
            return blok
        svoya = zastavka_rolika(src.group(1))
        if svoya:
            if re.search(r"poster=['\"][^'\"]*['\"]", blok):
                blok = re.sub(r"poster=['\"][^'\"]*['\"]", "poster='%s'" % svoya, blok, count=1)
            else:
                blok = blok.replace("<video", "<video poster='%s'" % svoya, 1)
        else:
            blok = re.sub(r"\s*poster=['\"]\s*['\"]", "", blok)
        for atr in ("autoplay", "muted", "playsinline"):
            if atr not in blok.split(">")[0]:
                blok = blok.replace("<video", "<video %s" % atr, 1)
        return blok
    return re.sub(r"<video\b.*?</video>", _zamena, html, flags=re.S)

def video_bez_dublej(html):
    bedy = []
    bloki = re.findall(r"<video\b.*?</video>", html, re.S)
    srcy, postery = [], []
    for b in bloki:
        shapka = b.split(">")[0]
        s = re.search(r"<source[^>]*src=['\"]([^'\"]+)['\"]", b) or \
            re.search(r"<video[^>]*\ssrc=['\"]([^'\"]+)['\"]", b)
        po = re.search(r"poster=['\"]([^'\"]*)['\"]", b)
        if s:
            srcy.append(s.group(1))
            fajl = os.path.join(VIDEO, s.group(1).split("?")[0].lstrip("/"))
            if not os.path.exists(fajl):
                bedy.append("VIDEO VALIDATION FAIL — нет файла %s" % s.group(1))
            elif os.path.getsize(fajl) == 0:
                bedy.append("VIDEO VALIDATION FAIL — пустой файл %s" % s.group(1))
        if po is not None:
            postery.append(po.group(1))
            if not po.group(1).strip():
                bedy.append("VIDEO VALIDATION FAIL — пустой poster (чёрный экран)")
            else:
                fp = os.path.join(VIDEO, po.group(1).split("?")[0].lstrip("/"))
                if not os.path.exists(fp):
                    bedy.append("VIDEO VALIDATION FAIL — заставки нет на диске %s" % po.group(1))
        if "autoplay" in shapka and "muted" not in shapka:
            bedy.append("VIDEO VALIDATION FAIL — autoplay без muted")
        if bloki and "playsinline" not in shapka:
            bedy.append("VIDEO VALIDATION FAIL — нет playsinline")
    if len(srcy) != len(set(srcy)):
        bedy.append("VIDEO DUPLICATE DETECTED — одинаковый src у двух роликов")
    if len(srcy) > 1 and len(postery) > 1 and len(set(postery)) == 1:
        bedy.append("VIDEO DUPLICATE DETECTED — одинаковая заставка у разных роликов")
    return bedy

# Плитка каталога и превью заполняются целиком: пропорции сохраняются,
# кадрируется только лишний фон, положение подбирается по области автомобиля.
PHOTOFIT_VERSION = "UA-PHOTOFIT-V1"

PHOTOFIT_CSS = """<style id="mcf-photofit-css">
/* Фото в плитках и превью заполняют контейнер, не деформируясь */
.plitka img,.karta img,.card img,.preview img,.tile img,
a[href*="UA-"] > img,figure.foto img{
object-fit:cover!important;width:100%!important;height:100%!important;
display:block!important;background:transparent!important}
</style>"""

def _fokus_procenty(kod, imya):
    try:
        put = os.path.join(VIDEO, "foto", kod, imya)
        if os.path.exists(put):
            x, y = fokus_fajla(put)
            return x, y
    except Exception:
        pass
    return 50, 50

def zapolnit_foto(html):
    """Каждому фото машины в плитке — заполнение контейнера и положение по
    автомобилю. Кадры этапов (/video/stage) не трогаются: они под замком."""
    def _teg(m):
        teg = m.group(0)
        if "/stage/" in teg or "data-mcf-fit" in teg:
            return teg
        src = re.search(r'src=["\']([^"\']*/foto/(UA-\d{4})/([^"\']+))["\']', teg)
        if not src:
            return teg
        x, y = _fokus_procenty(src.group(2), src.group(3))
        stil = ("object-fit:cover!important;object-position:%d%% %d%%!important;"
                "width:100%%!important;height:100%%!important" % (x, y))
        m2 = re.search(r'style=["\']([^"\']*)["\']', teg)
        if m2:
            staryy = re.sub(r"object-(fit|position)\s*:[^;]*;?", "", m2.group(1))
            teg = teg.replace(m2.group(0), 'style="%s;%s" data-mcf-fit="1"'
                              % (staryy.rstrip(";"), stil), 1)
        else:
            teg = teg[:-1].rstrip("/") + ' style="%s" data-mcf-fit="1">' % stil
        return teg
    html = re.sub(r"<img[^>]*>", _teg, html)
    if 'id="mcf-photofit-css"' not in html:
        html = html.replace("</head>", PHOTOFIT_CSS + "</head>", 1) \
            if "</head>" in html else PHOTOFIT_CSS + html
    return html

def _ua_master_stage_span(source):
    start_count = source.count(_UA_STAGE_START)

def _ua083_dedup_catalog(source):
    pattern = _ua083_re.compile(
        _ua083_re.escape(_UA083_CAT_START) + r".*?" +
        _ua083_re.escape(_UA083_CAT_END), _ua083_re.S,
    )
    matches = list(pattern.finditer(source))
    if not matches:
        return source
    output = []
    cursor = 0
    previous_end = 0
    for match in matches:
        output.append(source[cursor:match.start()])
        upper = source[previous_end:match.start()]
        output.append(_ua083_dedup_block(match.group(0), upper))
        cursor = match.end()
        previous_end = match.end()
    output.append(source[cursor:])
    return "".join(output)

def _ua068_card_errors(source, kod, row):
    errors = []
    if _ua068_re.search(
            r"carhistory(?:\.kr)?|проверить\s+vin|перевірити\s+vin|2\s*200\s*krw",
            source, _ua068_re.I):
        errors.append("VIN advertisement forbidden")
    if source.count(_UA068_VIN_START) != 1 or source.count(_UA068_VIN_END) != 1:
        errors.append("VIN Guard blocks != 1")
    errors.extend(_ua068_primary_action_errors(source, kod, row))
    if source.count(_UA068_STAGE_START) != 1 or source.count(_UA068_STAGE_END) != 1:
        errors.append("stage anchors != 1")
    if source.count(_UA068_DIAG) != 1:
        errors.append("diagnostics anchors != 1")
    diag_links = _ua068_re.findall(
        r'href=["\'](?:[^"\']*/)?' + _ua068_re.escape(str(kod)) + r'-diag\.html(?:\?[^"\']*)?["\']',
        source, _ua068_re.I)
    if len(diag_links) != 1:
        errors.append("diagnostics links != 1")
    if _ua068_forbidden_count(source):
        errors.append("forbidden sea wording remains")
    if _ua068_re.search(
            r'class=["\'][^"\']*\b(?:mcf-etap|mcf-track|mcf-diag-off)\b[^"\']*["\']',
            source, _ua068_re.I):
        errors.append("legacy duplicate UI remains")
    if _ua068_native_stage_span(source) is not None:
        errors.append("native stage duplicate remains")
    vin = str((row or {}).get("vin") or "").strip().upper()
    if not vin or vin not in source:
        errors.append("VIN missing")
    engine = _ua068_engine(row)
    if engine and not _ua068_re.search(_ua068_re.escape(engine) + r"\s*см³", source):
        errors.append("engine mismatch")
    stage = _ua068_stage(row)
    if 'data-ua-stage="%d"' % stage not in source:
        errors.append("stage copy mismatch")
    if stage == 4 and ("Выдача и осмотр — Киев" in source or "Видача та огляд — Київ" in source):
        errors.append("redundant Kyiv issue/inspection copy remains")
    expected_videos = _ua068_video_count(kod, row, source)
    if 'data-ua-video-count="%d"' % expected_videos not in source:
        errors.append("video count mismatch")
    return errors
