def sklonenie(n):
    if n % 10 == 1 and n % 100 != 11:
        return "день"
    if n % 10 in (2, 3, 4) and n % 100 not in (12, 13, 14):
        return "дня"
    return "дней"

def ostatok_dney(m):
    """Сколько дней до Киева. Ручная дата важнее расчёта по этапу."""
    eta = (m.get("eta_manual") or "").strip()
    if eta:
        for obr in ("%Y-%m-%d", "%d.%m.%Y", "%Y-%m-%dT%H:%M:%S"):
            try:
                cel = time.mktime(time.strptime(eta[:19], obr))
                return max(int((cel - time.time()) // 86400), 0)
            except Exception:
                continue
    return OSTATOK_PO_ETAPU.get(nomer_etapa(m), 0)


def data_vydachi(dney):
    t = time.localtime(time.time() + dney * 86400)
    return "%d %s %d" % (t.tm_mday, MESYACY_V115[t.tm_mon - 1], t.tm_year)

def srok(m):
    eta = (m.get("eta_manual") or "").strip()
    if eta:
        for obr in ("%Y-%m-%d", "%d.%m.%Y", "%Y-%m-%dT%H:%M:%S"):
            try:
                cel = time.mktime(time.strptime(eta[:19], obr))
                ostalos = int((cel - time.time()) // 86400)
                if ostalos > 0:
                    return "До прибытия в Киев — %d %s" % (ostalos, sklonenie(ostalos))
                return "Прибытие в Киев ожидается со дня на день"

                continue
    s = (m.get("status") or "").lower()
    if s.startswith("ua_"):
        return "Машина в Киеве — можно приехать и посмотреть"
    if s == "ge_to_kyiv":
        try:
            pamyat = json.loads(chitat(os.path.join(BAZA_DIR, ".etapy.json"), 200000))
            kogda = (pamyat.get(nomer(m)) or {}).get("s_kogda")
        except Exception:
            kogda = None
        if kogda:
            ostalos = SROK_DNEY - int((time.time() - kogda) // 86400)
            if ostalos > 0:
                return "До прибытия в Киев — %d %s" % (ostalos, sklonenie(ostalos))
            return "Прибытие в Киев ожидается со дня на день"
    d = ostatok_dney(m)  # UA-V115
    return "До Киева — %d %s" % (d, sklonenie(d))

def razmer_jpeg(put):
    try:
        with open(put, "rb") as f:
            d = f.read(200000)
    except Exception:
        return (0, 0)
    i = 2
    while i < len(d) - 9:
        if d[i] != 0xFF:
            i += 1
            continue
        marker = d[i + 1]
        if marker in (0xC0, 0xC1, 0xC2, 0xC3, 0xC5, 0xC6, 0xC7, 0xC9, 0xCA, 0xCB, 0xCD, 0xCE, 0xCF):
            return ((d[i + 7] << 8) + d[i + 8], (d[i + 5] << 8) + d[i + 6])
        if marker in (0xD8, 0xD9) or 0xD0 <= marker <= 0xD7:
            i += 2
            continue
        i += 2 + (d[i + 2] << 8) + d[i + 3]
    return (0, 0)


SREDNIJ = 1000   # для ленты в карточке и главной плитки каталога
MALYJ   = 440    # для мини-ленты и обложки главной


def variant(otn, papka):
    """foto/UA-0001/003.jpg → foto/UA-0001/m/003.jpg"""
    chasti = otn.rsplit("/", 1)
    return "%s/%s/%s" % (chasti[0], papka, chasti[1])


def umenshit(otn, papka, shirina, kachestvo):
    """Делает уменьшенную копию кадра. Возвращает путь для страницы."""
    ishod = os.path.join(PAPKA_VID, otn)
    cel_otn = variant(otn, papka)
    cel = os.path.join(PAPKA_VID, cel_otn)
    try:
        if (os.path.exists(cel) and os.path.getsize(cel) > 1000
                and os.path.getmtime(cel) >= os.path.getmtime(ishod)):
            return cel_otn
    except Exception:
        pass
    try:
        from PIL import Image
    except Exception:
        return otn
    try:
        im = Image.open(ishod)
        if im.mode not in ("RGB", "L"):
            im = im.convert("RGB")
        if im.width > shirina:
            im = im.resize((shirina, max(int(im.height * shirina / im.width), 1)),
                           Image.LANCZOS)
        os.makedirs(os.path.dirname(cel), exist_ok=True)
        im.save(cel, "JPEG", quality=kachestvo, optimize=True, progressive=True)
        return cel_otn
    except Exception:
        return otn

def dostat_iz_stranic(nom, papka):
    """Запасной источник: кадры, вшитые в ранее собранные страницы."""
    kandidaty = []
    for shablon in (os.path.join(PAPKA_VID, nom + "*.html"),
                    os.path.join(PAPKA_SITE, nom + "*.html"),
                    os.path.join(BAZA_DIR, "staroe", "**", nom + "*.html")):
        kandidaty += glob.glob(shablon, recursive=True)
    kadry = []
    for put in kandidaty:
        try:
            najdeno = re.findall(
                r"data:image/(?:jpeg|jpg|png);base64,([A-Za-z0-9+/=]{2000,})", chitat(put))
        except Exception:
            continue
        if len(najdeno) > len(kadry):
            kadry = najdeno
    for i, kusok in enumerate(kadry, 1):
        put = os.path.join(papka, "%03d.jpg" % i)
        try:
            if not os.path.exists(put) or os.path.getsize(put) < 1000:
                with open(put, "wb") as f:
                    f.write(base64.b64decode(kusok + "=" * (-len(kusok) % 4)))
        except Exception:
            continue
    return len(kadry)


def kadry_mashiny(m):
    nom = nomer(m)
    papka = os.path.join(PAPKA_FOTO, nom)
    os.makedirs(papka, exist_ok=True)
    est = sorted(glob.glob(os.path.join(papka, "*.jpg")))
    if not est:
        dostat_iz_stranic(nom, papka)
        est = sorted(glob.glob(os.path.join(papka, "*.jpg")))
    puti = ["foto/%s/%s" % (nom, os.path.basename(x)) for x in est
            if os.path.getsize(x) > 1000]
    from photo_visibility import visible_names
    visible = set(visible_names(m, [os.path.basename(p) for p in puti]))
    puti = [p for p in puti if os.path.basename(p) in visible]
    # первым — первый горизонтальный кадр
    if not m.get("cover_photo"):
        for otn in puti:
            sh, vy = razmer_jpeg(os.path.join(PAPKA_VID, otn))
            if sh and vy and sh / float(vy) >= 1.2:
                if puti[0] != otn:
                    puti = [otn] + [x for x in puti if x != otn]
                break
    return puti

def _ua_stage_number(m):
    status = str(m.get("status") or "").strip().lower()
    if status.startswith("kr_"):
        return 1
    if status.startswith("sea_"):
        return 2
    if status == "ge_to_kyiv" or status.startswith("ge_"):
        return 3
    if status.startswith("ua_"):
        return 4
    try:
        saved = int(m.get("stage") or 0)
        if 1 <= saved <= 4:
            return saved
    except Exception:
        pass
    return 1


def _ua_stage_date(value):
    text = str(value or "").strip()
    if not text:
        return None
    for pattern, width in (("%Y-%m-%d", 10), ("%d.%m.%Y", 10), ("%d/%m/%Y", 10)):
        try:
            return _ua_stage_dt.datetime.strptime(text[:width], pattern).date()
        except Exception:
            pass
    return None

    zone = _ua_stage_dt.timezone(_ua_stage_dt.timedelta(hours=3))
    return _ua_stage_dt.datetime.now(zone).date()


def _ua_stage_pretty(value):
    return "%d %s %d" % (value.day, _UA_STAGE_MONTHS[value.month - 1], value.year)


def _ua_stage_plural(value):
    value = abs(int(value))
    if value % 10 == 1 and value % 100 != 11:
        return "день"
    if value % 10 in (2, 3, 4) and value % 100 not in (12, 13, 14):
        return "дня"
    return "дней"

def _ua_stage_eta(m, stage):
    status = str(m.get("status") or "").strip().lower()
    if stage >= 4 or status.startswith("ua_"):
        return ("arrived", None, None)
    manual = _ua_stage_date(m.get("eta_manual"))
    if manual is not None:
        target = manual
    elif status == "ge_to_kyiv":
        released = _ua_stage_saved_release(m)
        target = released + _ua_stage_dt.timedelta(days=15) if released else None
    elif status.startswith("ge_"):
        target = None
    else:
        shipped = _ua_stage_date(m.get("sea_date_out"))
        target = shipped + _ua_stage_dt.timedelta(days=75) if shipped else None
    if target is None:
        return ("georgia" if status.startswith("ge_") else "unknown", None, None)
    days = max((target - _ua_stage_today()).days, 0)
    return ("due" if days else "soon", days, target)

def zapisat(imya_bez, soderzhimoe):
    """Пишем во все копии: обе папки, все адреса с хэшем."""
    # UA-SHCHIT-V3: общие страницы сайта ведёт только новый сборщик
    try:
        _sh_imya = os.path.basename(str(imya_bez)).strip().lower()
        if _sh_imya.replace(".html", "") in ("index", "katalog", "podbor", "info"):
            try:
                with io.open("/home/Carix/shchit_log.txt", "a", encoding="utf-8") as _sh_zh:
                    _sh_zh.write("%s  запрет zapisat: %s\n"
                                 % (time.strftime("%Y-%m-%d %H:%M:%S"), imya_bez))
            except Exception:
                pass
            return True
    except Exception:
        pass
    # >>> UA-ART-SITE-GE-PERSISTENCE-V1: STRANICA_WRITE
    from site_ge_inject import inject_html as _ua_site_ge_inject_html
    soderzhimoe = _ua_site_ge_inject_html(soderzhimoe)
    # <<< UA-ART-SITE-GE-PERSISTENCE-V1: STRANICA_WRITE
    celi = []
    for papka in (PAPKA_VID, PAPKA_SITE):
        os.makedirs(papka, exist_ok=True)
        celi.append(os.path.join(papka, imya_bez + ".html"))
        for _variant in glob.glob(os.path.join(papka, imya_bez + "-*.html")):
            # UA-DIAG-FILENAME-GUARD-V1: запись карточки не имеет права
            # перезаписывать её отдельную страницу диагностики.
            _base = os.path.basename(_variant)
            if (str(imya_bez).startswith("UA-")
                    and not str(imya_bez).endswith("-diag")
                    and _base.startswith(str(imya_bez) + "-diag")):
                continue
            celi.append(_variant)
    for put in set(celi):
        try:
            with io.open(put, "w", encoding="utf-8") as f:
                f.write(soderzhimoe)
        except Exception:
            continue
    return len(set(celi))


def otklyuchit_staroe():
    """Глушим прежний фоновый сборщик — иначе он вернёт старую версию страницы."""
    ubrano = []
    # 1. сам файл модуля превращаем в пустышку, копию оставляем рядом
    for imya in ("karta_iz_crm.py",):
        put = os.path.join(BAZA_DIR, imya)
        if not os.path.exists(put):
            continue
        try:
            soderzhimoe = chitat(put)
            if soderzhimoe.strip() == "pass":
                continue
            with io.open(put + ".vyklyuchen", "w", encoding="utf-8") as f:
                f.write(soderzhimoe)
            with io.open(put, "w", encoding="utf-8") as f:
                f.write("pass\n")
            ubrano.append(imya)
        except Exception:
            continue
    for ver in ("3.10", "3.13"):
        uc = os.path.expanduser("~/.local/lib/python%s/site-packages/usercustomize.py" % ver)
        if not os.path.exists(uc):
            continue
        try:
            tekst = chitat(uc)
        except Exception:
            continue
        stroki, tronuli = [], False
        for s in tekst.split("\n"):
            if s.strip() and not s.strip().startswith("#") and "karta_iz_crm" in s:
                stroki.append(" " * (len(s) - len(s.lstrip())) + "pass  # старый сборщик выключен")
                tronuli = True
            else:
                stroki.append(s)
        if not tronuli:
            continue
        novyj = "\n".join(stroki)
        try:
            compile(novyj, uc, "exec")
        except SyntaxError:
            continue
        try:
            with io.open(uc + ".proshlyy", "w", encoding="utf-8") as f:
                f.write(tekst)
            with io.open(uc, "w", encoding="utf-8") as f:
                f.write(novyj)
            ubrano.append(ver)
        except Exception:
            continue
    return ubrano

        return 0

def _v169_dney_do(metka):
    """Разница в календарных сутках по киевской дате. Меньше нуля не бывает."""
    seychas = time.strftime("%Y-%m-%d",
                            time.localtime(time.time() + V169_SDVIG_KIEV))
    cel = time.strftime("%Y-%m-%d", time.localtime(metka))
    try:
        a = time.mktime(time.strptime(seychas, "%Y-%m-%d"))
        b = time.mktime(time.strptime(cel, "%Y-%m-%d"))
    except Exception:
        return 0
    return max(int(round((b - a) / 86400.0)), 0)

def _v174_proverit(imya, novaya, staraya):
    """Возвращает список нарушений структуры. Пусто — можно публиковать."""
    opis = _v174_opisanie()
    if not opis.get("CARD_TEMPLATE_LOCKED"):
        return []
    bedy = []
    poryadok = opis.get("poryadok") or []
    mesta = []
    for punkt in poryadok:
        nazvanie = punkt.get("razdel")
        obrazec = punkt.get("obrazec")
        if not obrazec:
            continue
        gde = novaya.find(obrazec)
        if gde < 0:
            bedy.append("пропал раздел «%s»" % nazvanie)
        else:
            mesta.append((nazvanie, gde))
    if not bedy:
        proshloe = -1
        for nazvanie, gde in mesta:
            if gde < proshloe:
                bedy.append("нарушен порядок разделов на «%s»" % nazvanie)
                break
            proshloe = gde
    if staraya:
        bylo = staraya.count("<img")
        stalo = novaya.count("<img")
        if bylo >= 5 and stalo < bylo * 0.8:
            bedy.append("фотографий было %d, стало %d" % (bylo, stalo))
    return bedy

def _ua083_dedup_catalog(source):
    pattern = _ua083_re.compile(
        _ua083_re.escape(_UA083_CAT_START) + r".*?" +
        _ua083_re.escape(_UA083_CAT_END), _ua083_re.S,
    )
    matches = list(pattern.finditer(source))
    if not matches:
        return source
    output = []
    cursor = 0
    previous_end = 0
    for match in matches:
        output.append(source[cursor:match.start()])
        upper = source[previous_end:match.start()]
        output.append(_ua083_dedup_block(match.group(0), upper))
        cursor = match.end()
        previous_end = match.end()
    output.append(source[cursor:])
    return "".join(output)

    source = _ua068_replace_marked(source, _UA068_CAT_START, _UA068_CAT_END)
    # Catalogs currently exist in two layouts: modern <article> cards and
    # legacy standalone <a> tiles.  A page may also contain both layouts
    # (or a second link to one car).  Insert exactly one canonical block per
    # identifier instead of choosing one page-wide parser branch.
    seen = set()

    def render_once(match):
        block = match.group(0)
        found = _ua068_re.search(r"UA-[0-9]{4,}", block, _ua068_re.I)
        if not found:
            return _ua068_terms(block)
        identifier = found.group(0).upper()

            return _ua068_terms(block)
        rendered = _ua068_catalog_item(block, rows)
        if 'data-ua-card="%s"' % identifier in rendered:
            seen.add(identifier)
        return rendered

    article = _ua068_re.compile(
        r'<article\b[^>]*>.*?</article\s*>', _ua068_re.I | _ua068_re.S)
    source = article.sub(render_once, source)

        r'<a\b(?=[^>]*href=["\'][^"\']*UA-[0-9]{4,}\.html(?:\?[^"\']*)?["\'])[^>]*>.*?</a\s*>',
        _ua068_re.I | _ua068_re.S)

    missing = [identifier for identifier in rows if identifier not in seen]

        (r"На пароме\s*·\s*\d+", "На пароме · %d" % counts[2]),
        (r"На поромі\s*·\s*\d+", "На поромі · %d" % counts[2]),
        (r"В Корее\s*·\s*\d+", "В Корее · %d" % counts[1]),

def _ua068_card_errors(source, kod, row):
    errors = []
    if _ua068_re.search(
            r"carhistory(?:\.kr)?|проверить\s+vin|перевірити\s+vin|2\s*200\s*krw",
            source, _ua068_re.I):
        errors.append("VIN advertisement forbidden")
    if source.count(_UA068_VIN_START) != 1 or source.count(_UA068_VIN_END) != 1:
        errors.append("VIN Guard blocks != 1")
    errors.extend(_ua068_primary_action_errors(source, kod, row))
    if source.count(_UA068_STAGE_START) != 1 or source.count(_UA068_STAGE_END) != 1:
        errors.append("stage anchors != 1")
    if source.count(_UA068_DIAG) != 1:
        errors.append("diagnostics anchors != 1")
    diag_links = _ua068_re.findall(
        r'href=["\'](?:[^"\']*/)?' + _ua068_re.escape(str(kod)) + r'-diag\.html(?:\?[^"\']*)?["\']',
        source, _ua068_re.I)
    if len(diag_links) != 1:
        errors.append("diagnostics links != 1")
    if _ua068_forbidden_count(source):
        errors.append("forbidden sea wording remains")
    if _ua068_re.search(
            r'class=["\'][^"\']*\b(?:mcf-etap|mcf-track|mcf-diag-off)\b[^"\']*["\']',
            source, _ua068_re.I):
        errors.append("legacy duplicate UI remains")
    if _ua068_native_stage_span(source) is not None:
        errors.append("native stage duplicate remains")
    vin = str((row or {}).get("vin") or "").strip().upper()
    if not vin or vin not in source:
        errors.append("VIN missing")
    engine = _ua068_engine(row)
    if engine and not _ua068_re.search(_ua068_re.escape(engine) + r"\s*см³", source):
        errors.append("engine mismatch")
    stage = _ua068_stage(row)
    if 'data-ua-stage="%d"' % stage not in source:
        errors.append("stage copy mismatch")
    if stage == 4 and ("Выдача и осмотр — Киев" in source or "Видача та огляд — Київ" in source):
        errors.append("redundant Kyiv issue/inspection copy remains")
    expected_videos = _ua068_video_count(kod, row, source)
    if 'data-ua-video-count="%d"' % expected_videos not in source:
        errors.append("video count mismatch")
    return errors
