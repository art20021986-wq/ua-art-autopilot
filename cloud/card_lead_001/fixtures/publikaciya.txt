def _sha(put):
    try:
        h = hashlib.sha256()
        with open(put, "rb") as f:
            for k in iter(lambda: f.read(65536), b""):
                h.update(k)
        return h.hexdigest()
    except Exception:
        return "-"


def _sha_teksta(t):
    if isinstance(t, str):
        t = t.encode("utf-8")
    return hashlib.sha256(t).hexdigest()


def _chitat(put):
    try:
        with io.open(put, "r", encoding="utf-8", errors="ignore") as f:
            return f.read()
    except Exception:
        return ""


def _zhurnal(stroki):
    try:
        with io.open(LOG, "a", encoding="utf-8") as f:
            for s in stroki:
                f.write("%s  %s\n" % (time.strftime("%Y-%m-%d %H:%M:%S"), s))
    except Exception:
        pass


def _kartochki(krome=None):
    """sha всех карточек и диагностик, кроме указанной машины."""
    d = {}
    for papka in (VIDEO, SITE):
        if not os.path.isdir(papka):
            continue
        for imya in os.listdir(papka):
            if not (imya.startswith("UA-") and imya.endswith(".html")):
                continue
            if krome and imya.startswith(krome):
                continue
            d[os.path.join(papka, imya)] = _sha(os.path.join(papka, imya))
    return d

def proverit(html, kod):
    """Возвращает список причин отказа. Пустой список = страница годная."""
    prichiny = []
    if not html or len(html) < 5000:
        prichiny.append("страница короче 5 КБ")
        return prichiny
    n64 = html.count("data:image")
    if n64 > 2:
        prichiny.append("вшитые base64-фотографии: %d — чужой шаблон" % n64)
    if len(html) > 600000:
        prichiny.append("страница тяжелее 600 КБ (%d байт)" % len(html))
    if kod not in html:
        prichiny.append("нет номера машины %s" % kod)
    chuzhie = sorted(set(re.findall(r"UA-\d{4}", html)) - set([kod]))
    if chuzhie:
        prichiny.append("данные другой машины: %s" % ", ".join(chuzhie))
    if "</html>" not in html:
        prichiny.append("страница оборвана, нет </html>")
    if not re.search(r"\d[\d\s\u00a0]{2,}\s*\$", html):
        prichiny.append("нет цены")
    if "иагностик" not in html:
        prichiny.append("нет раздела диагностики")
    if (kod + "-diag.html") not in html:
        prichiny.append("нет ссылки на страницу диагностики")
    if "упить" not in html:
        prichiny.append("нет кнопки Купить")
    if "оделиться" not in html:
        prichiny.append("нет кнопки Поделиться")
    if "index.html" not in html:
        prichiny.append("нет навигации на главную")
    if "wa.me" not in html:
        prichiny.append("нет контакта WhatsApp")
    if "<img" not in html:
        prichiny.append("нет фотографий")
    if "aspect-ratio:16/10" not in html.replace(" ", ""):
        prichiny.append("правило показа фото не совпадает с эталоном 16/10")
    return prichiny


def _master(kod):
    """Собирает карточку и диагностику мастер-генератором. Возвращает (html, diag, mashina)."""
    import stranica as S
    spisok = S.mashiny()
    m = None
    for x in spisok:
        try:
            if (S.nomer(x) or "").upper() == kod.upper():
                m = x
                break
        except Exception:
            continue
    if m is None:
        return None, None, None
    kadry = S.kadry_mashiny(m)
    html = S.sobrat_kartochku(m, kadry)
    diag = None
    try:
        if S.est_diagnostika(m):
            diag = S.sobrat_diagnostiku(m)
    except Exception:
        diag = None
    if isinstance(html, bytes):
        html = html.decode("utf-8", "ignore")
    if isinstance(diag, bytes):
        diag = diag.decode("utf-8", "ignore")
    return html, diag, m

def opublikovat(kod, proba=False):
    """Единственная разрешённая публикация карточки. (ok, текст для владельца)"""
    kod = (kod or "").strip().upper()
    if not re.match(r"^UA-\d{3,5}$", kod):
        return False, "Публикация отменена: непонятный номер машины."

    do = _kartochki(krome=kod)          # чужие карточки до работы
    try:
        html, diag, m = _master(kod)
    except Exception as e:
        _zhurnal(["%s FAIL сборка мастером: %s" % (kod, e)])
        return False, "Публикация отменена: сборщик не смог собрать %s (%s). Старая страница цела." % (kod, e)
    if not html:
        _zhurnal(["%s FAIL машина не найдена в CRM" % kod])
        return False, "Публикация отменена: %s не найдена среди машин CRM." % kod

    prichiny = proverit(html, kod)
    if prichiny:
        _zhurnal(["%s FAIL валидация: %s" % (kod, "; ".join(prichiny))])
        return False, ("Публикация отменена, страница не прошла проверку:\n· %s\n"
                       "Прежняя страница на сайте не тронута." % "\n· ".join(prichiny))

    if proba:
        return True, "Проба: страница %s собрана и проверена, ничего не записано." % kod

    papka_rez = os.path.join(REZERV_KORE, "%s_%s" % (kod, time.strftime("%Y%m%d_%H%M%S")))
    celi = []
    for papka in (VIDEO, SITE):
        if not os.path.isdir(papka):
            continue
        celi.append(os.path.join(papka, kod + ".html"))
        for imya in os.listdir(papka):
            if re.match(r"^%s-[0-9a-f]{6,10}\.html$" % kod, imya):
                celi.append(os.path.join(papka, imya))
        if diag:
            celi.append(os.path.join(papka, kod + "-diag.html"))
    try:
        os.makedirs(papka_rez, exist_ok=True)
        for put in celi:
            if os.path.exists(put):
                shutil.copy2(put, os.path.join(
                    papka_rez, put.replace(BASE + "/", "").replace("/", "__")))
    except Exception as e:
        _zhurnal(["%s FAIL резерв: %s" % (kod, e)])
        return False, "Публикация отменена: не удалось сделать резерв (%s)." % e

    bylo = dict((put, _sha(put)) for put in celi)
    try:
        for put in celi:
            _zapisat_atomarno(put, diag if put.endswith("-diag.html") else html)
    except Exception as e:
        _otkat(papka_rez, celi)
        _zhurnal(["%s FAIL запись: %s — выполнен откат" % (kod, e)])
        return False, "Публикация отменена при записи (%s). Выполнен откат." % e

    posle = _kartochki(krome=kod)
    izmenilis = [k for k in do if do.get(k) != posle.get(k)]
    if izmenilis:
        _otkat(papka_rez, celi)
        _zhurnal(["%s FAIL регрессия: изменились %s — откат" % (kod, ", ".join(izmenilis))])
        return False, ("Публикация отменена: изменились чужие карточки (%s). "
                       "Выполнен откат." % ", ".join(os.path.basename(x) for x in izmenilis))

    stalo = dict((put, _sha(put)) for put in celi)
    _zhurnal(["%s PASS публикация · мастер stranica.sobrat_kartochku · файлов %d"
              % (kod, len(celi))]
             + ["    %s  %s -> %s" % (os.path.basename(k), (bylo.get(k) or "-")[:12],
                                      (stalo.get(k) or "-")[:12]) for k in celi])
    hvost = "" if diag else "\nМатериалов диагностики в CRM нет — страница диагностики не трогалась."
    return True, ("Страница %s обновлена на сайте по утверждённому шаблону.%s\n"
                  "https://www.uaart.com.ua/video/%s.html?v=%d"
                  % (kod, hvost, kod, int(time.time())))


def _otkat(papka_rez, celi):
    for put in celi:
        kopiya = os.path.join(papka_rez, put.replace(BASE + "/", "").replace("/", "__"))
        try:
            if os.path.exists(kopiya):
                shutil.copy2(kopiya, put)
        except Exception:
            pass
